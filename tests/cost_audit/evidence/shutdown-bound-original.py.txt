"""Isolated negative check of the synthetic tool, never a product test."""
import importlib.util
import json
from pathlib import Path
import subprocess
import sys
import threading
import time
from unittest.mock import patch

PROBE = Path('/tmp/issue118-cost-audit-20261006/tests/cost_audit/shutdown_poll_probe.py')
OUT = Path('/tmp/issue118-wait-review/stuck-shutdown-receipt.json')


def child():
    spec = importlib.util.spec_from_file_location('synthetic_shutdown_probe', PROBE)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    # The synthetic cleanup wake can make BaseServer attempt its deliberately
    # absent request handler. Suppress only that daemon-thread diagnostic.
    threading.excepthook = lambda args: None
    never = threading.Event()
    started = time.perf_counter()
    try:
        with patch.object(module.socketserver.BaseServer, 'shutdown', lambda self: never.wait()):
            module.measure(.01)
    except RuntimeError as error:
        elapsed = time.perf_counter() - started
        if str(error) != 'synthetic shutdown exceeded its bound':
            raise
        if elapsed >= 8:
            raise AssertionError('injected stuck shutdown was not bounded')
        print(json.dumps({'result': 'bounded rejection', 'error': str(error),
                          'elapsed_seconds': elapsed,
                          'normal_probe_runs': 0, 'product_suite_runs': 0}))
        return 0
    raise AssertionError('stuck shutdown unexpectedly returned a successful measurement')


if __name__ == '__main__':
    if '--child' in sys.argv:
        sys.exit(child())
    started = time.perf_counter()
    proc = subprocess.run([sys.executable, __file__, '--child'], capture_output=True, text=True, timeout=9)
    record = {'command': [sys.executable, __file__, '--child'], 'outer_deadline_seconds': 9,
              'shell_exit': proc.returncode, 'outer_elapsed_seconds': time.perf_counter() - started,
              'stdout': proc.stdout, 'stderr': proc.stderr}
    OUT.write_text(json.dumps(record, indent=2) + '\n')
    print(json.dumps(record))
    sys.exit(proc.returncode)
