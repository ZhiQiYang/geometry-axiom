import hashlib,json,os,platform,resource,subprocess,sys,tarfile,time
from pathlib import Path
R=Path('/workspace/geometry-axiom');O=Path('/tmp/pf3-native-final-v2-20261015');O.mkdir(exist_ok=False);os.chdir(R)
def save(name,v):(O/name).write_text(json.dumps(v,ensure_ascii=False,indent=2)+'\n')
def git(*args):return subprocess.check_output(['git',*args]).decode().strip()
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
tracked=[os.fsdecode(p) for p in subprocess.check_output(['git','ls-files','-z']).split(b'\0') if p]
def snapshot():return {p:sha(R/p) for p in tracked if (R/p).is_file()}
before=snapshot();head=git('rev-parse','HEAD');assert not git('status','--porcelain')
save('tracked-before.json',before);save('expected-before.json',json.loads((R/'tests/run_all.expected.json').read_text()))
resource.setrlimit(resource.RLIMIT_STACK,(16777216,resource.getrlimit(resource.RLIMIT_STACK)[1]))
env={**os.environ,'MUTATE_CACHE':'/tmp/pf3-native-empty-cache-v2'};assert not Path(env['MUTATE_CACHE']).exists()
argv=[sys.executable,'-u','tests/run_all.py','--build'];started_ns=time.time_ns();start=time.monotonic()
save('started.json',dict(head=head,argv=argv,startedUnixNs=started_ns,stackBytes=16777216,mutationCache=env['MUTATE_CACHE']))
with (O/'stdout').open('wb') as so,(O/'stderr').open('wb') as se:p=subprocess.run(argv,cwd=R,env=env,stdout=so,stderr=se)
after=snapshot();save('tracked-after.json',after);save('expected-after.json',json.loads((R/'tests/run_all.expected.json').read_text()))
artifacts={};members=[]
with tarfile.open(O/'binaries.tar.gz','w:gz') as archive:
    for name in ['bridge','core','integrity','model-base','engineering-json-test','pf3-algorithm']:
        path=Path('/tmp/geometry-tests')/name;side=path.with_name(path.name+'.fingerprint.json')
        artifacts[name]=dict(sha256=sha(path),sidecar=json.loads(side.read_text()));archive.add(path,arcname=name);archive.add(side,arcname=side.name)
save('binary-final.json',artifacts)
logs=Path('/tmp/geometry-tests/run_all-logs')
with tarfile.open(O/'logs.tar.gz','w:gz') as archive:
    for f in sorted(logs.rglob('*')):
        if f.is_file() and f.stat().st_mtime_ns>=started_ns:
            name=str(f.relative_to(logs));members.append(dict(path=name,bytes=f.stat().st_size,sha256=sha(f)));archive.add(f,arcname=name)
save('log-manifest.json',members)
receipt=dict(headBefore=head,headAfter=git('rev-parse','HEAD'),argv=argv,exit=p.returncode,seconds=time.monotonic()-start,
             trackedRegularFiles=len(before),trackedExcluded=[],trackedBeforeAfterEqual=before==after,statusAfter=git('status','--porcelain'),
             platform=platform.platform(),python=platform.python_version(),node=subprocess.check_output(['node','--version']).decode().strip(),
             gcc=subprocess.check_output(['gcc','--version']).decode().splitlines()[0],stackBytes=16777216,mutationCache=env['MUTATE_CACHE'],
             mutationCacheExistedBefore=False,currentRunLogMembers=len(members),logArchiveSha256=sha(O/'logs.tar.gz'),binaryArchiveSha256=sha(O/'binaries.tar.gz'))
save('receipt.json',receipt);print(json.dumps(receipt),flush=True)
raise SystemExit(p.returncode)
