def copy(src,target):
    target.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(src,target)
def archive(path,files):
    manifest=[]
    with tarfile.open(path,'w:gz') as t:
        for src,name in files:
            t.add(src,arcname=name,recursive=False);manifest.append(dict(path=name,bytes=src.stat().st_size,sha256=hashlib.sha256(src.read_bytes()).hexdigest()))
    path.with_name(path.name+'.inventory.json').write_text(json.dumps(manifest,indent=2)+'\n')
