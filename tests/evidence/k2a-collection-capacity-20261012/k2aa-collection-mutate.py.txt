from pathlib import Path
import shutil,subprocess,json,hashlib
root=Path('/workspace/geometry-k2aa-review');out=Path('/tmp/k2aa-collection-mutants');out.mkdir(exist_ok=True)
private=out/'private';assert not private.exists()
(private/'tests').mkdir(parents=True)
shutil.copytree(root/'geometry-web',private/'geometry-web',ignore=shutil.ignore_patterns('node_modules','__pycache__','feedback','*.exe'))
for name in ['test_grid_drafts_lifecycle_capacity.js','test_reference_workspace_races.js']:shutil.copy2(root/'tests'/name,private/'tests'/name)
for name in ['evidence','fixtures']:(private/'tests'/name).symlink_to(root/'tests'/name,target_is_directory=True)
file=private/'geometry-web/feature-attachments.js';source=file.read_text()
anchor='try{boundedJSON({schema:SCHEMA,records:[{...record,schema:value.schema,value}]});}'
cases=[('force-collapse-byte-boundary','B1 collection projection','tests/test_grid_drafts_lifecycle_capacity.js'),('force-collapse-writer-save','K2a B1 collection two','tests/test_reference_workspace_races.js'),('force-collapse-old-reader','B1 previous collection','tests/test_grid_drafts_lifecycle_capacity.js'),('force-collapse-old-import','K2a B1 previous collection','tests/test_reference_workspace_races.js')]
def run(command,path):
 p=subprocess.run(command,cwd=private,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True);path.write_text(p.stdout);return p
results=[]
for ident,pattern,test in cases:
 folder=out/ident;folder.mkdir();cmd=['node','--test','--test-name-pattern='+pattern,test]
 assert source.count(anchor)==1
 baseline=run(cmd,folder/'baseline.log');assert baseline.returncode==0,baseline.stdout
 try:
  file.write_text(source.replace(anchor,'try{/* incorrectly force collection into the smaller single quota */}'))
  build=run(['node','--check',str(file)],folder/'build.log');assert build.returncode==0
  red=run(cmd,folder/'red.log');assert red.returncode==1 and 'AssertionError [ERR_ASSERTION]' in red.stdout and 'ℹ skipped 0' in red.stdout and '✖ '+pattern in red.stdout,red.stdout
  if 'byte-boundary' not in ident:assert '容量' in red.stdout,red.stdout
 finally:file.write_text(source)
 restored=run(cmd,folder/'restored.log');assert restored.returncode==0,restored.stdout
 results.append({'id':ident,'source':'geometry-web/feature-attachments.js','sourceSHA256':hashlib.sha256(source.encode()).hexdigest(),'before':anchor,'after':'try{/* incorrectly force collection into the smaller single quota */}','command':cmd,'baseline':baseline.returncode,'compile':build.returncode,'red':red.returncode,'restored':restored.returncode,'designatedAssertion':True,'skipped':0})
 print(ident,'baseline 0 / build 0 / assertion red 1 / restored 0',flush=True)
(out/'results.json').write_text(json.dumps(results,ensure_ascii=False,indent=2)+'\n')
