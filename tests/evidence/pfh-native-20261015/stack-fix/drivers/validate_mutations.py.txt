import hashlib,json,os,shutil,subprocess,sys
from pathlib import Path
R=Path('/tmp/geometry-209-segments');O=Path('/tmp/geometry-wake41/official-stack-final');U=O/'unified-cache';U.mkdir(exist_ok=True)
sys.path.insert(0,str(R/'tests'));import mutate
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
records={};receipts=[];audit=[];baseline_hash=None
for group in range(3):
 part=O/str(group);receipt=json.loads((part/'receipt.json').read_text());assert receipt['exit']==0;receipts.append(receipt)
 selected=mutate.load_records(part/'results.json');assert set(selected)==set(receipt['ids']);assert not set(records)&set(selected);records.update(selected)
 for mid,record in selected.items():
  if record['verdict']=='retired':continue
  execution=record['execution'];source=part/'cache'/mid/'_mut.exe';base=part/'cache'/'_baseline'/'baseline.exe'
  assert sha(source)==execution['exe_sha256'] and sha(base)==execution['baseline_exe_sha256']
  if baseline_hash is None:
   baseline_hash=sha(base);target=U/'_baseline';target.mkdir(exist_ok=True)
   for p in [base,base.with_name('baseline.exe.fingerprint.json')]:shutil.copy2(p,target/p.name)
  assert sha(base)==baseline_hash,'different executed baseline bytes; cannot coalesce artifact namespaces'
  target=U/mid;target.mkdir(exist_ok=True)
  for p in [source,source.with_name('_mut.exe.fingerprint.json')]:shutil.copy2(p,target/p.name)
  audit.append(dict(id=mid,original=str(source),copied=str(target/'_mut.exe'),exeSha256=sha(source),copySha256=sha(target/'_mut.exe'),baselineSha256=sha(base)))
assert len(records)==50 and len(audit)==47
mutate.save_records(R/'tests/mutants_results.json',records)
argv=[sys.executable,'tests/mutate.py','--check','--killers']
with (O/'check.stdout').open('wb') as so,(O/'check.stderr').open('wb') as se:p=subprocess.run(argv,cwd=R,env={**os.environ,'MUTATE_CACHE':str(U)},stdout=so,stderr=se)
summary=dict(head=receipts[0]['head'],receipts=receipts,mergedRecords=len(records),actualExecutedArtifactAudit=audit,unifiedCache=str(U),baselineSha256=baseline_hash,checkArgv=argv,checkExit=p.returncode)
(O/'summary.json').write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps(dict(records=len(records),executed=len(audit),checkExit=p.returncode,baselineSha256=baseline_hash)))
raise SystemExit(p.returncode)
