from concurrent.futures import ThreadPoolExecutor
import json,os,resource,subprocess,sys,time
from pathlib import Path
ROOT=Path('/tmp/geometry-209-segments');OUT=Path('/tmp/geometry-wake41/official-stack-final');OUT.mkdir(exist_ok=True)
sys.path.insert(0,str(ROOT/'tests'));import mutate
resource.setrlimit(resource.RLIMIT_STACK,(16777216,resource.getrlimit(resource.RLIMIT_STACK)[1]))
ids=[m['id'] for m in mutate.MUTANTS];assert len(set(ids))==len(ids)==50
head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip()
groups=[ids[i::3] for i in range(3)]
def run(i):
    out=OUT/str(i);out.mkdir(exist_ok=True);cache=out/'cache';assert not cache.exists()
    argv=[sys.executable,'-u','tests/mutate.py',*groups[i]];started=time.monotonic()
    env={**os.environ,'MUTATE_CACHE':str(cache),'MUTATE_RESULTS':str(out/'results.json')}
    with (out/'stdout').open('wb') as so,(out/'stderr').open('wb') as se:p=subprocess.run(argv,cwd=ROOT,env=env,stdout=so,stderr=se)
    receipt=dict(head=head,argv=argv,ids=groups[i],exit=p.returncode,seconds=time.monotonic()-started,stackBytes=16777216,freshCache=str(cache))
    (out/'receipt.json').write_text(json.dumps(receipt,indent=2)+'\n');print(json.dumps(receipt),flush=True);return receipt
with ThreadPoolExecutor(max_workers=3) as pool:receipts=list(pool.map(run,range(3)))
assert all(r['exit']==0 for r in receipts),receipts
merged={}
for i,expected in enumerate(groups):
    records=mutate.load_records(OUT/str(i)/'results.json');assert set(records)==set(expected)
    assert not set(merged)&set(records);merged.update(records)
assert set(merged)==set(ids)
mutate.save_records(ROOT/'tests/mutants_results.json',merged)
argv=[sys.executable,'tests/mutate.py','--check','--killers']
with (OUT/'check.stdout').open('wb') as so,(OUT/'check.stderr').open('wb') as se:p=subprocess.run(argv,cwd=ROOT,stdout=so,stderr=se)
summary=dict(head=head,partition=groups,receipts=receipts,mergedRecords=len(merged),checkArgv=argv,checkExit=p.returncode)
(OUT/'summary.json').write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps(summary),flush=True)
raise SystemExit(p.returncode)
