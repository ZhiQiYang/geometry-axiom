import hashlib,json,tarfile
from pathlib import Path
R=Path('/tmp/geometry-209-segments');E=R/'tests/evidence/pfh-native-20261015'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
manifest=json.loads((E/'final-manifest.json').read_text())
assert set(manifest)=={str(p.relative_to(E)) for p in E.rglob('*') if p.is_file() and p.name!='final-manifest.json'}
for name,row in manifest.items():
 p=E/name;assert p.stat().st_size==row['bytes'] and sha(p)==row['sha256'],name
for archive_name,manifest_name in [('native/native-current-logs.tar.gz','native/native-log-manifest.json'),('official-components/executed-sidecars.tar.gz','official-components/sidecar-manifest.json')]:
 rows=json.loads((E/manifest_name).read_text())
 with tarfile.open(E/archive_name) as archive:
  assert set(archive.getnames())=={r['path'] for r in rows}
  for row in rows:assert hashlib.sha256(archive.extractfile(row['path']).read()).hexdigest()==row['sha256']
r=json.loads((E/'native/native.json').read_text());assert sha(E/'native/native-current-logs.tar.gz')==r['archiveSha256']
with tarfile.open(E/'native/tracked-snapshots.tar.gz') as archive:
 before=json.load(archive.extractfile('tracked-before.json'));after=json.load(archive.extractfile('tracked-after.json'));assert before==after and len(before)==r['trackedRegularFiles']
audit=json.loads((E/'components-checkpoint/source-audit.json').read_text())
for p,row in audit['sources'].items():assert sha(R/p)==row['sha256'],p
for p,digest in audit['protectedSha256'].items():assert sha(R/p)==digest,p
assert json.loads((E/'native/expected-before.json').read_text())==json.loads((E/'native/expected-after.json').read_text())
print(json.dumps(dict(manifestFiles=len(manifest),nativeLogMembers=r['currentRunLogMembers'],sidecarMembers=48,trackedFiles=len(before),productSourcesUnchanged=True,protectedUnchanged=True),ensure_ascii=False))
