R=Path('/workspace/geometry-axiom');E=R/'tests/evidence/pf3-p134-20261015';E.mkdir(parents=True,exist_ok=False)
def copy(src,name):target=E/name;target.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(src,target)

    manifest=[]

    (E/(prefix+'.inventory.json')).write_text(json.dumps(manifest,indent=2)+'\n')
