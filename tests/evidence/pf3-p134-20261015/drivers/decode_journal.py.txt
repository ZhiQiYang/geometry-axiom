"""Reassemble raw native journal from GitHub comment JSONs, without extraction.

Fetch each linked part with gh api repos/ZhiQiYang/geometry-axiom/issues/comments/ID > part.json
python decode_journal.py.txt output.tar.gz part-1.json part-2.json ...
"""
import base64,hashlib,json,pathlib,re,sys
parts={};expected=None;count=None
for name in sys.argv[2:]:
    body=json.loads(pathlib.Path(name).read_text())['body']
    match=re.search(r'P134_NATIVE_ARCHIVE_PART (\d+)/(\d+) SHA256=([0-9a-f]{64})',body)
    block=re.search(r'```base64\n([A-Za-z0-9+/=\n]+)\n```',body)
    assert match and block,name
    index,total=int(match[1]),int(match[2])
    assert index not in parts and (expected is None or expected==match[3]) and (count is None or count==total)
    expected,count=match[3],total;parts[index]=''.join(block[1].split())
assert set(parts)==set(range(1,count+1))
raw=base64.b64decode(''.join(parts[i] for i in range(1,count+1)),validate=True)
assert hashlib.sha256(raw).hexdigest()==expected
pathlib.Path(sys.argv[1]).write_bytes(raw)
print(json.dumps(dict(path=sys.argv[1],sha256=expected,bytes=len(raw))))
