import hashlib,json,subprocess,sys,urllib.request
from pathlib import Path
root=Path('/tmp/geometry-control-diagnostics');out=root/'tests/evidence/control-diagnostics-integration-20261013'
mode=sys.argv[1]
def snapshot():
    paths=subprocess.check_output(['git','ls-files'],cwd=root).decode().splitlines()
    chosen=[]
    for n in paths:
        p=Path(n)
        if n=='current_geometry.c' or (n.startswith(('geometry-web/','geometry-core/')) and p.suffix in {'.c','.h','.inc','.py','.js','.html','.css'} and not any(part in {'docs','prototypes','evidence'} for part in p.parts)) or n in {'tests/run_all.py','tests/run_all.expected.json','tests/run_tests.py','tests/mutants_results.json'}:
            chosen.append(n)
    data={'head':subprocess.check_output(['git','rev-parse','HEAD'],cwd=root).decode().strip(),'files':{n:hashlib.sha256((root/n).read_bytes()).hexdigest() for n in chosen},'binaries':{}}
    for name in ['core','bridge','integrity','model-base','engineering-json-test']:
        p=Path('/tmp/control-diag-integration-binaries')/name
        data['binaries'][name]=hashlib.sha256(p.read_bytes()).hexdigest()
        side=p.with_name(p.name+'.fingerprint.json')
        (out/(name+'-fingerprint.json')).write_bytes(side.read_bytes())
    return data
if mode in {'before','after'}:
    data=snapshot()
    if mode=='after':
        assert data==json.loads((out/'identity-before.json').read_text())
    (out/('identity-'+mode+'.json')).write_text(json.dumps(data,indent=2)+'\n')
    print(mode,data['head'],len(data['files']))
else:
    for port,label in [(8974,'canvas'),(8975,'grid')]:
        raw=urllib.request.urlopen(f'http://127.0.0.1:{port}/runtime').read()
        (out/(label+'-runtime.json')).write_bytes(raw)
        r=subprocess.run(['python','geometry-web/runtime_version.py','--probe',str(port)],cwd=root,capture_output=True,text=True)
        (out/(label+'-probe.json')).write_text(json.dumps({'exit':r.returncode,'stdout':r.stdout,'stderr':r.stderr},indent=2)+'\n')
        assert r.returncode==0
        print(label,r.stdout.strip())
