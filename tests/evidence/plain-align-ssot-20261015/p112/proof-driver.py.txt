import pathlib, subprocess, re, json, hashlib, difflib
root=pathlib.Path('/tmp/geometry-framing-proposal')
out=root/'tests/evidence/plain-align-ssot-20261015/p112'
base='79412a858f712981fa0ba15a6b293c3e0bf06f21'
def original(path):
    return subprocess.check_output(['git','show',base+':'+path],cwd=root)
plain='geometry-guide/公理層-白話.md'
old=original(plain).decode(); new=(root/plain).read_text()
def sections(text):
    pieces=re.split(r'^## ([DIR]\d+[a-z]?)\s*\n',text,flags=re.M)
    return dict(zip(pieces[1::2],pieces[2::2]))
old_sections,new_sections=sections(old),sections(new)
assert old_sections.keys()==new_sections.keys()
changed=[key for key in old_sections if old_sections[key]!=new_sections[key]]
assert changed==['D1','I10','I53','R45'],changed
header_old=old.split('## D1',1)[0]; header_new=new.split('## D1',1)[0]
changes=[(header_old.splitlines()[a:b],header_new.splitlines()[c:d]) for tag,a,b,c,d in difflib.SequenceMatcher(None,header_old.splitlines(),header_new.splitlines()).get_opcodes() if tag!='equal']
assert len(changes)==1 and len(changes[0][0])==len(changes[0][1])==1
assert [s for s in old.splitlines() if '🔒' in s]==[s for s in new.splitlines() if '🔒' in s]
path='geometry-guide/公理層-閱讀器.html'
before=original(path).decode(); after=(root/path).read_text()
def data(text,name):
    return json.loads(re.search(r'<script\b[^>]*id="'+name+r'"[^>]*>(.*?)</script>',text,re.S)[1])
old_clauses={r['id']:r for r in data(before,'clause-data')}
new_clauses={r['id']:r for r in data(after,'clause-data')}
assert old_clauses.keys()==new_clauses.keys()
assert all(r['body']==new_clauses[key]['body'] for key,r in old_clauses.items())
locked=[key for key,row in old_clauses.items() if '🔒' in row['body']]
assert all(old_clauses[key]==new_clauses[key] for key in locked)
old_state,new_state=data(before,'axiom-review-data'),data(after,'axiom-review-data')
assert old_state['comments']==new_state['comments']
raw_locked=[(len([s for s in before.splitlines()[a:b] if '🔒' in s]),len([s for s in after.splitlines()[c:d] if '🔒' in s])) for tag,a,b,c,d in difflib.SequenceMatcher(None,before.splitlines(),after.splitlines()).get_opcodes() if tag!='equal']
assert sum(a for a,b in raw_locked)==sum(b for a,b in raw_locked)==2
protected=['公理層.md','BACKLOG.md','決策佇列.md','current_geometry.c','tests/mutants_results.json','tests/run_all.py','tests/run_all.expected.json','geometry-guide/sync_axiom_reader.py']
assert all(original(p)==(root/p).read_bytes() for p in protected)
proof=dict(base=base,changedPlainSections=changed,headerChange=changes,all204ClauseBodiesIdentical=True,lockedClauseIds=locked,lockedClauseObjectsIdentical=True,humanLockedLinesUnchanged=True,rawGeneratedLockedLinesChanged=2,commentsUnchanged=True,commentCount=len(old_state['comments']),canonicalProducedBySync=True,canonicalSha256=hashlib.sha256((root/path).read_bytes()).hexdigest())
(out/'scope-proof.json').write_text(json.dumps(proof,ensure_ascii=False,indent=2)+'\n')
print(json.dumps(proof,ensure_ascii=False))
