import pathlib, subprocess, json, hashlib, os, time
root=pathlib.Path('/tmp/geometry-framing-proposal')
out=root/'tests/evidence/plain-align-ssot-20261015/p112'
head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip()
paths=subprocess.check_output(['git','ls-files','-z','geometry-guide','geometry-core','geometry-web','tests','current_geometry.c','公理層.md','BACKLOG.md','決策佇列.md'],cwd=root).decode().rstrip('\0').split('\0')
paths=[p for p in paths if '/evidence/' not in p and '/release-evidence/' not in p]
def identity():
    return {p:hashlib.sha256((root/p).read_bytes()).hexdigest() for p in paths}
before=identity()
(out/'sources-before.json').write_text(json.dumps(before,ensure_ascii=False,indent=2)+'\n')
env=dict(os.environ,MUTATE_CACHE='/tmp/wait-trace-mutation-cache')
env.pop('GEOMETRY_WAIT_TRACE',None)
def run(name,command):
    start=time.monotonic()
    with (out/(name+'.stdout.log')).open('wb') as stdout,(out/(name+'.stderr.log')).open('wb') as stderr:
        p=subprocess.run(['bash','-c','ulimit -s 16384; exec '+command],cwd=root,env=env,stdout=stdout,stderr=stderr)
    receipt=dict(head=head,headAfter=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),exit=p.returncode,seconds=time.monotonic()-start)
    (out/(name+'.json')).write_text(json.dumps(receipt,indent=2)+'\n')
    return p.returncode
reader_exit=run('reader41','python3 tests/test_axiom_reader.py')
if reader_exit==0:
    run('native','python3 /tmp/plain-native-command.py')
after=identity()
(out/'sources-after.json').write_text(json.dumps(after,ensure_ascii=False,indent=2)+'\n')
(out/'gates.json').write_text(json.dumps(dict(head=head,headAfter=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),sourceCount=len(paths),sourcesEqual=before==after,readerExit=reader_exit),indent=2)+'\n')
