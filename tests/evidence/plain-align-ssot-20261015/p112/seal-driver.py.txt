import pathlib, json, hashlib, tarfile, shutil, re, subprocess
root=pathlib.Path('/tmp/geometry-framing-proposal')

logs=pathlib.Path('/tmp/plain-align-binaries/run_all-logs')
start=(out/'sources-before.json').stat().st_mtime
files=sorted(p for p in logs.rglob('*') if p.is_file() and p.stat().st_mtime>=start)
members=[]
with tarfile.open(out/'native-raw.tar.gz','w:gz') as archive:
    for p in files:
        raw=p.read_bytes(); name=str(p.relative_to(logs))
        members.append(dict(path=name,bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest()))
        archive.add(p,arcname=name)
with tarfile.open(out/'native-raw.tar.gz','r:gz') as archive:
    for row in members:
        raw=archive.extractfile(row['path']).read()
        assert len(raw)==row['bytes'] and hashlib.sha256(raw).hexdigest()==row['sha256']
(out/'native-members.json').write_text(json.dumps(members,indent=2)+'\n')
binary={}
def raw_clauses(html):
    encoded=re.search(r'<script\b[^>]*id="clause-data"[^>]*>(.*?)</script>',html,re.S)[1]
    decoder=json.JSONDecoder(); index=1; rows={}
    while True:
        while encoded[index] in ' \t\r\n,': index+=1
        if encoded[index]==']': break
        value,end=decoder.raw_decode(encoded,index)
        rows[value['id']]=encoded[index:end]
        index=end
    return rows
old_raw=raw_clauses(subprocess.check_output(['git','show','79412a85:geometry-guide/公理層-閱讀器.html'],cwd=root).decode())
new_raw=raw_clauses((root/'geometry-guide/公理層-閱讀器.html').read_text())
locked=json.loads((out/'scope-proof.json').read_text())['lockedClauseIds']
assert all(old_raw[key]==new_raw[key] for key in locked)
(out/'locked-serialized.json').write_text(json.dumps({key:hashlib.sha256(old_raw[key].encode()).hexdigest() for key in locked},indent=2)+'\n')
for p in pathlib.Path('/tmp/plain-align-binaries').iterdir():
    if p.is_file() and p.suffix!='.json':
        binary[p.name]=hashlib.sha256(p.read_bytes()).hexdigest()
    elif p.name.endswith('.fingerprint.json'):
        shutil.copyfile(p,out/p.name)
(out/'binary-final.json').write_text(json.dumps(binary,indent=2)+'\n')
