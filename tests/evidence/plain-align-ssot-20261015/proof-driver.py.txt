import pathlib, subprocess, re, json, hashlib, difflib
root=pathlib.Path('/tmp/geometry-framing-proposal')
out=root/'tests/evidence/plain-align-ssot-20261015'
base='79412a858f712981fa0ba15a6b293c3e0bf06f21'
def original(path):
    return subprocess.check_output(['git','show',base+':'+path],cwd=root)
plain='geometry-guide/公理層-白話.md'
old=original(plain).decode()
new=(root/plain).read_text()
def sections(text):
    pieces=re.split(r'^## ([DI R]\d+[a-z]?)\s*\n',text,flags=re.M)
    return dict(zip(pieces[1::2],pieces[2::2]))
old_sections,new_sections=sections(old),sections(new)
changed=[key for key in old_sections if old_sections[key]!=new_sections[key]]
assert changed==['D1','I10','I53','R45'],changed
assert [s for s in old.splitlines() if '🔒' in s]==[s for s in new.splitlines() if '🔒' in s]
html=original('geometry-guide/公理層-閱讀器.html').decode()
patch=pathlib.Path('/tmp/plain-align-reader-proposed.diff').read_text()
added='\n'.join(line[1:] for line in patch.splitlines() if line.startswith('+') and not line.startswith('+++'))
def clauses(text):
    return json.loads(re.search(r'<script\b[^>]*id="clause-data"[^>]*>(.*?)</script>',text,re.S)[1])
old_clauses={r['id']:r for r in clauses(html)}
new_clauses={r['id']:r for r in clauses(added)}
assert old_clauses.keys()==new_clauses.keys()
assert all(r['body']==new_clauses[key]['body'] for key,r in old_clauses.items())
locked=[key for key,row in old_clauses.items() if '🔒' in row['body']]
assert all(old_clauses[key]==new_clauses[key] for key in locked)
protected=['公理層.md','BACKLOG.md','決策佇列.md','current_geometry.c','tests/mutants_results.json','tests/run_all.py','tests/run_all.expected.json']
assert all(original(path)==(root/path).read_bytes() for path in protected)
proof=dict(base=base,changedPlainSections=changed,headerOnlyS8=True,all204ClauseBodiesIdentical=True,lockedClauseIds=locked,lockedClauseObjectsIdentical=True,plainLockedLinesIdentical=True,canonicalRestored=True,proposedPatchSha256=hashlib.sha256(pathlib.Path('/tmp/plain-align-reader-proposed.diff').read_bytes()).hexdigest())
(out/'scope-proof.json').write_text(json.dumps(proof,ensure_ascii=False,indent=2)+'\n')
(out/'plain-changes.json').write_text(json.dumps([dict(id=key,before=old_sections[key],after=new_sections[key]) for key in changed],ensure_ascii=False,indent=2)+'\n')
print(json.dumps(proof,ensure_ascii=False))
