import json, os, resource, subprocess, sys, time
from pathlib import Path
name=sys.argv[1];cwd=sys.argv[2];command=sys.argv[3:]
out=Path('/workspace/geometry-axiom/tests/evidence/geo261-load-residual-20261020/checks');out.mkdir(parents=True,exist_ok=True)
resource.setrlimit(resource.RLIMIT_STACK,(16777216,16777216))
meta={'name':name,'command':command,'cwd':cwd,'stack_bytes':16777216,'started':time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime())}
(out/(name+'.json')).write_text(json.dumps(meta,ensure_ascii=False,indent=2))
with (out/(name+'.stdout')).open('wb') as stdout,(out/(name+'.stderr')).open('wb') as stderr:
    r=subprocess.run(command,cwd=cwd,stdout=stdout,stderr=stderr)
meta.update(exit=r.returncode,finished=time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime()))
(out/(name+'.json')).write_text(json.dumps(meta,ensure_ascii=False,indent=2))
print(name,'exit='+str(r.returncode),flush=True)
sys.exit(r.returncode)
