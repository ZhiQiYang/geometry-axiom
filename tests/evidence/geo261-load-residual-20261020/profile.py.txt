import gzip,json,resource,subprocess
from pathlib import Path
root=Path('/workspace/geometry-axiom');out=root/'tests/evidence/geo261-load-residual-20261020';checks=out/'checks'
resource.setrlimit(resource.RLIMIT_STACK,(16777216,16777216))
for version in ['base','fix']:
    exe=Path('/tmp/geo261-residual')/(version+'-pg')/'bridge'
    for n in [32,64]:
        work=Path('/tmp/geo261-residual/profile')/f'{version}-{n}';work.mkdir(parents=True,exist_ok=True)
        stdin=f'loadfile /tmp/s{n}.scene\n'.encode()
        for step,command,inp in [('load',[str(exe)],stdin),('gprof',['gprof',str(exe),str(work/'gmon.out')],None)]:
            r=subprocess.run(command,input=inp,cwd=work,capture_output=True)
            name=f'profile-{version}-{n}-{step}'
            (checks/(name+'.stdout')).write_bytes(r.stdout);(checks/(name+'.stderr')).write_bytes(r.stderr)
            (checks/(name+'.json')).write_text(json.dumps({'command':command,'stdin':inp.decode() if inp else None,'cwd':str(work),'stack_bytes':16777216,'exit':r.returncode},indent=2))
            assert r.returncode==0,(name,r.stderr)
            if step=='load':assert json.loads(r.stdout)['ok']
            else:
                p=Path(f'/tmp/geo261-residual-base-{n}.gprof' if version=='base' else f'/tmp/geo261-residual-final-fix-{n}.gprof')
                p.write_bytes(r.stdout)
        (out/f'gmon-{version}-{n}.gz').write_bytes(gzip.compress((work/'gmon.out').read_bytes(),mtime=0))
        print(version,n,'load=0 gprof=0',flush=True)
