import hashlib
import json
import os
from pathlib import Path
import platform
import resource
import subprocess
import tarfile
import time

ROOT = Path('/workspace/geometry-axiom')
OUT = Path('/tmp/pfh-task21-20261015')
OUT.mkdir(exist_ok=True)
os.chdir(ROOT)

def save(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')

def git(*args):
    return subprocess.check_output(['git', *args]).decode().strip()

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

tracked = [os.fsdecode(p) for p in subprocess.check_output(['git', 'ls-files', '-z']).split(b'\0') if p]
def snapshot():
    return {p: sha(ROOT / p) for p in tracked if (ROOT / p).is_file()}

head = git('rev-parse', 'HEAD')
assert head == '4be9ff3453202984433c9256dff9dd60d5c8437a', head
assert not git('status', '--porcelain')
before = snapshot()
save('tracked-before.json', before)
resource.setrlimit(resource.RLIMIT_STACK, (16777216, resource.getrlimit(resource.RLIMIT_STACK)[1]))
cache = OUT / 'fresh-mutation-cache'
assert not cache.exists()
env = dict(os.environ, MUTATE_CACHE=str(cache))
started_ns = time.time_ns()
save('started.json', dict(head=head, startedUnixNs=started_ns, stackBytes=resource.getrlimit(resource.RLIMIT_STACK)[0], mutationCache=str(cache), mutationCacheExistedBefore=False))
commands = [('reader-check', ['python3', 'geometry-guide/sync_axiom_reader.py', '--check']), ('native', ['python3', '-u', 'tests/run_all.py', '--build'])]
results = {}
for label, argv in commands:
    started = time.monotonic()
    with (OUT / (label + '.stdout')).open('wb') as stdout, (OUT / (label + '.stderr')).open('wb') as stderr:
        result = subprocess.run(argv, stdout=stdout, stderr=stderr, env=env)
    results[label] = dict(argv=argv, exit=result.returncode, seconds=time.monotonic() - started)
    save(label + '.json', results[label])
    print(json.dumps(dict(label=label, **results[label])), flush=True)
after = snapshot()
save('tracked-after.json', after)
binary_dir = Path('/tmp/geometry-tests')
binary = {}
for name in ('bridge', 'core', 'integrity', 'model-base', 'engineering-json-test'):
    path = binary_dir / name
    sidecar = path.with_name(path.name + '.fingerprint.json')
    binary[name] = dict(exists=path.is_file(), sha256=sha(path) if path.is_file() else None, sidecar=json.loads(sidecar.read_text()) if sidecar.is_file() else None)
save('binary-final.json', binary)
logs = binary_dir / 'run_all-logs'
members = []
with tarfile.open(OUT / 'native-logs.tar.gz', 'w:gz') as archive:
    for path in sorted(logs.rglob('*')):
        if path.is_file() and path.stat().st_mtime_ns >= started_ns:
            name = str(path.relative_to(logs))
            members.append(dict(path=name, bytes=path.stat().st_size, sha256=sha(path)))
            archive.add(path, arcname=name)
save('native-log-manifest.json', members)
receipt = dict(headBefore=head, headAfter=git('rev-parse', 'HEAD'), commands=results,
               trackedRegularFiles=len(before), trackedExcluded=[], trackedBeforeAfterEqual=before == after,
               changedTracked=[p for p in before.keys() | after.keys() if before.get(p) != after.get(p)],
               statusAfter=git('status', '--porcelain'), platform=platform.platform(), python=platform.python_version(),
               node=subprocess.check_output(['node', '--version']).decode().strip(),
               gcc=subprocess.check_output(['gcc', '--version']).decode().splitlines()[0],
               stackBytes=resource.getrlimit(resource.RLIMIT_STACK)[0], mutationCache=str(cache), mutationCacheExistedBefore=False,
               currentRunLogMembers=len(members), archiveSha256=sha(OUT / 'native-logs.tar.gz'),
               formalWSLSeven=dict(executed=0, scheduled=False, status='unverified'))
save('verification.json', receipt)
print(json.dumps(receipt, ensure_ascii=False), flush=True)
raise SystemExit(0 if all(r['exit'] == 0 for r in results.values()) and before == after and head == receipt['headAfter'] and not receipt['statusAfter'] else 1)
