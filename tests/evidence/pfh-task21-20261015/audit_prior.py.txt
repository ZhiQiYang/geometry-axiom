import difflib
import hashlib
import json
from pathlib import Path
import subprocess
import tarfile

ROOT = Path('/workspace/geometry-axiom')
OUT = Path('/tmp/pfh-task21-20261015')

def digest(data):
    return hashlib.sha256(data).hexdigest()

def check_manifest(folder):
    manifest_path = ROOT / folder / 'final-manifest.json'
    manifest = json.loads(manifest_path.read_text())
    for name, value in manifest.items():
        data = (manifest_path.parent / name).read_bytes()
        assert len(data) == value['bytes'], name
        assert digest(data) == value['sha256'], name
    return dict(path=str(manifest_path.relative_to(ROOT)), sha256=digest(manifest_path.read_bytes()), verifiedFiles=len(manifest))

result = {'manifests': [check_manifest(p) for p in ('tests/evidence/pfh-native-20261015', 'tests/evidence/pfh-p126-20261015')]}
audit = json.loads((ROOT / 'tests/evidence/pfh-p126-20261015/source-audit.json').read_text())
for name, value in audit['unchangedProductSources'].items():
    assert digest((ROOT / name).read_bytes()) == value['sha256'], name
for name, value in audit['protectedUnchanged'].items():
    assert digest((ROOT / name).read_bytes()) == value, name
result['unchangedProductSourceFiles'] = len(audit['unchangedProductSources'])
result['unchangedProtectedFiles'] = list(audit['protectedUnchanged'])
name = 'geometry-guide/公理層-閱讀器.html'
old = subprocess.check_output(['git', 'show', '16ed25b8:' + name], cwd=ROOT).decode()
current = (ROOT / name).read_bytes()
patch = ''.join(difflib.unified_diff(old.splitlines(keepends=True), current.decode().splitlines(keepends=True), fromfile='a/' + name, tofile='b/' + name)).encode()
assert digest(patch) == audit['generatedPatchSha256']
assert digest(current) == audit['canonicalSha256']
result['generatedPatchSha256'] = digest(patch)
result['canonicalSha256'] = digest(current)
native_path = ROOT / 'tests/evidence/pfh-p126-20261015/native'
members = json.loads((native_path / 'native-log-manifest.json').read_text())
with tarfile.open(native_path / 'native-current-logs.tar.gz') as archive:
    actual_names = [m.name for m in archive.getmembers() if m.isfile()]
    assert set(actual_names) == {m['path'] for m in members}
    for item in members:
        data = archive.extractfile(item['path']).read()
        assert len(data) == item['bytes'] and digest(data) == item['sha256'], item['path']
result['previousNativeLogMembersVerified'] = len(members)
paths = subprocess.check_output(['git', 'diff', '--name-only', '-z', 'f6d128ac90ef0e028b5bf8f56e93847b14dc5c10', 'HEAD'], cwd=ROOT).decode().split('\0')
paths = [p for p in paths if p]
assert all(p.startswith('tests/evidence/pfh-p126-20261015/') or p == 'geometry-core/panel-fill/PFH-NATIVE.md' for p in paths), paths
result['deliverySincePreviousNativeOnlyEvidenceAndDocumentation'] = True
result['deliverySincePreviousNativeChangedPaths'] = paths
result['previousControlsAndMutations'] = 'retained per P126; not rerun in task21'
result['formalWSLSeven'] = {'status': 'unverified', 'executed': 0, 'scheduled': False}
(OUT / 'prior-evidence-audit.json').write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({k: v for k, v in result.items() if k != 'deliverySincePreviousNativeChangedPaths'}, ensure_ascii=False))
