#!/usr/bin/env python3
"""Pure-file archival; no compilation, tests, Git writes, or product writes."""
import collections
import datetime
import hashlib
import importlib.util
import io
import json
import os
from pathlib import Path
import re
import shutil
import stat
import sys
import tarfile

sys.dont_write_bytecode = True
S = 'b823903e44aea091b70fd2f45b74dff6229776dc'
OUT = Path('/workspace/geo323-ledger-publication-STAGING')
ROOTS = {
    'core-regenerate-e': Path('/workspace/geo323-core-regenerate-e'),
    'core-check-e': Path('/workspace/geo323-core-check-e'),
    'outer-ledger-regenerate-e': Path('/workspace/geo323-core-outer/ledger-regenerate-e'),
    'outer-ledger-check-e': Path('/workspace/geo323-core-outer/ledger-check-e'),
    'w-vocabulary-e': Path('/workspace/geo323-w-vocabulary-e'),
    'outer-w-vocabulary-e': Path('/workspace/geo323-core-outer/w-vocabulary-e'),
}
CLONE = ROOTS['core-regenerate-e'] / 'tmp/geometry-tests/mutants/_bridge-work' / S / 'repo'
MAX_RAW = 80 * 1024 * 1024
MAX_ARCHIVE = 95 * 1024 * 1024

def sha(data):
    return hashlib.sha256(data).hexdigest()

def filehash(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()

def dump(path, data):
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')

def load(path):
    return json.loads(path.read_text(encoding='utf-8'))

def main():
    if OUT.exists():
        raise RuntimeError('new staging directory required; will not overwrite')
    OUT.mkdir()
    (OUT / 'archives').mkdir()
    (OUT / 'inventories').mkdir()
    shutil.copyfile(__file__, OUT / 'archive_ledger_e.py.txt')
    ledger = load(ROOTS['core-regenerate-e'] / 'ledger.json')
    records = {k: v for k, v in ledger.items() if re.fullmatch(r'M\d+', k)}
    executed = {k: v for k, v in records.items() if v.get('execution', {}).get('head') == S}
    assert len(records) == 78 and len(executed) == 75
    assert collections.Counter(r['verdict'] for r in records.values()) == {'killed': 70, 'equivalent': 4, 'crashed': 1, 'retired': 3}
    receipts = {label: load(path / 'receipt.json') for label, path in ROOTS.items()}
    full = receipts['core-regenerate-e']
    check = receipts['core-check-e']
    for d in [full, check]:
        assert d['sha'] == S and d['ended_utc'] and not d['errors']
        assert d['source_before'] == d['source_after'] == {'head': S, 'status': ''}
    step_exits = {s['label']: s['original_command_result']['actual_process_returncode'] for s in full['steps']}
    assert full['exit_code'] == 0 and step_exits == {'precheck': 1, 'full-ledger': 0, 'checkafter': 0}
    assert check['exit_code'] == 1
    for label, code in [('outer-ledger-check-e', 1), ('outer-ledger-regenerate-e', 0), ('outer-w-vocabulary-e', 0)]:
        d = receipts[label]
        assert d['sha'] == S and d['outer_process']['actual_process_returncode'] == code
        assert not d['outer_process']['timeout_expired'] and d['outer_process']['ended_epoch']
    w = receipts['w-vocabulary-e']
    assert w['source_sha'] == S and w['source_before'] == w['source_after'] == {'head': S, 'status': ''}
    assert w['manifest_matches_full_mechanical_union'] is True
    assert all(w[k] == [] for k in ['missing_identifiers', 'unexpected_identifiers', 'missing_strings', 'unexpected_strings'])
    # Import original source-only helpers; never call main/run/compile/evaluate.
    sys.path.insert(0, str(CLONE / 'tests'))
    spec = importlib.util.spec_from_file_location('geo323_archive_original_mutate', CLONE / 'tests/mutate.py')
    mt = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = mt
    spec.loader.exec_module(mt)
    source = mt.Sources(CLONE)
    assert all(r['closure_sha256'] == source.closure for r in executed.values())
    definitions = {m['id']: m for m in mt.MUTANTS}
    inputs = []
    excluded = []
    generated = []
    tool_rows = []
    raw_checks = []
    def add_path(path, member):
        st = path.lstat()
        row = {'member': member, 'original_path': str(path), 'bytes': st.st_size, 'mode': stat.S_IMODE(st.st_mode), 'mtime_ns': st.st_mtime_ns}
        if stat.S_ISLNK(st.st_mode):
            row.update(kind='symlink', target=os.readlink(path), sha256=sha(os.readlink(path).encode()))
        elif stat.S_ISREG(st.st_mode):
            row.update(kind='file', sha256=filehash(path))
        else:
            row.update(kind='other', lstat_mode=st.st_mode)
            raise RuntimeError('unexpected nonregular source: ' + str(path))
        inputs.append((path, row, None))
        return row
    # Preserve original full command streams and every bridge build/test log.
    for label, root in ROOTS.items():
        for base, dirs, files in os.walk(root, followlinks=False):
            base = Path(base)
            for name in sorted(files):
                path = base / name
                rel = path.relative_to(root).as_posix()
                st = path.lstat()
                reason = None
                if path.is_relative_to(CLONE):
                    reason = 'rebuildable-source-clone-and-git'
                elif stat.S_ISREG(st.st_mode):
                    with path.open('rb') as f:
                        magic = f.read(4)
                    if magic == b'\x7fELF':
                        reason = 'ELF-runtime-binary'
                    elif path.suffix.lower() in ('.o', '.obj'):
                        reason = 'object-file'
                if reason:
                    row = {'original_path': str(path), 'would_be_member': label + '/' + rel, 'reason': reason,
                           'bytes': st.st_size, 'mode': stat.S_IMODE(st.st_mode), 'mtime_ns': st.st_mtime_ns,
                           'kind': 'symlink' if stat.S_ISLNK(st.st_mode) else 'file'}
                    row['target' if stat.S_ISLNK(st.st_mode) else 'sha256'] = os.readlink(path) if stat.S_ISLNK(st.st_mode) else filehash(path)
                    excluded.append(row)
                else:
                    add_path(path, label + '/' + rel)
            for name in list(dirs):
                path = base / name
                if path.is_symlink():
                    add_path(path, label + '/' + path.relative_to(root).as_posix())
                    dirs.remove(name)
    # Retain exact original input closure and tools independent of omitted clone.
    for rel in sorted(source.texts):
        add_path(CLONE / rel, 'source-closure/' + rel)
    for rel in ['tests/mutate.py', 'tests/mutation_bridge.py', 'tests/bridge_mutation_oracle.py', 'tests/run_tests.py', 'tests/binaries.py']:
        add_path(CLONE / rel, 'original-product-tools/' + rel)
    for label in ['outer-ledger-check-e', 'outer-ledger-regenerate-e', 'outer-w-vocabulary-e']:
        for path, expected in receipts[label]['tool_hashes'].items():
            p = Path(path)
            actual = filehash(p)
            assert actual == expected, ('external tool changed', path)
            member = 'original-external-tools/' + p.name
            if not any(r['member'] == member for _, r, _ in inputs):
                add_path(p, member)
            tool_rows.append({'receipt': label, 'path': path, 'sha256': actual})
    # Restore deleted generated source in memory using the original pure helper.
    for mid, record in sorted(executed.items()):
        m = definitions[mid]
        assert mt.mutant_key(m) == record['mutant_sha256']
        assert source.hashes[m['file']] == record['source_sha256']
        e = record['execution']
        if m.get('binary') == 'bridge':
            data = (CLONE / m['file']).read_bytes()
            assert data.count(m['old'].encode()) == 1
            data = data.replace(m['old'].encode(), m['new'].encode(), 1)
            expected = e['mutant_identity']['sidecar']['files'][m['file']]
            assert sha(data.replace(b'\r\n', b'\n')) == expected
            member = 'reconstructed-not-captured/' + mid + '/' + m['file']
            proof = {'type': 'bridge-mutant-sidecar-LF-source-hash', 'expected_sha256': expected, 'actual_sha256': sha(data.replace(b'\r\n', b'\n'))}
        else:
            data = mt.build_mutant_source(m, source).encode('utf-8')
            fp = load(ROOTS['core-regenerate-e'] / 'tmp/geometry-tests/mutants' / mid / '_mut.exe.fingerprint.json')
            expected = fp['fingerprint']['generated_source_sha256']
            assert sha(data) == expected
            member = 'reconstructed-not-captured/' + mid + '/_mutant_' + mid + '.c'
            proof = {'type': 'original-compile-fingerprint-generated-source-hash', 'expected_sha256': expected, 'actual_sha256': sha(data)}
        row = {'member': member, 'kind': 'file', 'bytes': len(data), 'sha256': sha(data), 'mode': 0o644, 'origin': 'reconstructed-not-captured', 'mutation_id': mid, 'source_sha': S, 'proof': proof}
        inputs.append((None, row, data))
        generated.append(row)
    # Validate persisted command stream hashes, including inner and outer capture.
    for label, d in receipts.items():
        if label == 'w-vocabulary-e':
            continue
        commands = d.get('commands', d.get('subcommands', []))
        if isinstance(commands, dict):
            commands = list(commands.values())
        for command in commands:
            folder = Path(command['directory'])
            for kind in ['stdout', 'stderr']:
                path = folder / (kind + '.bin')
                assert filehash(path) == command[kind + '_sha256']
                assert path.stat().st_size == command[kind + '_bytes']
                raw_checks.append({'receipt': label, 'path': str(path), 'bytes': command[kind + '_bytes'], 'sha256': command[kind + '_sha256']})
    bridge_checks = []
    base_mutants = ROOTS['core-regenerate-e'] / 'tmp/geometry-tests/mutants'
    for mid, rec in executed.items():
        e = rec['execution']
        if e['schema'] != 'geometry-bridge-mutation-execution/1':
            continue
        for phase in ['baseline_build', 'baseline', 'mutant_build', 'mutant', 'restored_build', 'restored']:
            cmd = e[phase]
            folder = base_mutants / ('_bridge-baseline/' + S if phase.startswith('baseline') else mid)
            for kind in ['stdout', 'stderr']:
                path = folder / cmd[kind + '_file']
                assert filehash(path) == cmd[kind + '_sha256']
                bridge_checks.append({'mutation_id': mid, 'phase': phase, 'stream': kind, 'path': str(path), 'bytes': path.stat().st_size, 'sha256': cmd[kind + '_sha256']})
    # Small packages: raw size <=80MiB guarantees margin under95MiB.
    archives = []
    batches = []
    batch = []
    size = 0
    for item in sorted(inputs, key=lambda x: x[1]['member']):
        if batch and size + item[1]['bytes'] > MAX_RAW:
            batches.append(batch); batch = []; size = 0
        batch.append(item); size += item[1]['bytes']
    if batch:
        batches.append(batch)
    assert len({r['member'] for _, r, _ in inputs}) == len(inputs)
    for index, batch in enumerate(batches, 1):
        name = 'ledger-e-raw-%02d.tar.gz' % index
        archive = OUT / 'archives' / name
        with tarfile.open(archive, 'w:gz', compresslevel=6) as tar:
            for path, row, data in batch:
                info = tarfile.TarInfo(row['member'])
                info.mode = row['mode']; info.mtime = 0
                if row['kind'] == 'symlink':
                    info.type = tarfile.SYMTYPE; info.linkname = row['target']; tar.addfile(info)
                else:
                    info.size = row['bytes']
                    if data is not None:
                        tar.addfile(info, io.BytesIO(data))
                    else:
                        with path.open('rb') as f:
                            tar.addfile(info, f)
        assert archive.stat().st_size < MAX_ARCHIVE
        # Verify actual archive path/length/content against each member inventory.
        expected = {r['member']: r for _, r, _ in batch}
        seen = set()
        with tarfile.open(archive, 'r:gz') as tar:
            for info in tar:
                assert info.name not in seen and info.name in expected
                seen.add(info.name); row = expected[info.name]
                if row['kind'] == 'symlink':
                    assert info.issym() and info.linkname == row['target']
                else:
                    assert info.isfile() and info.size == row['bytes']
                    h = hashlib.sha256()
                    with tar.extractfile(info) as member_stream:
                        for chunk in iter(lambda: member_stream.read(1024 * 1024), b''):
                            h.update(chunk)
                    assert h.hexdigest() == row['sha256']
        assert seen == set(expected)
        inv = 'inventories/' + name + '.members.json'
        dump(OUT / inv, {'archive': 'archives/' + name, 'member_count': len(expected), 'members': list(expected.values()), 'verified_from_archive': True})
        archives.append({'path': 'archives/' + name, 'bytes': archive.stat().st_size, 'sha256': filehash(archive), 'member_count': len(expected), 'member_inventory': inv})
    dump(OUT / 'EXCLUDED.json', {'policy': 'Recoverable source clone/.git and ELF/object binaries omitted; all omitted regular-file bytes were hashed; closure/tool snapshots are separately retained.', 'count': len(excluded), 'bytes': sum(r['bytes'] for r in excluded), 'entries': excluded})
    dump(OUT / 'RECONSTRUCTED.json', {'scope': 'Not original captured C; independently generated now with original pure transformation and matched original per-build fingerprint/sidecar.', 'count': len(generated), 'core': sum(r['proof']['type'].startswith('original-compile') for r in generated), 'bridge': sum(r['proof']['type'].startswith('bridge-') for r in generated), 'entries': generated})
    dump(OUT / 'RAW_PATHS.json', {'roots': {k: str(v) for k, v in ROOTS.items()}, 'preserved': [r for _, r, _ in inputs], 'command_stream_checks': raw_checks, 'bridge_stream_checks': bridge_checks, 'external_tools': tool_rows})
    summary = {'schema': 'geo323-ledger-publication-archive/1', 'source_sha': S, 'created_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'archival_only': True, 'tests_or_compiles_rerun': False, 'source_or_git_worktrees_modified': False, 'ledger_records': len(records), 'execution_head_count': len(executed), 'verdicts': dict(collections.Counter(r['verdict'] for r in records.values())), 'closure_sha256': source.closure, 'closure_file_count': len(source.texts), 'ledger_sha256': filehash(ROOTS['core-regenerate-e'] / 'ledger.json'), 'full_steps_exit': step_exits, 'initial_check_exit': check['exit_code'], 'outer_exits': {k: receipts[k]['outer_process']['actual_process_returncode'] for k in receipts if k.startswith('outer-')}, 'source_before_after_clean': True, 'w_vocabulary_complete': True, 'original_command_streams_checked': len(raw_checks), 'bridge_stream_references_checked': len(bridge_checks), 'bridge_unique_stream_paths': len({x['path'] for x in bridge_checks}), 'reconstructed_sources': len(generated), 'archived_members': len(inputs), 'archived_member_bytes': sum(r['bytes'] for _, r, _ in inputs), 'excluded_files': len(excluded), 'excluded_bytes': sum(r['bytes'] for r in excluded), 'archives': archives, 'limitations': ['Original core per-compile/per-attempt stdout/stderr were not persisted by tests/mutate.py; exact full-ledger command stdout/stderr and ledger structured outcomes/original stream hashes are preserved. No streams were inferred from hashes.', 'All 75 generated mutant C files here are reconstructed-not-captured and hash-matched to original build fingerprints/sidecars; original tool deletes/restores temporary C after builds.'], 'errors': []}
    dump(OUT / 'SUMMARY.json', summary)
    (OUT / 'README.md').write_text('''# #323 帳本完整重產收據封存\n\n受驗來源 `b823903e44aea091b70fd2f45b74dff6229776dc`。此目錄只整理既有已完成檔案，沒有重跑產品、測試或編譯，也沒有修改 Git 工作樹。\n\n初次 `--check` 原退出碼 1；重產中的 precheck/full-ledger/checkafter 為 1/0/0，來源前後皆同 S 且乾淨。78 筆含 75 筆同 S execution：70 killed、4 equivalent、M20 crashed；另 3 retired。M20 不列 killed。W vocabulary 是唯讀完整聯集核對，不代替架構三裁判。\n\n`SUMMARY.json` 是數量及結果索引；`RAW_PATHS.json` 保原路徑、指令 streams 核對、bridge streams 與工具；`archives/*.tar.gz` 內含完整 inner/outer 指令、原 stdout/stderr、環境、收據、前後來源指紋、新舊 ledger、bridge build/tests 原 logs、fingerprints 及來源閉包。每包小於 95 MiB；`inventories/*.members.json` 的每個 member 路徑、bytes、SHA256 已從實際壓縮包逐個讀回核對。`SHA256SUMS` 核本目錄所有出版檔（自身除外）。\n\n原 `tests/mutate.py` 對 core 內層 compile/test 未落盤完整 streams：attempt 只留結構化 result 和原 stdout/stderr SHA。這些原 ledger 資料及完整 full-ledger 正式命令 stdout/stderr 全保留；沒有憑雜湊反推或偽造未捕獲的原 stream。18 個 bridge 的 baseline/mutant/restored build/test streams 全數保留。\n\n原工具會刪除 core 暫存 C，bridge 亦恢復原來源。因此 `reconstructed-not-captured/` 明確是此次由原工具純轉換／原字節 replace 重建的 57 core 和 18 bridge variants，不聲稱當時已留存。core 逐檔比對當次 fingerprint.generated_source_sha256；bridge 比對當次 mutant sidecar.files 的原 LF source hash。完整證明列 `RECONSTRUCTED.json`。\n\n`EXCLUDED.json` 逐檔記錄可重建 source clone（含 .git）、ELF 和 object 的原路徑、大小、SHA 與 metadata；來源閉包與原工具另行保留。所有原檔仍在原位，沒有刪除。\n''', encoding='utf-8')
    sums = []
    for path in sorted(OUT.rglob('*')):
        if path.is_file() and path.name != 'SHA256SUMS':
            sums.append(filehash(path) + '  ' + path.relative_to(OUT).as_posix())
    (OUT / 'SHA256SUMS').write_text('\n'.join(sums) + '\n', encoding='ascii')
    print(json.dumps({'output': str(OUT), 'summary': summary}, ensure_ascii=False, indent=2))

if __name__ == '__main__':
    main()
