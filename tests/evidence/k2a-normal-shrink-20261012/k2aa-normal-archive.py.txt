from pathlib import Path
import shutil,json,hashlib,tarfile
root=Path('/workspace/geometry-k2aa-review');dest=root/'tests/evidence/k2a-normal-shrink-20261012';dest.mkdir(parents=True,exist_ok=True)
src=Path('/tmp/k2aa-normal-evidence')
for p in src.iterdir():
    if p.is_file():shutil.copyfile(p,dest/p.name)
    elif p.is_dir():shutil.copytree(p,dest/p.name,dirs_exist_ok=True)
for name in ['k2aa-normal-verify.py','k2aa-normal-runall-parent.py','k2aa-normal-browser-parent.py','k2aa-runall-isolated.py']:
    shutil.copyfile(Path('/tmp')/name,dest/(name+'.txt'))
for name in ['k2aa-normal-before.log','k2aa-normal-after.log','k2aa-normal-full.log','k2aa-normal-full-restored.log','k2aa-normal-list.log','k2aa-normal-runall.log']:
    shutil.copyfile(Path('/tmp')/name,dest/name)
logs=Path('/tmp/k2aa-final-binaries/run_all-logs')
with tarfile.open(dest/'runall-raw.tar.gz','w:gz') as archive:
    for p in sorted(logs.rglob('*')):
        if p.is_file():archive.add(p,arcname=str(p.relative_to(logs)),recursive=False)
members={str(p.relative_to(logs)):{'bytes':p.stat().st_size,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()} for p in sorted(logs.rglob('*')) if p.is_file()}
(dest/'runall-members.json').write_text(json.dumps(members,indent=2)+'\n')
inventory={str(p.relative_to(dest)):{'bytes':p.stat().st_size,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()} for p in sorted(dest.rglob('*')) if p.is_file() and p.name!='inventory.json'}
(dest/'inventory.json').write_text(json.dumps(inventory,indent=2)+'\n')
