from pathlib import Path
import subprocess,json,hashlib,time
root=Path('/workspace/geometry-k2aa-review')
out=Path('/tmp/k2aa-normal-evidence');out.mkdir(exist_ok=True)
paths=[root/'geometry-web/feature-attachments.js',root/'geometry-web/reference-workspace.js']
original={p:p.read_bytes() for p in paths}
cmd=['node','--test','--test-name-pattern=B1 normal collection admission|legacy normal collection confirmation','tests/test_grid_drafts_lifecycle_capacity.js','tests/test_reference_workspace_races.js']
results=[]
def run(name):
    p=subprocess.run(cmd,cwd=root,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
    (out/(name+'.log')).write_bytes(p.stdout)
    results.append({'name':name,'command':cmd,'exit':p.returncode,'sha256':hashlib.sha256(p.stdout).hexdigest(),'sources':{str(f.relative_to(root)):hashlib.sha256(f.read_bytes()).hexdigest() for f in paths}})
    return p
assert run('baseline').returncode==0
mutations=[('future-single',paths[0],b'      if (!stored && !value.orphans.includes(draft.draftId) && !draft.inputRefs.some(ref=>ref.lost))\n        boundedJSON({schema:SCHEMA,records:[{kind:GRID_KIND,schema:draft.schema,value:draft}]});\n'),('prepare-candidate',paths[1],b'    preflightGridConfirmation(kind,value);\n'),('begin-candidate',paths[1],b'    preflightGridConfirmation(prepared.kind,prepared.value);\n')]
for name,path,needle in mutations:
    assert original[path].count(needle)==1
    try:
        path.write_bytes(original[path].replace(needle,b''))
        check=subprocess.run(['node','--check',str(path)],capture_output=True)
        assert check.returncode==0
        p=run(name+'-mutant');assert p.returncode==1 and b'AssertionError' in p.stdout and b'Missing expected exception' in p.stdout
    finally:
        path.write_bytes(original[path])
    assert run(name+'-restored').returncode==0
(out/'results.json').write_text(json.dumps(results,ensure_ascii=False,indent=2)+'\n')
