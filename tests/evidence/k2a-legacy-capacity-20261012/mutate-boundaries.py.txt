def run(cmd,path):
    p=subprocess.run(cmd,cwd=private,text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
    path.write_text(p.stdout);return p
results=[]
for ident,rel,before,after,test,pattern in cases:
    folder=out/ident;folder.mkdir(exist_ok=True)
    file=private/rel;original=file.read_text();assert original.count(before)==1,(ident,original.count(before))
    command=['node','--test','--test-name-pattern='+pattern,test]
    baseline=run(command,folder/'baseline.log');assert baseline.returncode==0,(ident,'baseline',baseline.stdout)
    try:
        file.write_text(original.replace(before,after))
        build=run(['node','--check',rel],folder/'build.log');assert build.returncode==0
        red=run(command,folder/'red.log')
        assert red.returncode==1 and 'AssertionError [ERR_ASSERTION]' in red.stdout and 'ℹ skipped 0' in red.stdout and '✖ '+pattern in red.stdout,(ident,red.stdout)
        if ident in ['reject-stored-read','reject-legacy-scan','reject-unchanged-flush']:assert '容量' in red.stdout,(ident,'capacity reason required')
    finally:file.write_text(original)
    restored=run(command,folder/'restored.log');assert restored.returncode==0,(ident,'restored',restored.stdout)
    result=dict(id=ident,source=rel,sourceSHA256=hashlib.sha256(original.encode()).hexdigest(),before=before,after=after,command=command,baseline=baseline.returncode,build=build.returncode,red=red.returncode,restored=restored.returncode,designatedAssertion=True,skip=0)
    results.append(result);print(ident,'baseline green / build green / designated assertion red / restored green',flush=True)
(out/'results.json').write_text(json.dumps(results,ensure_ascii=False,indent=2)+'\n')
