assert receipt['exit']==0 and receipt['sourcesEqual'] and receipt['head']==receipt['headAfter']

logs=pathlib.Path('/tmp/wait-trace-binaries/run_all-logs'); start=(out/'sources-before.json').stat().st_mtime
members=[]
with tarfile.open(out/'native-raw.tar.gz','w:gz') as archive:
    for p in sorted(logs.rglob('*')):
        if p.is_file() and p.stat().st_mtime>=start:
            raw=p.read_bytes(); name=str(p.relative_to(logs)); archive.add(p,arcname=name)
            members.append(dict(path=name,bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest()))
(out/'native-members.json').write_text(json.dumps(members,indent=2)+'\n')
for folder,archive_name,manifest in ((out,'native-raw.tar.gz','native-members.json'),(out/'pre-admission-race','native-raw.tar.gz','native-members.json'),(out/'controlled-stop','partial-native.tar.gz','partial-members.json')):
    entries=json.loads((folder/manifest).read_text())
    with tarfile.open(folder/archive_name) as archive:
        for row in entries:
            raw=archive.extractfile(row['path']).read()
            assert len(raw)==row['bytes'] and hashlib.sha256(raw).hexdigest()==row['sha256']
binary={}
for name in ('core','bridge','integrity','model-base','engineering-json-test'):
    p=pathlib.Path('/tmp/wait-trace-binaries')/name
    binary[name]=hashlib.sha256(p.read_bytes()).hexdigest()
    shutil.copyfile(p.with_name(name+'.fingerprint.json'),out/(name+'.fingerprint.json'))
(out/'binary-final.json').write_text(json.dumps(binary,indent=2)+'\n')
(out/'counts.json').write_text(json.dumps(counts,indent=2)+'\n')
