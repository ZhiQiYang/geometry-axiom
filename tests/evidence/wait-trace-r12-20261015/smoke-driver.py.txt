import pathlib,tempfile,subprocess,json,os,urllib.request,uuid,time
root=pathlib.Path('/tmp/geometry-wait-trace'); out=root/'tests/evidence/wait-trace-r12-20261015'
feedback=pathlib.Path(tempfile.mkdtemp(prefix='wait-r12-feedback-'))
(feedback/'keep-report.txt').write_text('must remain')
(feedback/'wait-trace.jsonl').write_text('{"stage":"old","traceRunId":"old"}\n')
env=dict(os.environ,GEOMETRY_WAIT_TRACE='1',GEOMETRY_BROWSER_SUITE='canvas',GEOMETRY_FEEDBACK_DIR=str(feedback),GEOMETRY_BRIDGE_EXECUTABLE='/tmp/wait-trace-binaries/bridge')
prepared=subprocess.run(['python3','geometry-web/prepare_wait_trace.py'],cwd=root,env=env,capture_output=True,text=True)
assert prepared.returncode==0,prepared.stderr
env['GEOMETRY_WAIT_TRACE_RUN_ID']=prepared.stdout.strip()
with (out/'smoke-server.stdout.log').open('wb') as stdout,(out/'smoke-server.stderr.log').open('wb') as stderr:
    service=subprocess.Popen(['python3','geometry-web/serve.py','9042'],cwd=root,env=env,stdout=stdout,stderr=stderr)
    try:
        for attempt in range(100):
            try:
                with urllib.request.urlopen('http://127.0.0.1:9042/runtime',timeout=2) as response:
                    runtime=json.load(response); session=response.headers['X-Geometry-Session']
                break
            except OSError:
                if service.poll() is not None: raise
                time.sleep(.1)
        else: raise AssertionError('SERVER_NOT_READY')
        assert runtime['fresh'] is True,runtime
        data=dict(cmd='newid 1',requestId=str(uuid.uuid4()),session=session)
        request=urllib.request.Request('http://127.0.0.1:9042/api',data=json.dumps(data).encode(),headers={'Content-Type':'application/json','X-Geometry-Version':runtime['version']})
        with urllib.request.urlopen(request,timeout=5) as response: result=json.load(response)
        assert result['ok'] is True,result
    finally:
        service.terminate(); service.wait(timeout=10)
    assert service.returncode==0,service.returncode
archive=subprocess.run(['python3','geometry-web/archive_wait_trace.py',str(out/'cloud-artifact')],cwd=root,env=env,capture_output=True,text=True)
(out/'archive.stdout.log').write_text(archive.stdout); (out/'archive.stderr.log').write_text(archive.stderr)
(out/'archive.exit').write_text(str(archive.returncode)+'\n')
assert archive.returncode==0,archive.stdout+archive.stderr
rows=[json.loads(line) for line in (out/'cloud-artifact/wait-trace.jsonl').read_text().splitlines()]
assert all(row['traceRunId']==env['GEOMETRY_WAIT_TRACE_RUN_ID'] for row in rows)
assert {'source_lock','receipt_lock','bridge_lock','stdin_write','stdin_flush','response_queue','http_write'}<={row['stage'] for row in rows}
assert rows[-1]['stage']=='trace_complete' and rows[-1]['incomplete'] is False
assert (feedback/'keep-report.txt').read_text()=='must remain'
(out/'smoke.json').write_text(json.dumps(dict(head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),runtime=runtime,request=data,response=result,serviceExit=service.returncode,archiveExit=archive.returncode,records=len(rows),traceRunId=env['GEOMETRY_WAIT_TRACE_RUN_ID'],otherFeedbackPreserved=True,WSL=False),indent=2)+'\n')
print('CURRENT_RUN_COMPLETE_REAL_BRIDGE_PASS')
