"""Add the completed pure-file raw archive before publishing the ledger artifact."""
from pathlib import Path
import hashlib
import json
import shutil
import time

SOURCE = Path('/workspace/geo299316-ledger-raw-STAGING')
DEST = Path('/tmp/g261-next299316/tests/evidence/geo299-316-combined-20261022/ledger-e40762e9')
S = 'e40762e99b078335d78c65a3a2682a0c9ab875d0'


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


raw_summary = json.loads((SOURCE / 'SUMMARY.json').read_text())
assert raw_summary['source_sha'] == S and not raw_summary['errors']
assert raw_summary['tests_or_compiles_rerun'] is False
before = (DEST / 'SUMMARY.json').read_bytes()
primary = json.loads(before)
assert primary['execution_source_sha'] == S
assert primary['ledger_sha256'] == raw_summary['ledger_sha256']
assert primary['large_temporary_build_raw_published'] is False
target = DEST / 'complete-ledger-raw'
assert not target.exists()
for row in (SOURCE / 'SHA256SUMS').read_text().splitlines():
    expected, relative = row.split('  ', 1)
    assert digest((SOURCE / relative).read_bytes()) == expected
shutil.copytree(SOURCE, target)
for path in SOURCE.rglob('*'):
    if path.is_file():
        assert path.read_bytes() == (target / path.relative_to(SOURCE)).read_bytes()
(DEST / 'PRIMARY_SUMMARY_BEFORE_COMPLETE_RAW.json').write_bytes(before)
primary.update(large_temporary_build_raw_published=True,
               complete_ledger_raw_directory=target.name,
               complete_ledger_raw_summary_sha256=digest((target / 'SUMMARY.json').read_bytes()),
               complete_ledger_raw_members=raw_summary['archived_members'],
               complete_ledger_raw_member_bytes=raw_summary['archived_member_bytes'])
(DEST / 'SUMMARY.json').write_text(json.dumps(primary, ensure_ascii=False, indent=2) + '\n')
progress = DEST / 'PROGRESS.md'
text = progress.read_text()
text = text.replace('tmp 內大量 build／test raw 另補；原 core 內層未落盤 streams 不冒稱已保存。',
    '完整暫存 build／test raw 已放入 complete-ledger-raw/；原 core 內層未落盤 streams 仍如實明列，不冒稱已保存。')
text += f'''\n## 完整原 raw 封存已完成\n\n`complete-ledger-raw/` 已封 {raw_summary['archived_members']} members／{raw_summary['archived_member_bytes']} 原 bytes，包含全部原 bridge build／test stdout／stderr、fingerprints／sidecars、原 {raw_summary['preserved_original_core_record_bytes']} 個 core JSON 每段 bytes 核同，以及固定 S 的 {raw_summary['closure_file_count']} 個 C closure 與 {raw_summary['original_flat_python_and_root_scene_inputs']} 個原 flat Python／根 scene inputs。{raw_summary['original_command_streams_checked']} 個 top-level streams、{raw_summary['bridge_stream_references_checked']} 個 bridge stream references 已核原 hashes。\n\n{raw_summary['reconstructed_sources']} 份 variants 僅由原純 helper 重建並匹配原 fingerprint／sidecar，明標 reconstructed-not-captured。clone 子樹未遍歷／未 hash，完整排除項照列；其餘 ELF／objects 逐檔 hash。每包小於95 MiB，全部 member 已自實際封包逐 bytes／SHA256 核同。原 full2／outer1與缺 checkafter／source_after、collector preflight1、同 S 環境 retry0／strictcheck0 分欄保留。\n'''
progress.write_text(text)
shutil.copy2(__file__, DEST / 'include_complete_ledger_raw.py.txt')
(DEST / 'COMPLETE_RAW_AUGMENTATION.json').write_text(json.dumps({
    'schema': 'geo299316-complete-raw-augmentation/1', 'epoch': time.time(),
    'source_staging': str(SOURCE), 'destination': str(target),
    'primary_derived_summary_before_sha256': digest(before),
    'primary_derived_summary_after_sha256': digest((DEST / 'SUMMARY.json').read_bytes()),
    'original_primary_archive_and_command_receipts_modified': False,
    'product_or_test_code_modified': False,
    'raw_summary_sha256': digest((target / 'SUMMARY.json').read_bytes()),
    'all_source_files_copied_byte_identically': True,
}, ensure_ascii=False, indent=2) + '\n')
paths = [path for path in sorted(DEST.rglob('*')) if path.is_file() and path != DEST / 'SHA256SUMS']
(DEST / 'SHA256SUMS').write_text(''.join(digest(path.read_bytes()) + '  ' + path.relative_to(DEST).as_posix() + '\n' for path in paths))
print(json.dumps({'destination': str(target), 'files': len(paths), 'raw_members': raw_summary['archived_members']}, ensure_ascii=False))
