import hashlib

import io

import os
from pathlib import Path
import re

import sys
import tarfile

sys.dont_write_bytecode = True
S = 'e40762e99b078335d78c65a3a2682a0c9ab875d0'
OUT = Path('/workspace/geo299316-ledger-raw-STAGING')
ROOTS = {
    'original-full-phase': Path('/workspace/geo299316-ledger-regenerate-e40762e9'),
    'original-full-outer': Path('/workspace/geo299316-ledger-outer/combined-ledger-s-e40762e9'),
    'collector-preflight': Path('/workspace/geo299316-ledger-environment-retry-e40762e9'),
    'collector-preflight-outer': Path('/workspace/geo299316-ledger-retry-outer/combined-ledger-bridge-env-retry-e40762e9'),
    'bridge-environment-retry': Path('/workspace/geo299316-ledger-environment-retry-e40762e9-r2'),
    'bridge-retry-outer': Path('/workspace/geo299316-ledger-retry-outer/combined-ledger-bridge-env-retry-e40762e9-r2'),
}
FULL = ROOTS['original-full-phase']
RETRY = ROOTS['bridge-environment-retry']
CORE_CACHE = FULL / 'tmp/geometry-tests/mutants'
BRIDGE_CACHE = RETRY / 'cache'
CLONE = BRIDGE_CACHE / '_bridge-work' / S / 'repo'
PRUNED = {
    FULL / 'tmp/geometry-tests/mutants/_bridge-work',
    ROOTS['collector-preflight'] / 'cache/_bridge-work',

MAX_ARCHIVE = 95 * 1024 * 1024


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def filehash(path):
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b''):
            digest.update(chunk)
    return digest.hexdigest()


def dump(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')


def load(path):
    return json.loads(path.read_text(encoding='utf-8'))


def raw_records(raw):
    text, decoder = raw.decode('utf-8'), json.JSONDecoder()
    offset, records = 1, {}
    assert text[0] == '{'
    while True:
        while text[offset].isspace():
            offset += 1
        if text[offset] == '}':
            return records
        key, offset = decoder.raw_decode(text, offset)
        while text[offset].isspace():
            offset += 1
        assert text[offset] == ':'
        offset += 1
        while text[offset].isspace():
            offset += 1
        beginning = offset
        _, offset = decoder.raw_decode(text, offset)
        records[key] = text[beginning:offset].encode('utf-8')
        while text[offset].isspace():
            offset += 1
        if text[offset] == ',':
            offset += 1
        else:
            assert text[offset] == '}'


def step_exits(receipt):
    return [(row['label'], row['original_command_result']['actual_process_returncode'])
            for row in receipt['steps']]


def validate_receipts(receipts):
    full, retry = receipts['original-full-phase'], receipts['bridge-environment-retry']
    assert full['sha'] == S and full['ended_utc']
    assert full['source_before'] == {'head': S, 'status': ''}
    assert step_exits(full) == [('precheck', 1), ('full-ledger', 2)]
    assert full['exit_code'] == 1 and 'source_after' not in full
    assert any('No space left on device' in item for item in full['errors'])
    assert retry['source_sha'] == S and retry['ended_utc'] and retry['exit_code'] == 0
    assert not retry['errors']
    assert retry['source_before'] == retry['source_after'] == {'head': S, 'status': ''}
    assert step_exits(retry)[-3:] == [('strict-precheck', 1), ('bridge-environment-retry', 0), ('strict-checkafter', 0)]
    assert retry['complete_regeneration_in_two_environment_phases']
    assert retry['single_full_command_passed'] is False
    assert retry['original_full_phase_exit'] == 2 and retry['original_outer_exit'] == 1
    assert retry['original_partial_ledger_sha256'] == filehash(FULL / 'ledger.json')
    assert retry['ledger_sha256'] == filehash(RETRY / 'ledger.json')
    preflight = receipts['collector-preflight']
    assert preflight['source_sha'] == S and preflight['ended_utc'] and preflight['exit_code'] == 1
    assert preflight['source_before'] == preflight['source_after'] == {'head': S, 'status': ''}
    assert step_exits(preflight) == [('environment-clone', 0), ('environment-sparse', 0),
                                      ('environment-checkout', 0), ('source-oracle-bindings', 1)]
    for label, code in [('original-full-outer', 1), ('collector-preflight-outer', 1), ('bridge-retry-outer', 0)]:
        receipt = receipts[label]
        assert receipt['sha'] == S
        command = receipt['outer_process']
        assert command['actual_process_returncode'] == code
        assert command['ended_epoch'] and not command['timeout_expired']
    assert (CLONE / '.git/HEAD').read_text().strip() == S


def main():
    assert not sys.flags.optimize, 'archival assertions must remain active'
    assert not OUT.exists(), 'new staging directory required; no overwrite'
    receipts = {label: load(root / 'receipt.json') for label, root in ROOTS.items()}
    validate_receipts(receipts)
    ledger_raw = (RETRY / 'ledger.json').read_bytes()
    records = json.loads(ledger_raw)
    old_raw = (FULL / 'ledger.json').read_bytes()
    old_records = json.loads(old_raw)
    executed = {key: row for key, row in records.items() if row.get('verdict') != 'retired'}
    counts = dict(collections.Counter(row['verdict'] for row in records.values()))
    assert not set(counts) - {'killed', 'equivalent', 'crashed', 'retired'}
    assert all(row.get('execution', {}).get('head') == S for row in executed.values())
    old_executed = {key: row for key, row in old_records.items() if row.get('execution', {}).get('head') == S}
    old_slices, final_slices = raw_records(old_raw), raw_records(ledger_raw)
    preserved_core = []
    for key, row in old_executed.items():
        assert row['execution']['schema'] == 'geometry-mutation-execution/1'
        assert row == records[key] and old_slices[key] == final_slices[key]
        preserved_core.append({'id': key, 'record_bytes': len(old_slices[key]),
                               'original_sha256': sha(old_slices[key]),
                               'recovered_sha256': sha(final_slices[key]), 'exact_record_bytes_equal': True})
    # Only import pure source helpers; do not call main/run/compile/evaluate/check.
    sys.path.insert(0, str(CLONE / 'tests'))
    spec = importlib.util.spec_from_file_location('combined_archive_original_mutate', CLONE / 'tests/mutate.py')
    mt = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = mt
    spec.loader.exec_module(mt)
    import mutation_bridge
    source = mt.Sources(CLONE)
    definitions = {row['id']: row for row in mt.MUTANTS}
    assert set(records) == set(definitions)
    assert all(row['closure_sha256'] == source.closure for row in executed.values())
    assert all(row['execution']['tool_sha256'] == mt.tool_hash() for row in executed.values())
    bridge_definitions = [row for row in mt.MUTANTS if row.get('binary') == 'bridge']
    bridge_ids = [row['id'] for row in bridge_definitions]
    assert bridge_ids == receipts['bridge-environment-retry']['bridge_ids_to_retry']
    bindings = receipts['bridge-environment-retry']['original_source_oracle_bindings']
    assert bindings['source_closure'] == bindings['scratch_closure'] == source.closure
    assert bindings['restored_methods'] == mutation_bridge.RESTORED
    assert bindings['restored_method_count'] == len(mutation_bridge.RESTORED)
    bound_oracles = {row['id']: row for row in bindings['oracle_bindings']}
    for definition in bridge_definitions:
        key = definition['id']
        actual = mutation_bridge.oracle_hash(definition, CLONE)
        assert bound_oracles[key]['source_oracle_sha256'] == bound_oracles[key]['scratch_oracle_sha256'] == actual
        assert records[key]['execution']['oracle_sha256'] == actual
    OUT.mkdir()
    (OUT / 'archives').mkdir()
    (OUT / 'inventories').mkdir()
    shutil.copyfile(__file__, OUT / 'archive_ledger_raw.py.txt')
    inputs, excluded, generated, raw_checks, bridge_checks, tool_rows = [], [], [], [], [], []

    def add_path(path, member):
        information = path.lstat()
        row = {'member': member, 'original_path': str(path), 'bytes': information.st_size,
               'mode': stat.S_IMODE(information.st_mode), 'mtime_ns': information.st_mtime_ns}
        if stat.S_ISLNK(information.st_mode):
            row.update(kind='symlink', target=os.readlink(path), sha256=sha(os.readlink(path).encode()))
        else:
            assert stat.S_ISREG(information.st_mode), str(path)
            row.update(kind='file', sha256=filehash(path))
        inputs.append((path, row, None))

    # Prune whole reproducible scratch/Git directories without reading or hashing their contents.
    for label, root in ROOTS.items():
        for base, dirs, files in os.walk(root, followlinks=False):
            base = Path(base)
            for name in list(dirs):
                path = base / name
                member = label + '/' + path.relative_to(root).as_posix()
                if path in PRUNED or name == '.git':
                    info = path.lstat()
                    excluded.append({'original_path': str(path), 'would_be_member': member, 'kind': 'directory',
                                     'reason': 'rebuildable-scratch-source-and-git; subtree not traversed',
                                     'stat_bytes': info.st_size, 'mode': stat.S_IMODE(info.st_mode),
                                     'mtime_ns': info.st_mtime_ns, 'sha256': None,
                                     'content_bytes': None, 'contents_hashed': False})
                    dirs.remove(name)
                elif path.is_symlink():
                    add_path(path, member)
                    dirs.remove(name)
            for name in sorted(files):
                path = base / name
                info = path.lstat()
                member = label + '/' + path.relative_to(root).as_posix()
                reason = None
                if stat.S_ISREG(info.st_mode):
                    with path.open('rb') as stream:
                        magic = stream.read(4)
                    if magic == b'\x7fELF':
                        reason = 'ELF-runtime-binary'
                    elif path.suffix.lower() in ('.o', '.obj'):
                        reason = 'object-file'
                if reason:
                    excluded.append({'original_path': str(path), 'would_be_member': member, 'kind': 'file',
                                     'reason': reason, 'bytes': info.st_size, 'mode': stat.S_IMODE(info.st_mode),
                                     'mtime_ns': info.st_mtime_ns, 'sha256': filehash(path), 'contents_hashed': True})
                else:
                    add_path(path, member)
    # Independent closure/judge snapshots come only from the complete restored S scratch.
    for relative in sorted(source.texts):
        add_path(CLONE / relative, 'source-closure/' + relative)
    judge_paths = sorted({*CLONE.glob('tests/*.py'), *CLONE.glob('*.scene')})
    for path in judge_paths:
        add_path(path, 'original-oracle-inputs/' + path.relative_to(CLONE).as_posix())
    # Locate the actual frozen external tool version, including the preflight's older recorder.
    for label in ('original-full-outer', 'collector-preflight-outer', 'bridge-retry-outer'):
        for original_path, expected in receipts[label]['tool_hashes'].items():
            candidates = [Path(original_path)]
            for root in ROOTS.values():
                candidates.extend([root / 'tools' / Path(original_path).name, root / Path(original_path).name])
                if Path(original_path).name == 'execute_f_extra.py':
                    candidates.append(root / 'launcher.py.txt')
            matches = [path for path in candidates if path.is_file() and filehash(path) == expected]
            assert matches, ('original external tool bytes not found', label, original_path, expected)
            actual_path = matches[0]
            member = 'original-external-tools/' + label + '/' + Path(original_path).name
            add_path(actual_path, member)
            tool_rows.append({'receipt': label, 'original_path': original_path, 'snapshot_source': str(actual_path),
                              'sha256': expected, 'member': member})
    # Reconstructed variants are new pure transforms, never described as original captured C.
    for key, record in sorted(executed.items()):
        definition = definitions[key]
        assert mt.mutant_key(definition) == record['mutant_sha256']
        assert source.hashes[mt.mutant_file(definition)] == record['source_sha256']
        evidence = record['execution']
        if definition.get('binary') == 'bridge':
            raw = (CLONE / definition['file']).read_bytes()
            assert raw.count(definition['old'].encode()) == 1
            raw = raw.replace(definition['old'].encode(), definition['new'].encode(), 1)
            expected = evidence['mutant_identity']['sidecar']['files'][definition['file']]
            actual = sha(raw.replace(b'\r\n', b'\n'))
            member = 'reconstructed-not-captured/' + key + '/' + definition['file']
            proof = {'type': 'bridge-mutant-sidecar-LF-source-hash', 'expected_sha256': expected, 'actual_sha256': actual}
        else:
            raw = mt.build_mutant_source(definition, source).encode('utf-8')
            fingerprint = load(CORE_CACHE / key / '_mut.exe.fingerprint.json')
            expected = fingerprint['fingerprint']['generated_source_sha256']
            actual = sha(raw)
            member = 'reconstructed-not-captured/' + key + '/_mutant_' + key + '.c'
            proof = {'type': 'original-compile-fingerprint-generated-source-hash', 'expected_sha256': expected, 'actual_sha256': actual}
        assert actual == expected, key
        row = {'member': member, 'kind': 'file', 'bytes': len(raw), 'sha256': sha(raw), 'mode': 0o644,
               'origin': 'reconstructed-not-captured', 'mutation_id': key, 'source_sha': S, 'proof': proof}
        inputs.append((None, row, raw))
        generated.append(row)
    for label, receipt in receipts.items():
        commands = receipt.get('commands', receipt.get('subcommands', []))
        if isinstance(commands, dict):
            commands = list(commands.values())
        for command in commands:
            for stream in ('stdout', 'stderr'):
                path = Path(command['directory']) / (stream + '.bin')
                assert filehash(path) == command[stream + '_sha256']
                assert path.stat().st_size == command[stream + '_bytes']
                raw_checks.append({'receipt': label, 'path': str(path), 'bytes': path.stat().st_size,
                                   'sha256': command[stream + '_sha256']})
    for key, record in executed.items():
        evidence = record['execution']
        if evidence['schema'] != 'geometry-bridge-mutation-execution/1':
            continue
        phases = [(name, evidence[name]) for name in ('baseline_build', 'baseline', 'mutant_build', 'mutant', 'restored_build', 'restored')]
        phases.extend(('mutant_attempt_' + str(index), item)
                      for index, item in enumerate(evidence.get('mutant_attempts', []), 1))
        for phase, command in phases:
            folder = BRIDGE_CACHE / ('_bridge-baseline/' + S if phase.startswith('baseline') else key)
            for stream in ('stdout', 'stderr'):
                path = folder / command[stream + '_file']
                assert path.resolve().is_relative_to(folder.resolve())
                assert filehash(path) == command[stream + '_sha256']
                bridge_checks.append({'mutation_id': key, 'phase': phase, 'stream': stream,
                                      'path': str(path), 'bytes': path.stat().st_size, 'sha256': command[stream + '_sha256']})
    assert len({row['member'] for _, row, _ in inputs}) == len(inputs)
    batches, batch, size = [], [], 0
    for item in sorted(inputs, key=lambda entry: entry[1]['member']):
        assert item[1]['bytes'] <= MAX_RAW, ('single raw member exceeds batch bound', item[1]['member'])
        if batch and size + item[1]['bytes'] > MAX_RAW:
            batches.append(batch)
            batch, size = [], 0
        batch.append(item)
        size += item[1]['bytes']
    if batch:
        batches.append(batch)
    archives = []
    for number, batch in enumerate(batches, 1):
        name = 'ledger-raw-%02d.tar.gz' % number
        archive = OUT / 'archives' / name
        with tarfile.open(archive, 'w:gz', compresslevel=6) as handle:
            for path, row, raw in batch:
                info = tarfile.TarInfo(row['member'])
                info.mode, info.mtime = row['mode'], 0
                if row['kind'] == 'symlink':
                    info.type, info.linkname = tarfile.SYMTYPE, row['target']
                    handle.addfile(info)
                else:
                    info.size = row['bytes']
                    if raw is not None:
                        handle.addfile(info, io.BytesIO(raw))
                    else:
                        with path.open('rb') as stream:
                            handle.addfile(info, stream)
        assert archive.stat().st_size < MAX_ARCHIVE
        expected, seen = {row['member']: row for _, row, _ in batch}, set()
        with tarfile.open(archive, 'r:gz') as handle:
            for info in handle:
                assert info.name not in seen and info.name in expected
                seen.add(info.name)
                row = expected[info.name]
                if row['kind'] == 'symlink':
                    assert info.issym() and info.linkname == row['target']
                else:
                    assert info.isfile() and info.size == row['bytes']
                    digest = hashlib.sha256()
                    with handle.extractfile(info) as stream:
                        for chunk in iter(lambda: stream.read(1024 * 1024), b''):
                            digest.update(chunk)
                    assert digest.hexdigest() == row['sha256']
        assert seen == set(expected)
        inventory = 'inventories/' + name + '.members.json'
        dump(OUT / inventory, {'archive': 'archives/' + name, 'member_count': len(expected),
                               'members': list(expected.values()), 'verified_from_archive': True})
        archives.append({'path': 'archives/' + name, 'bytes': archive.stat().st_size,
                         'sha256': filehash(archive), 'member_count': len(expected), 'member_inventory': inventory})
    limitations = [
        'Original core per-compile/per-attempt stdout/stderr were not persisted by tests/mutate.py. The full original command stdout/stderr, structured results and original stream hashes are preserved; absent streams were not inferred or fabricated.',
        'All generated variants here are reconstructed-not-captured from fixed S with the original pure transformation and matched to original build fingerprints or mutant sidecar LF hashes.',
        'Original full command returned 2 and outer returned 1 due to ENOSPC; original checkafter/source_after are missing and remain missing. Only the affected bridge phase was retried at the same S, followed by the original strict check.',
        'Rebuildable scratch source/Git subtrees are pruned without traversal or content hashes. Their directory metadata is explicitly marked unhashed; exact C closure, flat Python judges and root scene inputs are separately preserved.',
    ]
    dump(OUT / 'EXCLUDED.json', {'policy': limitations[-1] + ' ELF/object files outside those subtrees are individually SHA256-hashed.',
                                'count': len(excluded), 'hashed_excluded_file_bytes': sum(row.get('bytes', 0) for row in excluded if row.get('contents_hashed')),
                                'entries': excluded})
    dump(OUT / 'RECONSTRUCTED.json', {'scope': limitations[1], 'count': len(generated),
                                     'core': sum(row['proof']['type'].startswith('original-compile') for row in generated),
                                     'bridge': sum(row['proof']['type'].startswith('bridge-') for row in generated), 'entries': generated})
    dump(OUT / 'PRESERVED_CORE_RECORDS.json', {'scope': 'Exact JSON value bytes, not only parsed equality, retained across the environment retry.',
                                             'count': len(preserved_core), 'entries': preserved_core})
    dump(OUT / 'RAW_PATHS.json', {'roots': {key: str(value) for key, value in ROOTS.items()},
                                 'preserved': [row for _, row, _ in inputs], 'command_stream_checks': raw_checks,
                                 'bridge_stream_checks': bridge_checks, 'external_tools': tool_rows})
    summary = {'schema': 'geo299316-ledger-raw-archive/1', 'source_sha': S,
               'created_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
               'archival_only': True, 'tests_or_compiles_rerun': False, 'source_or_git_worktrees_modified': False,
               'ledger_records': len(records), 'execution_head_count': len(executed), 'verdicts': counts,
               'closure_sha256': source.closure, 'closure_file_count': len(source.texts),
               'original_flat_python_and_root_scene_inputs': len(judge_paths), 'ledger_sha256': sha(ledger_raw),
               'original_full_steps_exit': step_exits(receipts['original-full-phase']),
               'collector_preflight_steps_exit': step_exits(receipts['collector-preflight']),
               'retry_steps_exit': step_exits(receipts['bridge-environment-retry']),
               'outer_exits': {label: receipts[label]['outer_process']['actual_process_returncode']
                               for label in ('original-full-outer', 'collector-preflight-outer', 'bridge-retry-outer')},
               'original_source_after_missing': True, 'retry_source_before_after_clean': True,
               'single_full_command_passed': False, 'preserved_original_core_record_bytes': len(preserved_core),
               'restored_original_method_count': len(mutation_bridge.RESTORED),
               'original_command_streams_checked': len(raw_checks), 'bridge_stream_references_checked': len(bridge_checks),
               'bridge_unique_stream_paths': len({row['path'] for row in bridge_checks}),
               'reconstructed_sources': len(generated), 'archived_members': len(inputs),
               'archived_member_bytes': sum(row['bytes'] for _, row, _ in inputs), 'archives': archives,
               'limitations': limitations, 'errors': []}
    dump(OUT / 'SUMMARY.json', summary)
    (OUT / 'README.md').write_text(f'''# #299+#316 帳本原始輸出封存

固定來源 S：`{S}`。本工具只整理已完成檔案及以原純函式重建突變來源；沒有執行產品、裁判、編譯、Git 寫入或安裝 ledger。

原整份重產 full 命令退出 2、outer 退出 1，fresh bridge checkout 因 ENOSPC 失敗；原 checkafter／source_after 沒有錄到，保持缺漏。collector 順序前置查核另外退出 1，發生於任何 mutate 命令前。修好雲端容量後同 S 只重跑原受影響 bridge IDs，原 strict check 接續通過；完整結果分欄見 SUMMARY.json，不能稱單次 full 命令通過。

動態帳本 {len(records)} 筆、{len(executed)} execution 均綁 S；分類 `{counts}`。原 {len(preserved_core)} 個 core 紀錄逐 JSON value bytes 等同，完整證明在 PRESERVED_CORE_RECORDS.json。原 RESTORED 方法數 {len(mutation_bridge.RESTORED)}，不依舊交接口頭數字增加或刪除。

RAW_PATHS.json 記所有原命令 stdout／stderr 及 bridge baseline／mutant／restored build/test streams 核對；archives 每包 raw 不超過80 MiB、壓縮檔小於95 MiB，inventories 每個 member 已自實際封包逐 bytes／SHA256 核回。來源 include closure、完整 flat Python judges、根 scene inputs 及實際外部工具版本另行保存。

原 core 內層 compile／attempt streams 未全部落盤，只留結構化結果及原 hashes；不以 hash 反推不存在的 stdout。RECONSTRUCTED.json 的 {len(generated)} 個 variants 明標 reconstructed-not-captured，core 比對原 fingerprint generated_source_sha256，bridge 比對原 mutant sidecar LF source hash，不能稱當時已捕獲原暫存 C。

EXCLUDED.json 對可重建 scratch／Git 子樹只記目錄 metadata、不讀內容也不假稱算過 hash；C closure／judge inputs 已獨立保存。其餘 ELF／objects 逐檔存原路徑、bytes、SHA256。所有原檔仍留原位。
''', encoding='utf-8')
    (OUT / 'SHA256SUMS').write_text(''.join(filehash(path) + '  ' + path.relative_to(OUT).as_posix() + '\n'
        for path in sorted(OUT.rglob('*')) if path.is_file() and path.name != 'SHA256SUMS'), encoding='utf-8')
    print(json.dumps({'output': str(OUT), 'summary': summary}, ensure_ascii=False, indent=2))


if __name__ == '__main__':
    main()
