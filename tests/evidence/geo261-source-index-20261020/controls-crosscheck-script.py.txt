"""Cross-check the hard-coded edge tables in the harness against (1) the C `edge-search` output
(product hash) and (2) an independent Python re-implementation of geo_source_edge_hash.
Usage: python3 srcidx_cmp.py <harness.c> <edge-search.out>"""
import re
import sys

M = (1 << 64) - 1
P = 11400714785074694791
FNV = 14695981039346656037


def hs(h, v):
    h ^= v & M
    h = (h * P) & M
    return h ^ (h >> 32)


def edge_hash(c, s, k):
    return hs(hs(hs(FNV, c), s), k)


text = open(sys.argv[1], encoding="utf-8").read()
group_src = re.search(r"edge_group\[EDGE_GROUP\]=\{(.*?)\};", text, re.S).group(1)
pairs_src = re.search(r"edge_pairs\[EDGE_PAIRS\]=\{(.*?)\};", text, re.S).group(1)
group = [tuple(map(int, m)) for m in re.findall(r"\{(\d+),(\d+),(\d+)\}", group_src)]
pairs = [tuple(map(int, m)) for m in re.findall(r"\{(\d+),(\d+),(\d+),(\d+)\}", pairs_src)]
print("table sizes:", len(group), len(pairs))

out = open(sys.argv[2], encoding="utf-8").read()
sec_g, sec_p = out.split("edge_pairs:")
cgroup = [tuple(map(int, m)) for m in re.findall(r"\{(\d+),(\d+),(\d+)\},", sec_g)]
cpairs = [tuple(map(int, m)) for m in re.findall(r"\{(\d+),(\d+),(\d+),(\d+)\},", sec_p)]
print("C edge-search == hard-coded table:", cgroup == group and cpairs == pairs)

N = 512
pgroup, ppairs = [], []
for c in range(N):
    for s in range(N):
        if c == s:
            continue
        sl = [edge_hash(c, s, k) & 4095 for k in range(3)]
        for k in range(3):
            if sl[k] == 4095 and len(pgroup) < 32:
                pgroup.append((c, s, k))
        for a in range(3):
            for b in range(a + 1, 3):
                if sl[a] == sl[b] and sl[a] != 4095 and len(ppairs) < 8:
                    ppairs.append((c, s, a, b))
print("independent Python search == hard-coded table:", pgroup == group and ppairs == pairs)
print("python re-check all group slots == 4095:", all(edge_hash(*g) & 4095 == 4095 for g in group))
print("python re-check all pair slots equal:", all(edge_hash(c, s, a) & 4095 == edge_hash(c, s, b) & 4095 for c, s, a, b in pairs))
print("distinct edges:", len(set(group) | {(c, s, k) for c, s, a, b in pairs for k in (a, b)}), "expected", 32 + 16)
ok = cgroup == group and cpairs == pairs and pgroup == group and ppairs == pairs
sys.exit(0 if ok else 1)
