"""Independent Python re-implementation of geo_source_edge_hash (source.c lines 29-47),
used only to cross-check the edge keys that the C harness finds with the product hash."""
import sys

M = (1 << 64) - 1
P = 11400714785074694791
FNV = 14695981039346656037


def hs(h, v):
    h ^= v & M
    h = (h * P) & M
    return h ^ (h >> 32)


def edge_hash(c, s, k):
    return hs(hs(hs(FNV, c), s), k)


N = 512
SLOT = 4095
MASK = 4095
GROUP = 32
PAIRS = 8

group = []
pairs = []
for c in range(N):
    for s in range(N):
        if c == s:
            continue
        slots = [edge_hash(c, s, k) & MASK for k in range(3)]
        for k in range(3):
            if slots[k] == SLOT and len(group) < GROUP:
                group.append((c, s, k))
        if len(pairs) < PAIRS:
            for a in range(3):
                for b in range(a + 1, 3):
                    if slots[a] == slots[b] and slots[a] != SLOT and len(pairs) < PAIRS:
                        pairs.append((c, s, a, b, slots[a]))
print("group", len(group))
for g in group:
    print(g)
print("pairs", len(pairs))
for p in pairs:
    print(p)
# totals available in the whole space, to show how much slack there is
total_slot = sum(1 for c in range(N) for s in range(N) if c != s for k in range(3)
                 if edge_hash(c, s, k) & MASK == SLOT)
total_pairs = sum(1 for c in range(N) for s in range(N) if c != s
                  for a in range(3) for b in range(a + 1, 3)
                  if edge_hash(c, s, a) & MASK == edge_hash(c, s, b) & MASK)
print("space: triples in slot", SLOT, "=", total_slot, "; same-slot kind pairs =", total_pairs)
