import subprocess, pathlib, time, json, os
root = pathlib.Path('/tmp/geometry-wait-trace')
out = root / 'tests/evidence/wait-trace-20261014'
head = subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip()
env = dict(os.environ, MUTATE_CACHE='/tmp/wait-trace-mutation-cache')
env.pop('GEOMETRY_WAIT_TRACE',None)
started = time.monotonic()
with (out/'native.stdout.log').open('wb') as stdout, (out/'native.stderr.log').open('wb') as stderr:
    result = subprocess.run(['bash','-c','ulimit -s 16384; exec python /tmp/wait-trace-native.py'],cwd=root,env=env,stdout=stdout,stderr=stderr)
after = subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip()
(out/'native.json').write_text(json.dumps(dict(head=head,headAfter=after,exit=result.returncode,seconds=time.monotonic()-started),indent=2)+'\n')
print((out/'native.json').read_text())
