import sys, pathlib, tempfile, os, json
sys.path.insert(0, '/tmp/geometry-wait-trace/tests')
from test_serve import BridgeServerTests
directory=tempfile.mkdtemp(prefix='receipt-trace-probe-')
os.environ.update(GEOMETRY_WAIT_TRACE='1', GEOMETRY_FEEDBACK_DIR=directory)
t=BridgeServerTests(); t.setUp(); m=t.server_module
receipt='12345678-1234-1234-1234-123456789012'
other='87654321-4321-4321-4321-210987654321'
try:
    t.responses.put(b'{"ok":true,"text":"native-result"}\n')
    result=m.recorded_api({'cmd':'newid 1','requestId':receipt,'session':m.root_context.session})
    data={'receipt':receipt,'requestId':other,'cmd':'ignored','session':m.root_context.session}
    handler=t.post_handler('/api',json.dumps(data).encode())
    handler.do_POST()
    reply=json.loads(handler._send.call_args.args[0])
    assert reply=={'ok':True,'result':json.loads(result)}
    rows=[json.loads(line) for line in (pathlib.Path(directory)/'wait-trace.jsonl').read_text().splitlines()]
    print(json.dumps(dict(request=data,response=reply,rows=rows),indent=2))
    receipt_rows=[row for row in rows if row['stage']=='receipt_lock']
    assert len(receipt_rows)==1
    assert receipt_rows[0]['businessRequestId']==receipt, 'WAIT_TRACE_RECEIPT_ID_PRECEDENCE'
    assert receipt_rows[0]['command']=='receipt', 'WAIT_TRACE_RECEIPT_COMMAND'
finally:
    t.tearDown()
