import subprocess, pathlib, os, json, time, urllib.request, uuid, tempfile
root=pathlib.Path('/tmp/geometry-wait-trace')
out=root/'tests/evidence/wait-trace-20261014/final'
feedback=pathlib.Path(tempfile.mkdtemp(prefix='wait-trace-smoke-'))
env=dict(os.environ,GEOMETRY_WAIT_TRACE='1',GEOMETRY_FEEDBACK_DIR=str(feedback),GEOMETRY_BRIDGE_EXECUTABLE='/tmp/wait-trace-binaries/bridge')
started=time.monotonic()
with (out/'smoke-server.stdout.log').open('wb') as stdout,(out/'smoke-server.stderr.log').open('wb') as stderr:
    process=subprocess.Popen(['python','geometry-web/serve.py','9040'],cwd=root,env=env,stdout=stdout,stderr=stderr)
    try:
        for _ in range(100):
            try:
                with urllib.request.urlopen('http://127.0.0.1:9040/runtime',timeout=2) as response:
                    runtime=json.load(response)
                    session=response.headers['X-Geometry-Session']
                break
            except OSError:
                if process.poll() is not None: raise
                time.sleep(.1)
        else: raise AssertionError('SMOKE_SERVER_NOT_READY')
        assert runtime['fresh'] is True
        data={'cmd':'newid 1','requestId':str(uuid.uuid4()),'session':session}
        request=urllib.request.Request('http://127.0.0.1:9040/api',data=json.dumps(data).encode(),headers={'Content-Type':'application/json','X-Geometry-Version':runtime['version']})
        with urllib.request.urlopen(request,timeout=5) as response:
            result=json.load(response)
        assert result['ok'] is True, result
        rows=[json.loads(line) for line in (feedback/'wait-trace.jsonl').read_text().splitlines()]
        api=[r for r in rows if r['route']=='/api']
        assert {'source_lock','receipt_lock','bridge_lock','stdin_write','stdin_flush','response_queue','http_write'} <= {r['stage'] for r in api}
        (out/'smoke-trace.jsonl').write_bytes((feedback/'wait-trace.jsonl').read_bytes())
        (out/'smoke.json').write_text(json.dumps(dict(runtime=runtime,request=data,response=result,exit=0,seconds=time.monotonic()-started),indent=2)+'\n')
    finally:
        process.terminate()
        process.wait(timeout=10)
print('REAL_BRIDGE_TRACE_PASS')
