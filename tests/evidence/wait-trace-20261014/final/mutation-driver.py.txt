import pathlib, subprocess, json, time
root = pathlib.Path('/tmp/geometry-wait-trace')
out = root / 'tests/evidence/wait-trace-20261014/final'
out.mkdir(parents=True, exist_ok=True)
rows = []
def run(label, args):
    started = time.monotonic()
    result = subprocess.run(args, cwd=root, capture_output=True)
    (out / (label+'.stdout.log')).write_bytes(result.stdout)
    (out / (label+'.stderr.log')).write_bytes(result.stderr)
    rows.append(dict(label=label, argv=args, exit=result.returncode, seconds=time.monotonic()-started))
    return result.returncode
py = ['python', 'tests/test_wait_trace.py']
js = ['node', '--test', 'tests/test_transport_failure_trace.js']
assert run('fixed-python',py)==0
assert run('fixed-js',js)==0
for label, file, old, new, args in [
    ('include-diagnostic-io', 'geometry-web/wait_trace.py', "    emit(stage + '_begin')\n    started = time.monotonic()", "    started = time.monotonic()\n    emit(stage + '_begin')", py),
    ('remove-wsl-canvas-gate', '.github/workflows/integration-tests.yml', 'canvas:*[Mm]icrosoft*) export GEOMETRY_WAIT_TRACE=1 ;;', 'canvas:*[Mm]icrosoft*) : ;;', py),
    ('default-on', 'geometry-web/wait_trace.py', "ENABLED = os.environ.get('GEOMETRY_WAIT_TRACE') == '1'", 'ENABLED = True', py),
    ('overwrite-first', 'geometry-web/transport_failure_trace.js', 'if (!first && (', 'if ((', js),
    ('remove-bridge-hook', 'geometry-web/serve.py', 'root_context = wait_trace.native(NativeContext(BRIDGE_CMD))', 'root_context = NativeContext(BRIDGE_CMD)', py),
]:
    path = root / file
    original = path.read_bytes()
    assert old.encode() in original
    try:
        path.write_bytes(original.replace(old.encode(),new.encode(),1))
        assert run(label,args)!=0, label
    finally:
        path.write_bytes(original)
    assert run(label+'-restored',args)==0
(out / 'mutations.json').write_text(json.dumps(rows,indent=2)+'\n')
