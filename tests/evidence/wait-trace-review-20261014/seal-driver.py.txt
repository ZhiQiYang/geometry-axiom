import pathlib, json, hashlib, tarfile, shutil, subprocess, sys, re
root=pathlib.Path('/tmp/geometry-wait-trace')
out=root/'tests/evidence/wait-trace-review-20261014'
receipt=json.loads((out/'native.json').read_text())
assert (receipt['exit']==0 or '--failed' in sys.argv) and receipt['sourcesEqual'] and receipt['head']==receipt['headAfter'], receipt
logs=pathlib.Path('/tmp/wait-trace-binaries/run_all-logs')
started=(out/'sources-before.json').stat().st_mtime
files=sorted(p for p in logs.rglob('*') if p.is_file() and p.stat().st_mtime >= started)
members=[]
with tarfile.open(out/'native-raw.tar.gz','w:gz') as archive:
    for p in files:
        name=str(p.relative_to(logs))
        raw=p.read_bytes()
        members.append(dict(path=name,bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest()))
        archive.add(p,arcname=name)
(out/'native-members.json').write_text(json.dumps(members,indent=2)+'\n')
with tarfile.open(out/'native-raw.tar.gz','r:gz') as archive:
    for entry in members:
        raw=archive.extractfile(entry['path']).read()
        assert len(raw)==entry['bytes'] and hashlib.sha256(raw).hexdigest()==entry['sha256']
binary={}
if '--failed' in sys.argv:
    print(json.dumps(dict(native=receipt,archiveMembers=len(members),failed=True)))
    sys.exit(0)
for p in pathlib.Path('/tmp/wait-trace-binaries').glob('*'):
    if p.is_file() and p.suffix!='.json':
        binary[p.name]=hashlib.sha256(p.read_bytes()).hexdigest()
    elif p.name.endswith('.fingerprint.json'):
        shutil.copyfile(p,out/p.name)
(out/'binary-final.json').write_text(json.dumps(binary,indent=2)+'\n')
protected=['公理層.md','geometry-guide/公理層-白話.md','geometry-guide/公理層-閱讀器.html','BACKLOG.md','決策佇列.md','current_geometry.c','tests/mutants_results.json']
different=subprocess.check_output(['git','diff','origin/main','--name-only','--',*protected],cwd=root,text=True)
assert not different,different
text=(out/'native.stdout.log').read_text()
counts=re.search(r'^合計.*?\s+(\d+)\s+(\d+)\s+(\d+)\s+(\d+)\s+([\d.]+)\s*$',text,re.M)
assert counts, 'native counts missing'
totals=dict(total=int(counts[1]),passed=int(counts[2]),failed=int(counts[3]),skipped=int(counts[4]))
assert totals==dict(total=2796,passed=2795,failed=0,skipped=1),totals
(out/'counts.json').write_text(json.dumps(totals,indent=2)+'\n')

(out/'README.md').write_text(readme)
inventory=[]
for p in sorted(out.rglob('*')):
    if p.is_file() and p.name!='inventory.json':
        raw=p.read_bytes()
        inventory.append(dict(path=str(p.relative_to(out)),bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest()))
