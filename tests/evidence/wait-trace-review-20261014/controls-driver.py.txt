import pathlib, subprocess, json
root = pathlib.Path('/tmp/geometry-wait-trace')
out = root / 'tests/evidence/wait-trace-review-20261014'
cases = [
 ('replace-original-exit', '.github/workflows/integration-tests.yml', b'            set +e\n', b'            :\n', 'test_archive_cleanup_preserves_failed_job_status'),
 ('remove-release', 'geometry-web/wait_trace.py', b"emit(self.name + '_release', holder=holder, heldSeconds=held_seconds)", b'pass', 'test_outer_release_records_each_holder_before_unlock'),
 ('inner-release', 'geometry-web/wait_trace.py', b'if self.depth == 1 and self.owner:', b'if self.owner:', 'test_outer_release_records_each_holder_before_unlock'),
 ('remove-archive-always', '.github/workflows/integration-tests.yml', b"if: always() && matrix.suite == 'canvas'", b"if: success() && matrix.suite == 'canvas'", 'test_archive_workflow_runs_even_on_failure'),
 ('allow-missing-archive', 'geometry-web/archive_wait_trace.py', b"receipt['status'] = 'missing-enabled-trace'\n        status = 1", b"receipt['status'] = 'missing-enabled-trace'\n        status = 0", 'test_runner_archive_keeps_raw_trace_and_effective_environment'),
 ('omit-archive-copy', 'geometry-web/archive_wait_trace.py', b'shutil.copyfile(source, copied)', b'pass', 'test_runner_archive_keeps_raw_trace_and_effective_environment'),
 ('omit-effective-env', '.github/workflows/integration-tests.yml', b'            printf \'GEOMETRY_WAIT_TRACE=%s\\n\' "${GEOMETRY_WAIT_TRACE-}" >> "$GITHUB_ENV"', b'            :', 'test_workflow_enables_only_wsl_canvas_before_server'),
]
results = []
def run(name, test):
    with (out / (name + '.log')).open('wb') as stream:
        p = subprocess.run(['python3', 'tests/test_wait_trace.py', 'WaitTraceTests.' + test], cwd=root, stdout=stream, stderr=subprocess.STDOUT)
    (out / (name + '.exit')).write_text(str(p.returncode) + '\n')
    return p.returncode
for name, file, old, new, test in cases:
    path = root / file
    original = path.read_bytes()
    assert old in original, name
    try:
        path.write_bytes(original.replace(old, new))
        red = run(name, test)
    finally:
        path.write_bytes(original)
    green = run(name + '-restored', test)
    results.append(dict(name=name, mutantExit=red, restoredExit=green))
    assert red == 1 and green == 0, results[-1]
(out / 'controls.json').write_text(json.dumps(results, indent=2) + '\n')
print(json.dumps(results))
