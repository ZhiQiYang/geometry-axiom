import pathlib, subprocess, time, json, os, hashlib
root = pathlib.Path('/tmp/geometry-wait-trace')
out = root / 'tests/evidence/wait-trace-review-20261014'
head = subprocess.check_output(['git','rev-parse','HEAD'], cwd=root, text=True).strip()
paths = subprocess.check_output(['git','ls-files','-z','geometry-web','geometry-core','tests/run_all.py','tests/run_all.expected.json','tests/test_wait_trace.py','tests/test_transport_failure_trace.js','.github/workflows/integration-tests.yml','current_geometry.c'],cwd=root).decode().rstrip('\0').split('\0')
paths = [p for p in paths if '/evidence/' not in p and '/release-evidence/' not in p]
def identity():
    return {p:hashlib.sha256((root/p).read_bytes()).hexdigest() for p in paths}
before=identity()
(out/'sources-before.json').write_text(json.dumps(before,indent=2)+'\n')
started=time.monotonic()
env=dict(os.environ,MUTATE_CACHE='/tmp/wait-trace-mutation-cache')
env.pop('GEOMETRY_WAIT_TRACE',None)
with (out/'native.stdout.log').open('wb') as stdout, (out/'native.stderr.log').open('wb') as stderr:
    run=subprocess.run(['bash','-c','ulimit -s 16384; exec python3 /tmp/wait-trace-native.py'],cwd=root,env=env,stdout=stdout,stderr=stderr)
after=identity()
(out/'sources-after.json').write_text(json.dumps(after,indent=2)+'\n')
(out/'native.json').write_text(json.dumps(dict(head=head,headAfter=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),exit=run.returncode,seconds=time.monotonic()-started,sourcesEqual=before==after,sourceCount=len(paths)),indent=2)+'\n')
