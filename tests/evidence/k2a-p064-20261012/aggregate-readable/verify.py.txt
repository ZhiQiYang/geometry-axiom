import base64
import hashlib
import io
import json
import tarfile
from pathlib import Path

root = Path(__file__).resolve().parent
manifest = json.loads((root / 'manifest.json').read_text())
pieces = []
for part in manifest['parts']:
    data = (root / part['file']).read_bytes()
    assert len(data) == part['bytes']
    assert hashlib.sha256(data).hexdigest() == part['sha256']
    pieces.append(b''.join(data.split()))
archive = base64.b64decode(b''.join(pieces), validate=True)
assert len(archive) == manifest['archive']['bytes']
assert hashlib.sha256(archive).hexdigest() == manifest['archive']['sha256']
assert archive == (root.parent / 'aggregate-raw.tar.gz').read_bytes()
expected = json.loads((root.parent / 'aggregate-members.json').read_text())
actual = {}
with tarfile.open(fileobj=io.BytesIO(archive), mode='r:gz') as tar:
    for entry in tar.getmembers():
        if not entry.isfile():
            continue
        assert entry.name not in actual
        data = tar.extractfile(entry).read()
        actual[entry.name] = {'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}
assert actual == expected
assert len(actual) == manifest['member_count']
print(f"Verified original archive and {len(actual)} file members; no tests rerun.")
