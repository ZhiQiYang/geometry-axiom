from pathlib import Path
import subprocess,json,sys
out=Path('/tmp/geometry-control-diagnostics/tests/evidence/grid-control-diagnostics-20261014')
records=[]
for mode,expected in [('baseline',1),('fixed',0)]:
 cmd=[sys.executable,'/tmp/grid-control-oracle.py',mode];p=subprocess.run(cmd,capture_output=True,text=True)
 (out/(mode+'-parent.stdout')).write_text(p.stdout);(out/(mode+'-parent.stderr')).write_text(p.stderr)
 assert p.returncode==expected,(mode,p.stderr)
 records.append({'command':cmd,'actual_returncode':p.returncode,'scope':'Extracted real ctl + actual Chromium DOM; sleep/click recorder, not historical P87 root cause'})
(out/'oracle-parents.json').write_text(json.dumps(records,indent=2)+'\n')
(out/'oracle.py.txt').write_bytes(Path('/tmp/grid-control-oracle.py').read_bytes())
(out/'oracle-parent.py.txt').write_bytes(Path(__file__).read_bytes())
(out/'initial-helper-error.log').write_bytes(Path('/tmp/grid-control-fixed.log').read_bytes())
print(json.dumps(records))
