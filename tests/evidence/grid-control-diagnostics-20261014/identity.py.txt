from pathlib import Path
import hashlib,json,subprocess,sys
root=Path('/tmp/geometry-control-diagnostics');out=root/'tests/evidence/grid-control-diagnostics-20261014'
paths=subprocess.check_output(['git','ls-files'],cwd=root,text=True).splitlines()
sources={p:hashlib.sha256((root/p).read_bytes()).hexdigest() for p in paths if Path(p).suffix in {'.c','.h','.inc','.js','.py','.html'} and '/evidence/' not in p and '/release-evidence/' not in p and '/node_modules/' not in p and '/vendor/' not in p}
bins={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in Path('/tmp/grid-diag-binaries').iterdir() if p.is_file() and p.name in {'core','bridge','integrity','model-base','engineering-json-test'}}
record={'head':subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),'sources':sources,'binaries':bins}
if sys.argv[1]=='after':assert record==json.loads((out/'identity-before.json').read_text())
(out/('identity-'+sys.argv[1]+'.json')).write_text(json.dumps(record,indent=2)+'\n')
print(len(sources),len(bins))
