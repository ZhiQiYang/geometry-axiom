"""核對裁判登記修正後的新架構三原命令、四檔及封存來源。"""
import hashlib
import json
from pathlib import Path
import re
import subprocess
import sys

HERE = Path(__file__).resolve().parent
SHA = sys.argv[1]
assert re.fullmatch('[0-9a-f]{40}', SHA)
PRODUCT = '52b951940f570011b257cd207741ba0a7f6d7ce9'
REPO = Path('D:/geometry-axiom-work/geo261-integration')
OUT = HERE / ('architecture-' + SHA[:8] + '-order-index')
FILES = {'product-architecture.json', 'product-architecture-scan.json',
         'product-architecture-receipt.json', 'PRODUCT-ARCHITECTURE.md'}


def read(path):
    return json.loads(path.read_text(encoding='utf-8-sig'))


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def main():
    assert not (OUT / 'verified.json').exists() and not (HERE / 'STOP-WSL').exists()
    assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=REPO, text=True).strip() == SHA
    assert set(subprocess.check_output(['git', 'diff', PRODUCT, SHA, '--name-only'], cwd=REPO, text=True).splitlines()) == {'tests/mutants_results.json'}
    run = read(OUT / 'receipt.json')
    audit = OUT / 'audit'
    receipt = read(audit / 'new/product-architecture-receipt.json')
    assert run['sha'] == receipt['sourceBaseHead'] == SHA and run['exit'] == 0 and receipt['scanPassed']
    assert digest((OUT / 'source.tar').read_bytes()) == run['archive_sha256']
    driver = Path('D:/geometry-axiom-work/geo261-research/refresh_product_architecture_next.py')
    assert digest(driver.read_bytes()) == run['driver_sha256'] == 'c786be4a0cd7288163c4004aeb06485997594d63a835f4e33a0dee6840b169c5'
    assert read(audit / 'RESULT.json')['allRawAndLfInputsUnchanged'] is True
    counts = receipt['counts']
    assert {k: counts[k] for k in ['executableFiles', 'ownedSources', 'exclusions', 'edges', 'diagnostics']} == dict(executableFiles=601, ownedSources=333, exclusions=268, edges=796, diagnostics=0)
    assert counts['layers'] == dict(K=69, W=67, H=191, vendor=6)
    assert receipt['manifestRefresh']['changedPreviouslySealedInputs'] == []
    assert not receipt['manifestRefresh']['staleExcludedVendorOrEmbedDigestsAfter']
    steps = []
    for name, total, skips in [('final-scan', None, 0), ('generic-tests', 205, 1), ('product-tests', 18, 0)]:
        row = read(audit / name / 'run.json')
        assert row['complete'] and row['exitCode'] == 0 and row['sourceBaseHead'] == SHA
        for key in ['stdout', 'stderr']:
            raw = (audit / name / key).read_bytes()
            assert digest(raw) == row[key]['rawSha256'] and len(raw) == row[key]['bytes']
        if total is None:
            scan = read(audit / name / 'stdout')
            assert scan['ok'] and scan['diagnostics'] == []
        else:
            text = (audit / name / 'stderr').read_text(encoding='utf-8')
            assert len(re.findall(r'^Ran ' + str(total) + r' tests in ', text, re.M)) == 1
            assert text.rstrip().endswith('OK (skipped=1)' if skips else 'OK')
        steps.append({k: row[k] for k in ['command', 'exitCode', 'seconds']})
    hashes = {}
    assert set(run['outputs']) == FILES
    for name, value in run['outputs'].items():
        raw = (REPO / 'geometry-core/domain-decoupling' / name).read_bytes()
        assert raw == (audit / 'new' / name).read_bytes() and digest(raw) == value
        hashes[name] = value
    for name, key in [('product-architecture.json', 'manifestSha256'), ('product-architecture-scan.json', 'scanSha256'), ('PRODUCT-ARCHITECTURE.md', 'reportSha256')]:
        assert digest((audit / 'new' / name).read_bytes().replace(b'\r\n', b'\n')) == receipt[key]
    changed = subprocess.check_output(['git', 'diff', '--name-only'], cwd=REPO, text=True).splitlines()
    assert set(changed) == {'geometry-core/domain-decoupling/PRODUCT-ARCHITECTURE.md', 'geometry-core/domain-decoupling/product-architecture-receipt.json'}
    result = {'passed': True, 'source_sha': SHA, 'counts': counts, 'steps': steps, 'output_hashes': hashes,
              'generic_tests': dict(total=205, passed=204, skipped=1), 'product_tests': dict(total=18, passed=18, skipped=0),
              'actual_changed_files': changed, 'verifier_sha256': digest(Path(__file__).read_bytes())}
    (OUT / 'verified.json').write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
    print(json.dumps(result, ensure_ascii=False))


if __name__ == '__main__':
    main()
