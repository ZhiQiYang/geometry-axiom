def stamp():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def sha_file(path):
    value = hashlib.sha256()
    with path.open('rb') as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b''):
            value.update(chunk)
    return value.hexdigest()


def save(path, value):
    with path.open('x', encoding='utf-8') as handle:
        handle.write(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


def load(path):
    return json.loads(Path(path).read_bytes())


def checked_result(path, source, count, code, failures=0):
    result = load(path)
    assert result['source_sha'] == source and result['exit'] == code
    assert result['tests_run'] == count and result['failures'] == failures
    assert result['errors'] == 0 and result['skipped'] == 0
    assert result['tracked_dirty'] == result['tracked_dirty_after'] == ''
    return result


def comparisons():
    candidate = Path('/dev/shm/pr348-profile-s796-n128')
    baseline = Path('/tmp/pr348-runs/baseline-profile-128')
    result = []
    for name in ('input.scene', 'stdout.bin', 'stderr.bin'):
        first, second = (candidate / name).read_bytes(), (baseline / name).read_bytes()
        assert first == second, name
        result.append({'name': name, 'candidate_path': str(candidate / name), 'baseline_path': str(baseline / name),
                       'bytes': len(first), 'sha256': hashlib.sha256(first).hexdigest(), 'bytes_equal': True})
    return {'candidate_source': S, 'baseline_source': BASE, 'checked_utc': stamp(),
            'comparisons': result, 'stdin_claimed_equal': False,
            'stdin_note': '原 stdin 的 loadfile 本地路徑不同，未宣稱 stdin bytes 相同；實際 scene bytes 全同。'}


def gather():
    rows, names = [], set()

    def add(path, member):
        path = Path(path)
        assert path.is_file() and not path.is_symlink(), ('不是原普通檔', str(path))
        assert '.git' not in path.parts, str(path)
        name = PurePosixPath(member)
        assert not name.is_absolute() and '..' not in name.parts and str(name) not in names
        names.add(str(name))
        rows.append((path, str(name)))

    for label, folder in DIRECTORIES.items():
        root = Path(folder)
        assert root.is_dir() and not root.is_symlink(), folder
        for path in sorted(root.rglob('*')):
            assert not path.is_symlink(), ('原資料有 symlink，須由主代理明確處理', str(path))
            if path.is_file():
                add(path, label + '/' + path.relative_to(root).as_posix())
    for path in EXECUTORS:
        add(path, 'executors/' + Path(path).name)
    for path in LAUNCH_STREAMS:
        add(path, 'launch-originals/' + Path(path).name)
    for path in TOOL_METADATA:
        add(path, 'local-observer-tool-metadata/' + Path(path).name)
    for name in BASE_PROFILE_FILES:
        add(Path('/tmp/pr348-runs/baseline-profile-128') / name, 'baseline-profile128-reference/' + name)
    for name in BASE_N256_FILES:
        add(Path('/dev/shm/pr348-tag-pair-256') / name, 'baseline-n256-reference/' + name)
    return rows
