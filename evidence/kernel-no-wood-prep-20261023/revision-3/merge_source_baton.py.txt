"""Merge only an explicitly handed-off #318 source and register the two tests.

No compiler/test/ledger execution. The integration repository is never written.
Invoke only after the predecessor owner has declared its source handoff.
"""
from pathlib import Path
import argparse
import ast
import datetime
import hashlib
import importlib.util
import json
import os
import re

ap = argparse.ArgumentParser(description=__doc__)
ap.add_argument('--baton-sha', required=True)
args = ap.parse_args()
assert re.fullmatch('[0-9a-f]{40}', args.baton_sha)
BASE = Path('/workspace/kernel-no-wood-prep')
WORK = Path('/workspace/kernel-no-wood-work')
URL = 'https://github.com/ZhiQiYang/geometry-axiom.git'
OWN = 'codex/kernel-without-woodworking-20261023'
PREDECESSOR = 'codex/issue318-reading-a-20261023'
CAPTURE = Path('/workspace/geo261-environment/final-f-contact-checker-proof-tools/capture.py')
assert hashlib.sha256(CAPTURE.read_bytes()).hexdigest() == 'c387835c03fce36be0861b7fa5228ae494ad2be94bfa9cd62d4f36b9782dc762'
OUT = BASE / ('source-baton-' + args.baton_sha[:8] + '-' + datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%SZ'))
OUT.mkdir()
spec = importlib.util.spec_from_file_location('no_wood_baton_capture', CAPTURE)
capture = importlib.util.module_from_spec(spec)
spec.loader.exec_module(capture)
recorder = capture.Recorder(OUT, dict(os.environ, LANG='C.UTF-8', LC_ALL='C.UTF-8', PYTHONDONTWRITEBYTECODE='1'), args.baton_sha)
report = {'schema': 'kernel-no-wood-source-baton-publication/1', 'predecessor_source_sha': args.baton_sha,
          'commands': recorder.rows, 'compiler_tests_or_ledger_executed': False,
          'force_push_or_rebase': False, 'integration_repository_written': False, 'errors': []}


def git(*argv):
    return recorder.run(['git', '-C', str(WORK), *argv], cwd=BASE, timeout=120,
                        source_variant='explicit-predecessor-source-handoff-only')


code = 1
try:
    assert git('branch', '--show-current')[1].decode().strip() == OWN
    assert not git('status', '--porcelain=v1', '--untracked-files=all')[1]
    report['own_head_before'] = git('rev-parse', 'HEAD')[1].decode().strip()
    before_remote = git('ls-remote', '--heads', URL, 'refs/heads/' + OWN)[1].decode().split()[0]
    assert before_remote == report['own_head_before'], 'Publish own pending commits before source handoff'
    git('fetch', '--no-tags', URL, 'refs/heads/' + PREDECESSOR)
    remote_parent = git('rev-parse', 'FETCH_HEAD')[1].decode().strip()
    git('merge-base', '--is-ancestor', args.baton_sha, remote_parent)
    report['predecessor_public_ref_head'] = remote_parent
    if args.baton_sha == 'ae74e51bba1efef48949c077635966273a9e8990':
        parents = git('rev-list', '--parents', '-n', '1', args.baton_sha)[1].decode().split()
        assert parents[1] == 'fdacdde4e11aae050fa98f4b22a057f8d90d0572'
        assert parents[2].startswith('6cad1a3c')
        changed_paths = [path.decode() for path in git('diff', '--name-only', '-z', parents[1], args.baton_sha)[1].split(b'\0') if path]
        assert all('evidence' in Path(path).parts for path in changed_paths), 'Unexpected product/test/mode change in source baton merge'
        git('diff', '--raw', parents[1], args.baton_sha)
        report['predecessor_merge_static'] = {'parents': parents[1:], 'changed_paths': changed_paths,
                                             'all_non_evidence_blobs_and_modes_identical': True}
    git('sparse-checkout', 'add', '/tests/run_all.expected.json')
    git('merge', '--no-ff', '-m', 'Merge approved Reading A source for isolated template verification', args.baton_sha)
    report['merge_sha'] = git('rev-parse', 'HEAD')[1].decode().strip()
    leaf = WORK / 'tests/test_rectangular_template.py'
    leaf_raw = leaf.read_bytes()
    assert hashlib.sha256(leaf_raw).hexdigest() == 'e948fb78e81a075875a45041009fb379e58d783c8d6b579cd9fa2f81cd40910e'
    methods = sum(isinstance(node, ast.FunctionDef) and node.name.startswith('test_') for node in ast.walk(ast.parse(leaf_raw)))
    assert methods == 13
    registry = WORK / 'tests/run_all.expected.json'
    original = registry.read_bytes()
    data = json.loads(original)
    suite = 'py:test_rectangular_template'
    assert data['suites'][suite] == {'mode': 'count', 'count': 11, 'unit': 'unittest'}
    pattern = rb'("py:test_rectangular_template":\s*\{\s*"mode": "count",\s*"count": )11(,)'
    changed, matches = re.subn(pattern, rb'\g<1>13\2', original)
    assert matches == 1
    expected = json.loads(original)
    expected['suites'][suite]['count'] = 13
    assert json.loads(changed) == expected
    registry.write_bytes(changed)
    (OUT / 'expected-before.json').write_bytes(original)
    (OUT / 'expected-after.json').write_bytes(changed)
    git('diff', '--check', '--', 'tests/run_all.expected.json')
    git('add', '--', 'tests/run_all.expected.json')
    git('commit', '-m', 'Register isolated template dependency guards')
    report['product_source_sha'] = git('rev-parse', 'HEAD')[1].decode().strip()
    assert not git('status', '--porcelain=v1', '--untracked-files=all')[1]
    git('push', URL, 'refs/heads/' + OWN + ':refs/heads/' + OWN)
    remote_after = git('ls-remote', '--heads', URL, 'refs/heads/' + OWN)[1].decode().split()[0]
    assert remote_after == report['product_source_sha']
    report.update(remote_exact_sha=remote_after, test_method_count=methods,
                  registry_count_before=11, registry_count_after=13, dynamic_proofs='UNRUN')
    code = 0
except BaseException as error:
    report['errors'].append(type(error).__name__ + ': ' + str(error))
finally:
    report['exit_code'] = code
    report['ended_utc'] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    capture.dump(OUT / 'receipt.json', report)
print(json.dumps({'out': str(OUT), 'exit_code': code, 'source_sha': report.get('product_source_sha'),
                  'dynamic_proofs': report.get('dynamic_proofs', 'UNRUN'), 'errors': report['errors']}, ensure_ascii=False))
raise SystemExit(code)
