def h(b):return hashlib.sha256(b).hexdigest()

def controls(text):
    env={}
    def v(x):
        if isinstance(x,ast.Constant):return x.value
        if isinstance(x,ast.Name):return env[x.id]
        if isinstance(x,(ast.Tuple,ast.List)):return tuple(v(y) for y in x.elts)
        if isinstance(x,ast.Call) and isinstance(x.func,ast.Attribute) and x.func.attr=='replace':
            assert not x.keywords;return v(x.func.value).replace(*(v(y) for y in x.args))
        raise AssertionError(ast.dump(x))
    for n in ast.parse(text).body:
        if isinstance(n,ast.Assign) and len(n.targets)==1 and isinstance(n.targets[0],ast.Name) and n.targets[0].id in ('COLLISION_FILTER','STOP_FILTER','CONTROLS'):
            env[n.targets[0].id]=v(n.value)
    return env['CONTROLS']
