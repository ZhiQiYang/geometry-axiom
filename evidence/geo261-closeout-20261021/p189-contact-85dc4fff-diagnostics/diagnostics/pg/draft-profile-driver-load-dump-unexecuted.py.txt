#!/usr/bin/env python3
"""獨立的單次 N64 診斷；正式樣本與 binary 完全分離。"""
from pathlib import Path
import datetime
import hashlib
import io
import json
import os
import resource
import signal
import subprocess
import sys
import tarfile
import tempfile
import time
import traceback

D = Path(__file__).resolve().parent
REPO = Path('/workspace/geometry-axiom')
FORMAL = Path('/workspace/geo261-evidence/p189-contact-v2-85dc4fff44ce-g3bqypyh')
L = '85dc4fff44ce49b27f831b21cbdcb662c863302c'

def now(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def sha(data): return hashlib.sha256(data).hexdigest()
def write(path, data):
    with path.open('xb') as f:
        f.write(data); f.flush(); os.fsync(f.fileno())
def save(path, data): write(path, (json.dumps(data, ensure_ascii=False, indent=2)+'\n').encode())

def command(name, argv, cwd, env, stdin=None, timeout=900):
    receipt = {'name': name, 'argv': argv, 'cwd': str(cwd), 'started_utc': now(),
               'source_sha': L, 'stack_limit_bytes': list(resource.getrlimit(resource.RLIMIT_STACK)),
               'stdin': str(stdin) if stdin else None}
    started = time.monotonic()
    input_stream = stdin.open('rb') if stdin else subprocess.DEVNULL
    try:
        with (D/(name+'.stdout')).open('xb') as output, (D/(name+'.stderr')).open('xb') as error:
            child = subprocess.Popen(argv, cwd=cwd, env=env, stdin=input_stream, stdout=output,
                                     stderr=error, start_new_session=True)
            receipt.update(pid=child.pid, pgid=os.getpgid(child.pid))
            save(D/(name+'.started.json'), receipt)
            timed_out = False
            try:
                code = child.wait(timeout=timeout)
            except subprocess.TimeoutExpired:
                timed_out = True
                os.killpg(child.pid, signal.SIGKILL)
                code = child.wait()
            write(D/(name+'.exit-code'), (str(code)+'\n').encode())
            output.flush(); error.flush(); os.fsync(output.fileno()); os.fsync(error.fileno())
    finally:
        if stdin: input_stream.close()
    receipt.update(finished_utc=now(), seconds=time.monotonic()-started, actual_exit_code=code,
                   timed_out=timed_out, stdout_sha256=sha((D/(name+'.stdout')).read_bytes()),
                   stderr_sha256=sha((D/(name+'.stderr')).read_bytes()))
    save(D/(name+'.receipt.json'), receipt)
    if code or timed_out: raise RuntimeError(f'{name} 失敗，實際結束碼 {code}，timeout={timed_out}')
    return receipt

def main():
    signal.signal(signal.SIGHUP, signal.SIG_IGN)
    resource.setrlimit(resource.RLIMIT_STACK, (16777216, 16777216))
    resource.setrlimit(resource.RLIMIT_CORE, (0, 0))
    env = os.environ.copy(); env.pop('PYTHONOPTIMIZE', None); env.pop('GMON_OUT_PREFIX', None)
    env.update(PYTHONDONTWRITEBYTECODE='1', PYTHONIOENCODING='utf-8')
    prepared = json.loads((FORMAL/'prepare-receipt.json').read_bytes())
    formal_binary = Path(prepared['work'])/L[:8]/'bridge'
    expected_formal_sha = json.loads((D/'prepare-receipt.json').read_bytes())['formal_binary_sha256']
    if sha(formal_binary.read_bytes()) != expected_formal_sha: raise RuntimeError('正式binary已改變')
    source = Path(tempfile.mkdtemp(prefix='geo261-profile-contact-v2-src.', dir='/workspace'))
    archive_argv = ['git', 'archive', '--format=tar', L, '--', 'current_geometry.c', 'geometry-core',
                    'geometry-app', 'geometry-templates', 'geometry-web',
                    ':(exclude)geometry-core/domain-decoupling', ':(exclude)geometry-web/docs',
                    ':(exclude)geometry-web/prototypes', ':(exclude)geometry-web/vendor',
                    ':(exclude)geometry-web/node_modules']
    archive_result = command('source-archive', archive_argv, REPO, env)
    archive = (D/'source-archive.stdout').read_bytes()
    with tarfile.open(fileobj=io.BytesIO(archive), mode='r:') as tf: tf.extractall(source, filter='data')
    index = json.loads((FORMAL/'source-files-final.json').read_bytes())
    for item in index:
        data = (source/item['path']).read_bytes()
        if len(data) != item['bytes'] or sha(data) != item['sha256']: raise RuntimeError('新archive來源不符：'+item['path'])
    scene = Path(prepared['work'])/'scenes'/'s64.scene'
    scene_bytes = scene.read_bytes(); write(D/'scene-n64.scene', scene_bytes)
    expected_scene = next(row['scene_sha256'] for row in json.loads((FORMAL/'dump-equivalence.json').read_bytes())['scenes'] if row['scene']=='s64.scene')
    if sha(scene_bytes) != expected_scene: raise RuntimeError('N64輸入與正式輸入不符')
    run = D/'run01'; run.mkdir()
    write(run/'load.in', ('loadfile '+str(scene)+'\ndump\n').encode())
    binary = D/'bridge-profile'
    compile_argv = ['gcc', '-std=c11', '-O2', '-g', '-pg', '-fno-omit-frame-pointer',
                    'geometry-web/web_bridge.c', '-o', str(binary), '-lm']
    save(D/'profile-environment.json', {'started_utc': now(), 'source_sha': L, 'source': str(source),
         'source_archive_sha256': sha(archive), 'source_file_count': len(index), 'source_files_match_formal': True,
         'compiler': subprocess.check_output(['gcc','--version'], env=env).decode(),
         'compile_argv': compile_argv, 'scene': str(scene), 'scene_sha256': expected_scene,
         'driver_sha256': sha(Path(__file__).read_bytes()), 'pid': os.getpid(), 'pgid': os.getpgrp(),
         'environment_removed': ['PYTHONOPTIMIZE','GMON_OUT_PREFIX'],
         'stack_limit_bytes': list(resource.getrlimit(resource.RLIMIT_STACK)),
         'core_limit_bytes': list(resource.getrlimit(resource.RLIMIT_CORE)),
         'is_formal_P189_sample': False, 'diagnostic_run_count_planned': 1, 'diagnostic_request': 'loadfile+dump'})
    build = command('profile-build', compile_argv, source, env)
    binary_sha = sha(binary.read_bytes())
    load = command('profile-load-run01', [str(binary)], run, env, run/'load.in')
    gmon = run/'gmon.out'
    if not gmon.is_file(): raise RuntimeError('沒有 gmon.out')
    payload = (D/'profile-load-run01.stdout').read_bytes()
    responses = [json.loads(line) for line in payload.splitlines()]
    if len(responses) != 2 or any(r.get('ok') is not True for r in responses): raise RuntimeError('診斷loadfile/dump失敗')
    expected_stdout = (FORMAL/'dump'/'s64.scene-final.stdout').read_bytes()
    if payload != expected_stdout: raise RuntimeError('診斷load輸出與正式N64輸出不同')
    profile = command('profile-gprof-run01', ['gprof','-b','-p','-q',str(binary),str(gmon)], D, env)
    command('profile-nm', ['nm','-anS',str(binary)], D, env)
    command('profile-objdump', ['objdump','-drwC',str(binary)], D, env)
    if sha(formal_binary.read_bytes()) != expected_formal_sha: raise RuntimeError('正式binary在診斷中改變')
    save(D/'profile-completion.json', {'completed_utc':now(), 'source_sha':L, 'source':str(source),
         'archive':archive_result, 'build':build, 'load':load, 'gprof':profile,
         'profile_binary_sha256':binary_sha, 'gmon_sha256':sha(gmon.read_bytes()),
         'scene_sha256':expected_scene, 'load_stdout_matches_formal':True,
         'formal_binary_sha256':expected_formal_sha, 'formal_binary_unchanged':True,
         'formal_samples_untouched':True, 'diagnostic_run_count':1, 'diagnostic_request':'loadfile+dump',
         'warning':'帶-pg的時間及取樣分布只供診斷，不能代替正式門檻；可能改變內聯、函式邊界和執行成本。'})

if __name__ == '__main__':
    try:
        main(); code=0
    except BaseException:
        write(D/'profile-driver.exception', traceback.format_exc().encode()); code=2
    write(D/'profile-driver.exit-code', (str(code)+'\n').encode())
    sys.exit(code)
