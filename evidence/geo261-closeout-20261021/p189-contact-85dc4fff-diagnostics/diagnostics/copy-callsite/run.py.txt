"""外部連結包裝診斷；一次負載，正式原件前後核對，真 wait4。"""
from pathlib import Path
import hashlib,json,os,resource,sys,tarfile,io,traceback
D=Path(__file__).resolve().parent
OLD=Path('/workspace/geo261-evidence/p189-pcdiag-85dc4fff-undef-gqq2eu1g')
helper=(OLD/'pcdiag-driver.py').read_bytes()
assert hashlib.sha256(helper).hexdigest()=='b38d023f0cc9da48d084e7a99762eddc22f53869bb480128b5bc4f516f4391ad'
ns={'__name__':'診斷共用方法','__file__':str(OLD/'pcdiag-driver.py')}
exec(compile(helper,str(OLD/'pcdiag-driver.py'),'exec'),ns);ns['D']=D
run,save,sha=ns['run'],ns['save'],ns['sha']
def main():
    c=json.loads((D/'released-config.json').read_bytes());assert c['released'] is True
    assert sha(Path(__file__).read_bytes())==c['driver_sha256'] and sha((D/'copy_wrap.c').read_bytes())==c['wrapper_sha256']
    assert not (D/'started.json').exists()
    resource.setrlimit(resource.RLIMIT_STACK,(16777216,16777216));resource.setrlimit(resource.RLIMIT_CORE,(0,0))
    env=os.environ.copy();env.pop('PYTHONOPTIMIZE',None);env.pop('GMON_OUT_PREFIX',None);env['PYTHONDONTWRITEBYTECODE']='1'
    formal=Path(c['formal_out']);binary=Path(c['formal_binary'])
    def unchanged():
        assert sha(binary.read_bytes())==c['formal_binary_sha256']
        for name in ('samples','receipt'):assert sha((formal/(name+'.json')).read_bytes())==c['formal_'+name+'_sha256']
    unchanged();save(D/'started.json',{'utc':ns['now'](),'pid':os.getpid(),'config':c,'stack_bytes':resource.getrlimit(resource.RLIMIT_STACK)})
    archive=run('archive',c['archive_argv'],c['repo'],env)
    source=D/'source';source.mkdir()
    with tarfile.open(fileobj=io.BytesIO((D/'archive.stdout').read_bytes())) as t:t.extractall(source,filter='data')
    index=json.loads((formal/'source-files-final.json').read_bytes())
    for item in index:
        b=(source/item['path']).read_bytes();assert len(b)==item['bytes'] and sha(b)==item['sha256']
    build=run('build',c['compile_argv'],source,env)
    scene=Path(c['scene']);assert sha(scene.read_bytes())==c['scene_sha256']
    (D/'load.in').write_text('loadfile '+str(scene)+'\n');env['GEO261_COPY_LOG']=str(D/'calls.tsv')
    load=run('load',[str(D/'bridge-copy')],D,env,D/'load.in')
    assert (D/'load.stdout').read_bytes()==(formal/'timing/n64-r0-final.stdout').read_bytes()
    run('nm',['nm','-anS',str(D/'bridge-copy')],D,env)
    run('elf',['readelf','-h',str(D/'bridge-copy')],D,env)
    symbols=[]
    for line in (D/'nm.stdout').read_text().splitlines():
        p=line.split()
        if len(p)==4 and p[2] in ('T','t') and int(p[1],16):symbols.append((int(p[0],16),int(p[0],16)+int(p[1],16),p[3]))
    assert 'EXEC (Executable file)' in (D/'elf.stdout').read_text()
    rows=[]
    for line in (D/'calls.tsv').read_text().splitlines()[1:]:
        kind,pc,calls,size=line.split('\t')
        if kind=='lost':assert int(calls)==0 and int(size)==0;continue
        address=int(pc,16);rows.append({'kind':int(kind),'return_pc':pc,'calls':int(calls),'bytes':int(size),'exact_symbol_ranges':[name for start,end,name in symbols if start<=address-1<end]})
    unchanged();save(D/'summary.json',{'source_sha':c['source_sha'],'archive':archive,'build':build,'load':load,'rows':sorted(rows,key=lambda x:x['bytes'],reverse=True),'stdout_matches_formal':True,'formal_unchanged':True,'source_files':len(index),'limitations':'-no-pie 與連結包裝只用於呼叫點診斷，非原執行檔／性能成績；只計可執行檔的外部呼叫，未計 inline 操作或 libc 內部呼叫；tail call 的 return_pc 可能屬上一層。'})
if __name__=='__main__':
    try:main();code=0
    except BaseException:(D/'exception.txt').write_text(traceback.format_exc());code=2
    (D/'exit-code').write_text(str(code)+'\n');sys.exit(code)
