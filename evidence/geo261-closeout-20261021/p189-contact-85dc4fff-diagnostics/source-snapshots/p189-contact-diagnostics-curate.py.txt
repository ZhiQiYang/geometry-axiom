def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def readj(path):
    return json.loads(path.read_bytes())

def save(path, obj):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('x') as stream:
        json.dump(obj, stream, ensure_ascii=False, indent=2)
        stream.write('\n')

def copy(src, rel, role):
    raw = src.read_bytes()
    dst = B / rel
    dst.parent.mkdir(parents=True, exist_ok=True)
    with dst.open('xb') as stream:
        stream.write(raw)
    os.chmod(dst, 0o644)
    assert dst.read_bytes() == raw
    mapping.append({'source': str(src), 'published': str(rel), 'bytes': len(raw),
                    'sha256': digest(raw), 'original_bytes': True, 'role': role})

def tree(src, prefix):
    for current, dirs, files in os.walk(src):
        dirs.sort()
        for name in list(dirs):
            if name in ('source', '__pycache__', '.git'):
                dirs.remove(name)
                excluded.append({'source': str(Path(current) / name), 'kind': 'directory',
                                 'reason': '來源整樹／暫存不複製；來源由固定 commit 與 archive 收據重建'})
        for name in sorted(files):
            file = Path(current) / name
            rel = file.relative_to(src)
            assert not file.is_symlink(), file
            raw = file.read_bytes()
            if name in ('archive.stdout', 'source-archive.stdout') or raw.startswith(b'\x7fELF'):
                excluded.append({'source': str(file), 'bytes': len(raw), 'sha256': digest(raw),
                                 'reason': 'ELF 或來源 tar 原件只留外部；未作文字證據冒充'})
                continue
            if name == 'gmon.out':
                encoded = base64.b64encode(raw) + b'\n'
                dst = B / prefix / (str(rel) + '.base64')
                dst.parent.mkdir(parents=True, exist_ok=True)
                with dst.open('xb') as stream:
                    stream.write(encoded)
                assert base64.b64decode(dst.read_bytes()) == raw
                mapping.append({'source': str(file), 'published': str(dst.relative_to(B)),
                                'original_bytes': False, 'encoding': 'base64', 'raw_bytes': len(raw),
                                'raw_sha256': digest(raw), 'encoded_sha256': digest(encoded),
                                'decode_matches_original': True,
                                'role': '明示無損編碼的二進位資料；原 gmon 留外部'})
                continue
            raw.decode('utf-8')
            published = Path(prefix) / rel
            role = '既有文字原件；狀態依原收據，含失敗及草稿'
            if file.suffix in ('.py', '.h', '.c', '.sh'):
                published = Path(str(published) + '.txt')
                role = '外部輔助／來源的原 bytes 純文字快照，0644；非部署入口；未執行草稿依原檔名及收據'
            copy(file, published, role)
