constants = {}
for node in ast.parse(verifier_raw).body:
    if isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name):
        if node.targets[0].id in {'FORBIDDEN_SUFFIXES', 'EXECUTABLE_MAGICS'}:
            constants[node.targets[0].id] = ast.literal_eval(node.value)
raws, metadata = {}, {}
for directory, dirs, names in os.walk(ROOT, followlinks=False):
    assert all(not (Path(directory) / d).is_symlink() for d in dirs)
    for name in names:
        path = Path(directory) / name
        rel = path.relative_to(ROOT).as_posix()
        st = path.lstat()
        assert stat.S_ISREG(st.st_mode) and stat.S_IMODE(st.st_mode) == 0o644
        assert path.suffix.lower() not in constants['FORBIDDEN_SUFFIXES']
        assert name not in {'.gitignore', '.gitattributes'}
        raw = path.read_bytes()
        assert not any(raw.startswith(m) for m in constants['EXECUTABLE_MAGICS'])
        raw.decode('utf-8')
        raws[rel] = raw
        metadata[rel] = {'bytes': len(raw), 'sha256': sha(raw), 'mtimeNs': st.st_mtime_ns}
assert len(raws) == 2409 and sum(len(r) for r in raws.values()) == 45078348

    seen.add(rel)
    assert metadata[rel]['bytes'] == row['bytes'] and metadata[rel]['sha256'] == row['sha256']
    original = Path(row['source'])
    assert original.is_file() and not original.is_symlink() and original.read_bytes() == raws[rel]
assert set(raws) - seen == {'README.md', '原件清單.json', '未收錄清單.json', 'SHA256SUMS'}
sum_records = []
for name in sorted(n for n in raws if PurePosixPath(n).name == 'SHA256SUMS'):
    parent = PurePosixPath(name).parent
    prefix = '' if str(parent) == '.' else str(parent) + '/'
    wanted = {}
    for line in raws[name].decode().splitlines():
        m = re.fullmatch(r'([0-9a-f]{64})  (.+)', line)
        assert m and m[2] not in wanted and '..' not in PurePosixPath(m[2]).parts and not PurePosixPath(m[2]).is_absolute()
        wanted[m[2]] = m[1]
    actual = {n[len(prefix):] for n in raws if n.startswith(prefix)} - {'SHA256SUMS'}
    assert set(wanted) == actual, name
    assert all(metadata[prefix + n]['sha256'] == digest for n, digest in wanted.items())
    sum_records.append({'path': name, 'entries': len(wanted), 'exactSetAndHashes': True})
assert next(r for r in sum_records if r['path'] == 'SHA256SUMS')['entries'] == 2408

git_records = []
def git(label, *args):
    argv = ['git', '-C', '/workspace/geometry-axiom', *args]
    r = subprocess.run(argv, capture_output=True)
    (OUT / (label + '.stdout')).write_bytes(r.stdout)
    (OUT / (label + '.stderr')).write_bytes(r.stderr)
    (OUT / (label + '.exit-code')).write_text(str(r.returncode) + '\n')
    git_records.append({'argv': argv, 'exitCode': r.returncode})
    dump('git-commands.json', git_records)
    assert r.returncode == 0
    return r.stdout
assert git('L3-parent', 'rev-parse', L3 + '^').decode().strip() == S3
