"""只讀核對既有 codec 定向原件並新建純資料證據包；不執行產品。"""
import datetime
import hashlib
import json
from pathlib import Path
import subprocess

E = Path('/workspace/geo261-evidence')
S = E / 'wd-read-candidate'
D = E / 'final-bundle/codec-targeted-7ad3216a'
sha = lambda data: hashlib.sha256(data).hexdigest()
def js(value):
    return (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()
def put(path, raw):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('xb') as f:
        f.write(raw)
    path.chmod(0o644)

assert not D.exists(), '不得覆寫既有包'
r = json.loads((S / 'targeted/receipt.json').read_bytes())
c = json.loads((S / 'commit-receipt.json').read_bytes())
b = json.loads((S / 'targeted/build-log.json').read_bytes())
u = json.loads((S / 'targeted/unit-result.json').read_bytes())
t = json.loads((S / 'targeted/selected-tests.json').read_bytes())
assert r['rawWaitStatus'] == r['exitCode'] == u['exitCode'] == 0
assert (S / 'targeted/exit-code').read_text() == '0\n'
assert u['testsRun'] == u['selectedCount'] == len(t['resolvedTests']) == 14
assert not any(u[k] for k in ('failures','errors','skipped','expectedFailures','unexpectedSuccesses'))
assert len(b) == 88 and all(x['exit'] == 0 for x in b)
for name,key in [('run_targeted.py','runnerSha256'),('targeted_tests.py','testDriverSha256'),('targeted/stdout.log','stdoutSha256'),('targeted/stderr.log','stderrSha256'),('candidate.patch','patchSha256'),('definition_sources-candidate-code.txt','candidateRawSha256')]:
    assert sha((S / name).read_bytes()) == r[key]
argv = ['git','-C','/workspace/geometry-axiom','show',c['sourceSha'] + ':geometry-templates/woodworking/native/definition_sources.c']
g = subprocess.run(argv, capture_output=True)
assert g.returncode == 0
assert g.stdout == (S / 'definition_sources-candidate-code.txt').read_bytes()
assert sha(g.stdout) == c['sourceRawSha256'] == r['candidateRawSha256'] == u['build']['actualDefinitionSourceSha256']
originals = sorted(x for x in S.rglob('*') if x.is_file())
assert len(originals) == 18 and not any(x.is_symlink() for x in originals)
mapping = []
for f in originals:
    rel = f.relative_to(S).as_posix()
    target = '原件/' + rel + ('.txt' if f.suffix == '.py' else '')
    raw = f.read_bytes()
    put(D / target, raw)
    mapping.append(dict(source=str(f), destination=target, bytes=len(raw), sha256=sha(raw), originalBytes=True, kind='執行方法原碼的純文字資料快照' if f.suffix == '.py' else '既有原件'))
binding = dict(checkedUTC=datetime.datetime.now(datetime.timezone.utc).isoformat(), kind='提交後唯讀 bytes 核對，不是重跑測試', executionHead=r['sourceSha'], executionDirtyPath='geometry-templates/woodworking/native/definition_sources.c', trialBranch=c['branch'], trialSha=c['sourceSha'], argv=argv, actualExitCode=g.returncode, stderr=g.stderr.decode(), stdoutBytes=len(g.stdout), stdoutSha256=sha(g.stdout), savedStdout='原件/definition_sources-candidate-code.txt', candidateEqualsCommittedBlob=True, exactTestedSourceSha256=u['build']['actualDefinitionSourceSha256'])
put(D / '提交後來源綁定.json', js(binding))
put(D / '來源索引.json', js(dict(sourceRoot=str(S), originalFileCount=18, originalBytes=sum(x['bytes'] for x in mapping), files=mapping)))
put(D / '封裝方法.py.txt', Path(__file__).read_bytes())
readme = '''這是 codec 快速解碼候選的歷史定向驗證資料，14／14 通過，failure／error／skip 皆 0，88 次編譯與連結紀錄皆為 0。原件沒有產品重跑或補造。

實跑於 detached `43f85313c32410ac01406ab585846bbacafea278` 加唯一 `definition_sources.c` 補丁；之後同 bytes 提交至分支 `codex/geo261-p189-codec-trial-20261021`、SHA `7ad3216ac3ea00bec0adac7b38c3a3daf0a39d09`。這不是在 7ad 提交後重跑。`提交後來源綁定.json` 另保存本次 git show 真命令、退出碼及與實跑候選相同的 SHA-256。

原方法入口為 `env -u CC -u PYTHONOPTIMIZE python3 -B /workspace/geo261-evidence/wd-read-candidate/run_targeted.py`；實際 wait 的子命令是 `timeout 900 python3 -B /workspace/geo261-evidence/wd-read-candidate/targeted_tests.py`。cwd、環境及 16 MiB stack 以 `原件/targeted/command.json` 為準；`receipt.json` 保存 rawWaitStatus=0、exitCode=0、來源前後相同及 9.15574672299772 秒。開始 2026-10-21T01:02:05.425033+00:00，結束 01:02:14.581042+00:00。

完整測試層 stdout／stderr 原 bytes、14 個具名 test id、unit 結果、88 筆建置命令與各自 stdout／stderr／exit 均保留。build-log 是既有 NativeBuild 的文字紀錄；沒有另存各測試內每個執行子程序完整原始 wire stream，也沒有獨立工具傳輸 raw 日誌，不能把 unittest 的 OK 擴寫成這些不存在的原件。

沿用五個 definition 具名負控制、四個 H 返回值控制及 quantity 變異案例，未新增 decoder 專屬負控制。快速路徑撤回舊 loop 應維持功能綠，效能裁判另見既有性能包；本包不證明性能門檻或最終來源的七項完整驗收。完整 prefix 截斷、尾隨 byte、sticky failure／bounds 的最小專屬補測不在這 14 案範圍。

`原件/README.md` 和靜態審查保留當時狀態，其中「未推送／未跑性能」是歷史說明，不是當前主分支狀態。所有 18 個既有檔案原 bytes 都列於來源索引；三個 .py 只改副本路徑為 .py.txt、以 0644 保存為資料快照，未新增執行入口。沒有 binary、archive、pycache。SHA256SUMS 覆蓋本包其他每一檔；舊凍結包未改。
'''
put(D / 'README.md', readme.encode())
files = sorted(x for x in D.rglob('*') if x.is_file())
put(D / 'SHA256SUMS', ''.join(f'{sha(f.read_bytes())}  {f.relative_to(D).as_posix()}\n' for f in files).encode())
for item in mapping:
    assert (D / item['destination']).read_bytes() == Path(item['source']).read_bytes()
allfiles = sorted(x for x in D.rglob('*') if x.is_file())
assert all((f.stat().st_mode & 0o777) == 0o644 for f in allfiles)
report = dict(bundle=str(D), fileCount=len(allfiles), totalBytes=sum(f.stat().st_size for f in allfiles), originalFileCount=len(mapping), originalBytes=sum(x['bytes'] for x in mapping), allOriginalCopiesIdentical=True, sumsSha256=sha((D / 'SHA256SUMS').read_bytes()), productRuns=0, changedOldBundles=False)
put(E / 'final-bundle/codec-targeted-7ad3216a-copy-check.json', js(report))
print(json.dumps(report, ensure_ascii=False))
