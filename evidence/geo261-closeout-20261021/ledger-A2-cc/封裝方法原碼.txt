"""只讀整理已完成 A2 帳本與 L2 發布原件；不執行任何產品測試。"""
import argparse
import datetime
import hashlib
import json
from pathlib import Path
import stat

BASE=Path('/workspace/geo261-evidence')
RUN=BASE/'ledger-A2-cc-cbefb39b'
PUBLISH=RUN/'publish'
PREP=BASE/'ledger-L2-preparation'
DEST=BASE/'final-bundle/ledger-A2-cc'
A2='cbefb39b47fb0edcfaf003d2c1101929fea244c8'
L2='249571fe2477845d5330f6190dad965bf327f43b'
LEDGER_SHA='6bda315331050bd3478ed65211a8263508cb1a1f6f67cee3f35df377506e7f0b'

def sha(raw):return hashlib.sha256(raw).hexdigest()
def load(path):return json.loads(path.read_bytes())
def files(root):
    result=[]
    for path in sorted(root.rglob('*')):
        assert not path.is_symlink(),str(path)
        if path.is_file():result.append(path)
    return result
def save(path,value):
    with path.open('x',encoding='utf-8') as f:
        json.dump(value,f,ensure_ascii=False,indent=2);f.write('\n')
    path.chmod(0o644)

def main():
    if not __debug__:raise RuntimeError('禁止停用 assert')
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--execute',action='store_true')
    parser.add_argument('--peer',type=Path)
    args=parser.parse_args()
    if not args.execute:parser.error('須明確指定 --execute，尚未建立新包')
    assert not DEST.exists()
    raw=(RUN/'mutants_results.json').read_bytes()
    assert sha(raw)==LEDGER_SHA
    validation=load(RUN/'ledger-validation.json')
    assert validation['source_sha']==A2 and validation['ledger_sha256']==LEDGER_SHA
    assert validation['record_count']==78 and validation['execution_count']==75
    assert validation['verdict_counts']=={'killed':70,'crashed':1,'equivalent':4,'retired':3}
    assert not load(RUN/'verdict-difference-review.json')['differences']
    receipt=load(BASE/'ledger-L2-publication/receipt.json')
    assert receipt['sourceA2']==receipt['parent']==A2 and receipt['ledgerL2']==L2
    assert (BASE/'ledger-L2-publication/ledger-blob.stdout').read_bytes()==raw
    assert (BASE/'ledger-L2-publication/parent.stdout').read_text().strip()==A2
    assert (BASE/'ledger-L2-publication/A-to-L-files.stdout').read_text().splitlines()==['tests/mutants_results.json']
    assert (BASE/'ledger-L2-publication/command-source.txt').read_bytes()==(PREP/'import_ledger.py').read_bytes()
    for name in ('mutate','check','check-killers','supervisor','supervisor-observed'):
        assert (RUN/(name+'.exit-code')).read_text().strip()=='0'
    expected_publish={}
    for line in (PUBLISH/'SHA256SUMS').read_text().splitlines():
        digest,name=line.split('  ',1);assert name not in expected_publish
        expected_publish[name]=digest
    actual_publish={p.relative_to(PUBLISH).as_posix() for p in files(PUBLISH)}
    assert set(expected_publish)==actual_publish-{'SHA256SUMS'}
    sources=[]
    def add(path,relative,group):
        assert path.is_file() and not path.is_symlink()
        sources.append((path,Path(relative),group))
    for path in files(PUBLISH):
        relative=path.relative_to(PUBLISH)
        if relative.as_posix()!='SHA256SUMS':assert sha(path.read_bytes())==expected_publish[relative.as_posix()]
        if len(relative.parts)==1 and (RUN/path.name).is_file():assert path.read_bytes()==(RUN/path.name).read_bytes()
        add(path,Path('執行原件')/relative,'已凍結 publish 原件')
    for path in sorted(RUN.iterdir()):
        if not path.is_file():continue
        if (PUBLISH/path.name).exists():continue
        add(path,Path('後補收據')/path.name,'publish 完成後新增，未改舊索引')
    for path in files(PREP):
        relative=path.relative_to(PREP)
        if path.suffix=='.py':relative=relative.with_suffix('.py.txt')
        add(path,Path('準備方法')/relative,'外部方法原 bytes，僅文字保存')
    for dirname,target,group in [
        ('source-A2-publication','A2發布','來源公開原件'),
        ('ledger-L2-publication','L2發布','only-ledger 發布原件'),
        ('ledger-cc-launcher-selftest.r3o_cexg','工具歷史自測','相同 supervisor 的歷史自測，非 A2 當輪執行')]:
        for path in files(BASE/dirname):add(path,Path(target)/path.relative_to(BASE/dirname),group)
    add(BASE/'ledger-L2-importer-tool-observation.json','L2發布外層工具觀察.json','root 實際工具回覆轉錄，非 raw transport／非重新 wait')
    add(BASE/'ledger-L2-readonly-plan.md','歷史準備計畫.md','A2 尚未產生時的計畫原件')
    add(BASE/'ledger-cc-supervisor.py','原工具/ledger-cc-supervisor.py.txt','未修改原 supervisor')
    add(BASE/'ledger-cc-validate-and-package.py','原工具/ledger-cc-validate-and-package.py.txt','原驗證函式，當輪只重綁來源與路徑')
    add(BASE/'import_default_cc_ledger.py','原工具/import_default_cc_ledger.py.txt','歷史匯入器原件，當輪不直接執行')
    add(Path(__file__),'封裝方法原碼.txt','外部封裝方法，不是另一產品工具')
    for name in ('static-check-code.txt','static-check.json','negative.stdout','negative.stderr'):
        add(Path(__file__).parent/name,Path('封裝靜態核對')/name,'封裝方法的靜態／早拒絕核對，沒有產品執行')
    if args.peer:
        assert args.peer.is_file()
        if not any(path.resolve()==args.peer.resolve() for path,_,_ in sources):
            add(args.peer,Path('獨立審查')/args.peer.name,'獨立唯讀審查原件')
    assert len({str(rel) for _,rel,_ in sources})==len(sources)
    DEST.mkdir(parents=True)
    index=[]
    for source,relative,group in sources:
        data=source.read_bytes();target=DEST/relative;target.parent.mkdir(parents=True,exist_ok=True)
        with target.open('xb') as f:f.write(data)
        target.chmod(0o644);assert target.read_bytes()==data
        index.append({'source':str(source),'path':relative.as_posix(),'bytes':len(data),'sha256':sha(data),'group':group})
    save(DEST/'原件清單.json',index)
    save(DEST/'未收錄清單.json',[
        {'source':'當輪 cache／worktree 的可執行 binary、object、暫存檔','reason':'只收原工具引用的 raw／identity／sidecar；binary hash 已由正式 validator 核對，交付不複製可執行檔。'},
        {'source':'核心57筆的每個子程序原 stdout/stderr','reason':'正式工具沒有另存此 raw，只有結構化 result 與 SHA256；不能補造或由頂層輸出取代。'},
        {'source':'其他 final-bundle 舊包與 L2 原生／browser／P189 結果','reason':'本包只涵蓋 A2 帳本與 A2→L2 發布；不改舊包、不宣稱全驗收完成。'}])
    readme=f'''# A2 全量帳本與 L2 發布證據

分支為 `codex/geo261-closeout-20261021`。完整 ledger 實際來源是 A2 `{A2}`；其後 L2 `{L2}` 唯一父提交為 A2，且只變更 `tests/mutants_results.json`。75 筆 execution.head 保留真正 A2，沒有改標成在 L2 重跑。

`python3 -B -u tests/mutate.py`、`python3 -B tests/mutate.py --check`、`python3 -B tests/mutate.py --check --killers` 三階段均真碼 0；supervisor finally、父 launcher 真 wait、互動工具最後回覆及必要 validator 也都是 0。78 筆為 70 killed、僅 M20 crashed、4 equivalent、3 retired；逐 ID verdict 與 A2 原帳本完全一致。M20 是同場景三次 -11，屬偵測到崩潰，不能稱斷言殺死。

原帳本 5,170,289 bytes，SHA256 `{LEDGER_SHA}`。57 個核心 execution 的基準為 165/165；18 個 bridge execution 的 baseline/restored 各完整 27 方法，還原 bytes 與 binary identity 已核。全新 detached A2 worktree／空 cache，CC 與 PYTHONOPTIMIZE absent，預設 cc argv，stack 16 MiB、core 0。362 份來源／oracle 原件、184 個 source closure 檔案和 165 個場景均在執行前綁定 Git A2。

03:25 左右曾發生工具傳輸短暫斷線，原進程持續，沒有重啟 writer。03:26 原 session 仍可讀，最後 session 12445/chunk706db3 真 exit 0。啟動收據未保存 starttime tick，後來的程序 tick 只能配合 PID／argv／父子鏈與 boot 時間對照，不能冒稱有歷史 tick 逐值比對。相關原觀察完整保留。

L2 發布保存了逐命令 raw、終了碼、原 JSON Git blob、default check 0、普通 push 0 與 remote exact。最外層 importer 工具觀察由 root 轉錄實際回覆，清楚保留不是 raw transport／不是重新 wait 的界線。此包不重新執行 ledger、check、發布或產品測試。

獨立審查定版接受器真碼 0；之前兩次接受器核對誤判也保原碼／工具觀察：第一次錯把只保尾 3000 字的 failure.detail 當完整輸出，第二次未處理 FAIL 標頭後的測試說明文字。改核完整 mutant stdout 的具名 FAIL／AssertionError 後通過；沒有重跑或改寫正式 ledger。獨審沒有再雜湊 75 個 cache binary，此項由既有正式 validator 核對，不能把兩者說成同一次檢查。

核心原始證據的限制：既有工具沒有另外保存每個核心子程序的原 stdout/stderr，只保存結構化 oracle result 與其 SHA256。頂層 mutate.stdout 無法補足缺口。bridge 及 sidecar 原件依正式工具紀錄逐 bytes 收錄。工具歷史自測是相同 supervisor 的先前自測，不冒稱 A2 當輪重做。

`執行原件` 是原 frozen publish 的完整快照，保留其原 publication-index 與 SHA256SUMS。`後補收據` 收入 publish 完成後新增的真碼／審查，不修改原包。歷史準備計畫、啟動時方法與 A2 發布收據保留當時「尚未完成」的文字；本 README 說明此次帳本已完成，不據此宣稱 L2 其他驗收已完成。

本包原件清單只列逐 bytes 複製的原件，不把自己列入自己的雜湊；根 SHA256SUMS 精確涵蓋除自己以外的全部檔案，包含生成說明與巢狀索引。所有檔案 mode 為 0644；外部 Python 方法以 .txt 保原 bytes，沒有新增產品工具或可執行副本。封裝／獨審最終退出碼與本包驗證收據另置包外，避免自含循環。
'''
    (DEST/'README.md').write_text(readme,encoding='utf-8');(DEST/'README.md').chmod(0o644)
    (DEST/'來源對照.md').write_text('''# 原件來源對照

- `執行原件/`：當輪 frozen publish；其 publication-index 保留 cache／原輸出路徑鏈。
- `後補收據/`：publish 完成後新增原件，與歷史索引分開。
- `準備方法/`：prepare／launch／validate／import 的原方法及初版／修版紀錄，沒有重執行。
- `A2發布/`、`L2發布/`：實際公開來源與 only-ledger 發布的原 stdout／stderr／命令／碼。
- `工具歷史自測/`、`原工具/`：原方法與歷史自測，日期與來源保持原樣。
- 每個 copied path 的真正來源、bytes、SHA256 由根 `原件清單.json` 唯一列出；根 sums 另涵蓋生成文件。
''',encoding='utf-8');(DEST/'來源對照.md').chmod(0o644)
    paths=files(DEST)
    assert all(p.suffix not in {'.py','.js','.cjs'} and stat.S_IMODE(p.stat().st_mode)==0o644 for p in paths)
    (DEST/'SHA256SUMS').write_text(''.join(sha(p.read_bytes())+'  '+p.relative_to(DEST).as_posix()+'\n' for p in paths),encoding='utf-8')
    (DEST/'SHA256SUMS').chmod(0o644)
    allfiles=files(DEST)
    result={'completed_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'destination':str(DEST),
      'copied_originals':len(index),'files':len(allfiles),'bytes':sum(p.stat().st_size for p in allfiles),
      'checksum_entries':len(paths),'index_sha256':sha((DEST/'原件清單.json').read_bytes()),
      'sums_sha256':sha((DEST/'SHA256SUMS').read_bytes()),'product_tests_rerun':False}
    save(BASE/'ledger-A2-bundle-preparation/bundle-result.json',result)
    print(json.dumps(result,ensure_ascii=False))

if __name__=='__main__':main()
