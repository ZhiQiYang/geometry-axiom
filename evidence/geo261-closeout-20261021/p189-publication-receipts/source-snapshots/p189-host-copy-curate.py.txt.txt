def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def save(path, value):
    with path.open('x') as f:
        json.dump(value, f, ensure_ascii=False, indent=2)
        f.write('\n')

def regulars(path):
    for current, dirs, names in os.walk(path):
        dirs[:] = sorted(x for x in dirs if x not in ('__pycache__', '.git', 'source', 'footprint'))
        for name in sorted(names):
            p = Path(current) / name
            assert not p.is_symlink(), p
            yield p

def bundle(name, sources, readme, links=()):
    B = ROOT / name
    assert not B.exists(), B
    B.mkdir()
    mapping, excluded = [], []
    for src, rel in sources:
        assert 'footprint' not in src.relative_to(E).parts, src
        raw = src.read_bytes()
        if raw.startswith(b'\x7fELF') or src.suffix in ('.pyc', '.tar'):
            excluded.append({'source': str(src), 'bytes': len(raw), 'sha256': sha(raw), 'reason': '二進位／archive／暫存留外部'})
            continue
        raw.decode('utf-8')
        role = '既有文字原件；歷史狀態依收據，不因封裝改寫'
        if src.suffix in ('.py', '.c', '.h', '.inc', '.sh'):
            rel = Path(str(rel) + '.txt')
            role = '外部原 bytes 純文字來源快照；0644 資料，不是 active 入口'
        dst = B / rel
        dst.parent.mkdir(parents=True, exist_ok=True)
        with dst.open('xb') as f:
            f.write(raw)
        os.chmod(dst, 0o644)
        assert dst.read_bytes() == raw
        mapping.append({'source': str(src), 'published': str(rel), 'bytes': len(raw), 'sha256': sha(raw), 'role': role, 'original_bytes': True})
    save(B / 'artifact-mapping.json', {'files': mapping, 'excluded': excluded, 'new_product_executions': 0})
    if links:
        linked = []
        for directory in links:
            m = ROOT / directory / 'SHA256SUMS'
            linked.append({'bundle': '../' + directory, 'manifest': '../' + directory + '/SHA256SUMS', 'manifest_sha256': sha(m.read_bytes()) if m.exists() else None})
        save(B / 'linked-bundles.json', linked)
    (B / 'README.md').write_text(readme)
    files = sorted(x for x in B.rglob('*') if x.is_file())
    (B / 'SHA256SUMS').write_text(''.join(sha(x.read_bytes()) + '  ' + str(x.relative_to(B)) + '\n' for x in files))
    for row in mapping:
        assert (B / row['published']).read_bytes() == Path(row['source']).read_bytes()
    for line in (B / 'SHA256SUMS').read_text().splitlines():
        h, rel = line.split('  ', 1)
        assert sha((B / rel).read_bytes()) == h
    files = sorted(x for x in B.rglob('*') if x.is_file())
    assert not any(x.suffix in ('.py', '.c', '.h', '.inc', '.sh') for x in files)
    result = {'checked_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'bundle': str(B),
              'file_count': len(files), 'bytes': sum(x.stat().st_size for x in files),
              'manifest_sha256': sha((B / 'SHA256SUMS').read_bytes()), 'all_original_bytes_match': True,
              'all_manifest_hashes_match': True, 'new_product_executions': 0, 'validation_exit_code': 0,
              'packager_sha256': sha(Path(__file__).read_bytes())}
    save(E / (name + '-copy-check.json'), result)
    return result
