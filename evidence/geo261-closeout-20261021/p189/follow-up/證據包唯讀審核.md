import hashlib, json, statistics
p=Path('/workspace/geo261-evidence/final-bundle/p189')

for f in s.rglob('*'):
    if f.is_file(): assert f.read_bytes()==(p/f.relative_to(s)).read_bytes()

r=load('receipt.json'); rows=load('samples.json'); e=load('dump-equivalence.json')

    assert med==item['median_seconds']
    assert med['final']/med['main']==item['final_main_ratio']
all_rows=rows+[v for scene in e['scenes'] for v in scene['sides'].values()]
assert len(all_rows)==561
for x in all_rows:
    b=(p/x['stdout_file']).read_bytes()
    assert (p/x['stderr_file']).is_file()
