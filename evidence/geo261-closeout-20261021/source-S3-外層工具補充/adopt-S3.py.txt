from pathlib import Path
import datetime, hashlib, json, subprocess
ROOT = Path('/workspace/geometry-axiom')
OUT = Path('/workspace/geo261-evidence/source-S3-adoption')
SOURCE = '19401ca5448e162ee6140fed37b64b83b31ddec8'
PARENT = '249571fe2477845d5330f6190dad965bf327f43b'
BRANCH = 'codex/geo261-closeout-20261021'
OUT.mkdir(exist_ok=False)
commands = []
def run(label, *args):
    p = subprocess.run(args, cwd=ROOT, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    (OUT / (label + '.stdout')).write_bytes(p.stdout)
    (OUT / (label + '.stderr')).write_bytes(p.stderr)
    (OUT / (label + '.exit-code')).write_text(str(p.returncode) + '\n')
    commands.append({'label': label, 'argv': list(args), 'exitCode': p.returncode})
    (OUT / 'commands.json').write_text(json.dumps(commands, ensure_ascii=False, indent=2) + '\n')
    assert p.returncode == 0, (label, p.returncode)
    return p.stdout
assert run('head-before', 'git', 'rev-parse', 'HEAD').decode().strip() == PARENT
assert run('branch', 'git', 'branch', '--show-current').decode().strip() == BRANCH
assert run('status-before', 'git', 'status', '--porcelain=v1') == b''
assert run('source-parent', 'git', 'rev-parse', SOURCE + '^').decode().strip() == PARENT
assert run('changed-paths', 'git', 'diff', '--name-only', PARENT, SOURCE).decode() == 'tests/test_geo261_host_authority.py\n'
run('fast-forward', 'git', 'merge', '--ff-only', SOURCE)
assert run('head-after', 'git', 'rev-parse', 'HEAD').decode().strip() == SOURCE
assert run('status-after', 'git', 'status', '--porcelain=v1') == b''
run('push', 'git', 'push', 'origin', 'HEAD:refs/heads/' + BRANCH)
assert run('remote', 'git', 'ls-remote', '--heads', 'origin', 'refs/heads/' + BRANCH).decode().strip() == SOURCE + '\trefs/heads/' + BRANCH
(OUT / 'receipt.json').write_text(json.dumps({'schema':'geo261-source-adoption/1', 'sourceSha':SOURCE, 'previousHead':PARENT,
    'branch':BRANCH, 'finishedUtc':datetime.datetime.now(datetime.timezone.utc).isoformat(), 'remoteExact':True,
    'scope':'只快轉自己的候選分支，未更新 PR #288 integration 來源分支。',
    'freezeRelease':'L2 native session78097/chunk4b2423 真1；L2 canvas session17336/chunkd211b7 真0、owned cleanup 完成；此為協作觀察轉錄。',
    'commands':commands},ensure_ascii=False,indent=2)+'\n')
print(json.dumps({'sourceSha':SOURCE,'branch':BRANCH,'remoteExact':True},ensure_ascii=False))
