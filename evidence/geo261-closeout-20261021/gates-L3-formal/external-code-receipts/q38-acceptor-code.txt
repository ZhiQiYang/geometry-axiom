symbols={}
def value(n):
    if isinstance(n,ast.Constant):return n.value
    if isinstance(n,ast.Name):return symbols[n.id]
    if isinstance(n,(ast.List,ast.Tuple,ast.Set)):
        a=[value(x) for x in n.elts]
        return tuple(a) if isinstance(n,ast.Tuple) else set(a) if isinstance(n,ast.Set) else a
    if isinstance(n,ast.Dict):return {value(a):value(b) for a,b in zip(n.keys,n.values)}
    if isinstance(n,ast.BinOp) and isinstance(n.op,ast.Add):return value(n.left)+value(n.right)
    if isinstance(n,ast.Call) and isinstance(n.func,ast.Name) and n.func.id=='case':
        keys=['id','clause','desc','file','before','after','tests','cause','equivalent','group','error_types','more']
        args=[value(x) for x in n.args];kw={x.arg:value(x.value) for x in n.keywords}
        d=dict(equivalent=None,group='native',error_types=(),more=())
        d.update(zip(keys,args));d.update(kw)
        d['patches']=[(d['file'],d['before'],d['after']),*d['more']]
        return d
    raise ValueError(type(n).__name__)

def check_evidence(e,code,logpath):
    assert e['schema']=='section-profile-exceptions/1'
    assert len(e['requested'])==len(set(e['requested']))>0 and e['requested']==e['executed']
    assert e['tests_run']==len(e['executed'])
    assert e['skipped']==e['expected_failures']==e['unexpected_successes']==0
    ev=e['events'];assert len({(x['test_id'],x['subtest_id']) for x in ev})==len(ev)
    assert all(x['test_id'] in e['executed'] for x in ev)
    assert e['failures']==sum(x['kind']=='failure' for x in ev) and e['errors']==sum(x['kind']=='error' for x in ev)
    assert set(e['passed'])==set(e['executed'])-{x['test_id'] for x in ev}
    assert code==(1 if ev else 0)
    raw=logpath.read_text()
    assert f"Ran {e['tests_run']} test" in raw
    if code==0:assert raw.rstrip().endswith('OK')
    else:assert 'FAILED (' in raw
    for x in ev:
        assert x['test_id'].split('.')[-1] in raw
        assert x['exception_type'].split('.')[-1]+':' in raw
    return raw
