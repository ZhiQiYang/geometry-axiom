#!/usr/bin/env python3
"""正式 L 的外部啟動器：stdout 與子程序真退出碼直接落盤，避免工具 session 遺失。"""
import argparse
import datetime
import hashlib
import json
import os
from pathlib import Path
import re
import signal
import subprocess
import sys
import time


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--sha', required=True)
    parser.add_argument('--root', default='/workspace/geometry-axiom')
    parser.add_argument('--output', required=True)
    parser.add_argument('--port', type=int, required=True)
    parser.add_argument('--cdp-port', type=int, required=True)
    parser.add_argument('--release-note', required=True)
    parser.add_argument('--execute', action='store_true')
    args = parser.parse_args()
    if not re.fullmatch('[0-9a-f]{40}', args.sha) or not args.execute or not args.release_note.strip():
        parser.error('須有完整 40 位 L SHA、明示 execute 與 root 放行記錄；尚未啟動')
    root, output = Path(args.root).resolve(), Path(args.output).resolve()
    if not output.is_relative_to('/workspace/geo261-evidence') or output.exists():
        parser.error('輸出須為尚未存在的外部證據目錄')
    script = Path(__file__).with_name('run_canvas_job.py').resolve()
    raw = script.read_bytes()
    prefix = output.parent / output.name
    log = Path(str(prefix) + '-orchestrator.log')
    exit_file = Path(str(prefix) + '-launcher-exit.json')
    raw_exit_file = Path(str(prefix) + '.exit-code')
    original = Path(str(prefix) + '-wrapper-command-code.txt')
    launcher_original = Path(str(prefix) + '-launcher-command-code.txt')
    if any(p.exists() for p in (log, exit_file, raw_exit_file, original, launcher_original)):
        parser.error('本輪啟動紀錄已存在，拒絕覆寫')
    output.parent.mkdir(parents=True, exist_ok=True)
    original.write_bytes(raw)
    launcher_original.write_bytes(Path(__file__).read_bytes())
    argv = [sys.executable, str(script), '--sha', args.sha, '--root', str(root), '--output', str(output),
            '--label', 'L-formal', '--port', str(args.port), '--cdp-port', str(args.cdp_port),
            '--release-note', args.release_note, '--execute']
    started, started_utc = time.monotonic(), datetime.datetime.now(datetime.timezone.utc).isoformat()
    child = None
    forwarded = []

    def forward(number, frame):
        if child is not None and child.poll() is None:
            # 直接子程序尚未 wait/reap，PID 不能被重用；交給既有 wrapper 的 finally 精確清理。
            child.send_signal(number)
            forwarded.append(number)

    for number in (signal.SIGINT, signal.SIGTERM):
        signal.signal(number, forward)
    signal.signal(signal.SIGHUP, signal.SIG_IGN)
    with log.open('xb') as stream:
        child = subprocess.Popen(argv, cwd=root, stdin=subprocess.DEVNULL,
                                 stdout=stream, stderr=subprocess.STDOUT, start_new_session=True)
        code = child.wait()
        # 先落真正 wait() 回傳碼，後續讀檔／收尾即使失敗仍可追溯。
        raw_exit_file.write_text(str(code) + '\n')
    elapsed = time.monotonic() - started
    record = dict(schema='geo261-canvas-launcher/1', sourceSha=args.sha, argv=argv, pid=child.pid,
                  startedUTC=started_utc, finishedUTC=datetime.datetime.now(datetime.timezone.utc).isoformat(),
                  observedWrapperReturncode=code, elapsedSeconds=elapsed, within2100Seconds=elapsed <= 2100,
                  wrapperSha256=hashlib.sha256(raw).hexdigest(), wrapperBytesUnchanged=script.read_bytes() == raw,
                  launcherSha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
                  signalsForwarded=forwarded, stdoutFile=str(log), originalWrapperBytes=str(original),
                  rawExitCodeFile=str(raw_exit_file), originalLauncherBytes=str(launcher_original),
                  note='returncode 是本啟動器對自己建立的 wrapper 子程序 wait() 所得；工具 session 是否仍可讀另列，不由產品 receipt 推造外層碼')
    exit_file.write_text(json.dumps(record, ensure_ascii=False, indent=2) + '\n')
    if time.monotonic() - started > 2100:
        record.update(within2100Seconds=False, elapsedSeconds=time.monotonic()-started)
        exit_file.write_text(json.dumps(record, ensure_ascii=False, indent=2) + '\n')
    return code if code else (0 if record['within2100Seconds'] and record['wrapperBytesUnchanged'] else 1)


if __name__ == '__main__':
    sys.exit(main())
