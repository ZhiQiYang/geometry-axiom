from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json

BASE = Path('/workspace/geo261-evidence')
RUN = BASE / 'browser-five-L-formal-43f85313'
LAUNCH = BASE / 'browser-five-L-formal-43f85313-launch'
PREP = BASE / 'browser-five-preparation'
BUNDLE = BASE / 'final-bundle/browser-five-L-formal'
L = '43f85313c32410ac01406ab585846bbacafea278'

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def save(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')

assert not BUNDLE.exists(), '新證據根不得覆寫'
assert (RUN / 'independent-review.json').is_file()
assert (RUN / 'artifact-independent-review.json').is_file()
assert json.loads((RUN / 'summary.json').read_text())['allFiveCompleted']
assert (LAUNCH / '.exit-code').read_text().strip() == '0'
assert json.loads((LAUNCH / 'exec-observation.json').read_text())['observedLauncherProcessExitCode'] == 0
BUNDLE.mkdir(parents=True)
copies = []
excluded = []

def copy_one(source, relative):
    assert source.is_file() and not source.is_symlink()
    target = BUNDLE / relative
    assert not target.exists()
    raw = source.read_bytes()
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(raw)
    assert target.read_bytes() == raw
    copies.append({'source': str(source), 'path': relative.as_posix(), 'bytes': len(raw),
                   'sha256': sha(raw), 'sourceMtimeNs': source.stat().st_mtime_ns})

for source_root, prefix in ((RUN, 'run'), (LAUNCH, 'launch'), (PREP, 'preparation')):
    for source in sorted(source_root.rglob('*')):
        if not source.is_file():
            continue
        relative = Path(prefix) / source.relative_to(source_root)
        if source.suffix in ('.py', '.js', '.cjs', '.pyc'):
            raw = source.read_bytes()
            excluded.append({'source': str(source), 'bytes': len(raw), 'sha256': sha(raw),
                             'reason': '不另收可執行副本；正式執行的原 bytes 已以 code.txt／launcher-code.txt／wrapper-code.txt 保存。'})
            continue
        copy_one(source, relative)
copy_one(BASE / 'browser-five-plan.md', Path('preparation/執行計畫.md'))
copy_one(Path(__file__), Path('preparation/bundle-code.txt'))
assert len(excluded) == 2
save(BUNDLE / '原件清單.json', {'schema': 'geo261-evidence-copy-index/1', 'sourceSha': L,
     'capturedUTC': datetime.now(timezone.utc).isoformat(), 'count': len(copies),
     'bytes': sum(row['bytes'] for row in copies), 'files': copies})
save(BUNDLE / '未收錄清單.json', {'files': excluded,
     'note': '來源集合包含本輪 run、launch、preparation 與執行計畫；binary、Chrome profile、cache 不屬封裝來源。binary 身分由原 native-identity/sidecar 保存，81 PNG 與9份 gcanvas及合成PDF輸入完整收入。'})
