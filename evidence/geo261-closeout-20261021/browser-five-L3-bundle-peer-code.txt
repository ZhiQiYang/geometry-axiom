"""只讀核對凍結的L3五項browser證據包；不執行產品。"""
from pathlib import Path
import collections
import datetime
import hashlib
import json
import re
import stat

ROOT = Path('/workspace/geo261-evidence')
PACK = ROOT/'final-bundle/browser-five-L3-formal'
SOURCE = '6589248c9a94bdf0460d8f84dd1610e51048188b'
sha = lambda data: hashlib.sha256(data).hexdigest()
allpaths = list(PACK.rglob('*'))
assert not any(path.is_symlink() for path in allpaths)
files = {str(path.relative_to(PACK)): path for path in allpaths if path.is_file()}
data = {name: path.read_bytes() for name, path in files.items()}
index = json.loads(data['source-index.json'])
assert sha(data['source-index.json']) == 'f456ade11a89f8c0f6278d4050b269d30a9e1c3a27b0d40fc79293ed5c81c143'
assert sha(data['SHA256SUMS']) == 'd839a2f626ea889902b66a0f70c694d70d1bf9caf12131a645a6f6d6c4928798'
assert index['sourceSha'] == SOURCE
assert index['sourceFileCount'] == len(index['files']) == 284
assert len({row['artifactPath'] for row in index['files']}) == 284
rows = []
for row in index['files']:
    original = Path(row['sourcePath']).read_bytes()
    copied = data[row['artifactPath']]
    assert original == copied
    assert len(original) == row['bytes']
    assert sha(original) == row['sha256']
    rows.append({'path': row['artifactPath'], 'bytes': len(copied), 'sha256': sha(copied)})
assert sum(row['bytes'] for row in rows) == index['sourceTotalBytes'] == 9496607
assert len(files) == 287 and sum(map(len, data.values())) == 9633585
assert set(files) == {row['artifactPath'] for row in index['files']} | {'README.md', 'source-index.json', 'SHA256SUMS'}
assert all(stat.S_IMODE(path.stat().st_mode) == 0o644 for path in files.values())
assert not any(path.suffix.lower() in ('.py', '.js', '.cjs', '.exe', '.so') for path in files.values())
sums = {}
for line in data['SHA256SUMS'].decode('utf-8').splitlines():
    value, name = line.split('  ', 1)
    assert name not in sums
    sums[name] = value
assert set(sums) == set(files) - {'SHA256SUMS'}
assert len(sums) == 286
for name, value in sums.items():
    assert sha(data[name]) == value
suffixes = collections.Counter(path.suffix.lower() for path in files.values())
assert suffixes['.png'] == 81 and suffixes['.gcanvas'] == 9 and suffixes['.pdf'] == 1
readme = data['README.md'].decode('utf-8')
links = re.findall(r'\[[^\]]+\]\(([^)]+)\)', readme)
assert all((PACK/link).exists() for link in links)
peerrows = [row for row in rows if row['path'].startswith('peer/')]
assert len(peerrows) == 25 and sum(row['bytes'] for row in peerrows) == 158724
for boundary in ('本輪 reused', '原 partial harness', '47 個具名 PASS', '868個斷言不是868個真人端到端情境', 'Injected lost success response', '15 件107195 bytes', '另核原 wrapper 已存9件非PNG', '事後捕捉不冒充在原job期限內', '本輪沒有新跑撤回突變', '未逐張人工目視 PNG'):
    assert boundary in readme, boundary
peerreadme = data['peer/README.md'].decode('utf-8')
assert '兩次資料探索錯誤只保有真工具碼與摘要' in peerreadme
assert '當時未另存完整stdout/stderr' in peerreadme
out = {
    'schema': 'geo261-browser-five-L3-bundle-peer/1',
    'checkedUTC': datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'sourceSha': SOURCE,
    'validationExitCode': 0,
    'readOnly': True,
    'originalCount': 284, 'fileCount': 287, 'bytes': 9633585, 'checksums': 286,
    'pngCount': 81, 'gcanvasCount': 9, 'pdfCount': 1,
    'exactOriginalBytes': True, 'exactFileSet': True, 'allRegular0644': True,
    'noExecutableCopies': True,
    'sourceIndexSha256': sha(data['source-index.json']),
    'sha256sumsSha256': sha(data['SHA256SUMS']),
    'files': rows, 'readmeLinksChecked': links, 'includedPeerOriginals': 25,
    'readmeClarification': '包README將探索錯誤與首版接受器錯誤並列後稱完整streams已保留，文字範圍過廣；完整原碼/stdout/stderr/真wait/差分屬於首版接受器。兩次探索只有真工具碼及摘要，未另存完整stdout/stderr。包內peer/README.md與peer/tool-observations.json已精確說明；本外層再次澄清，不改凍結包。',
    'limitations': [
        '這是來源原件、索引、SUMS與README的有限唯讀核查，不重跑產品或既有接受器，不改子包。',
        'PNG逐bytes/hash核對；未逐張目視。',
        '五項原scope/reused/原退出碼的結論沿包內原收據與獨審；不能由此代替其他job。',
        '探索錯誤的完整工具輸出未另存，不能把摘要當rawtransport或完整stderr。'
    ]
}
with (ROOT/'browser-five-L3-bundle-peer.json').open('x', encoding='utf-8') as stream:
    stream.write(json.dumps(out, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({key: out[key] for key in ('validationExitCode','originalCount','fileCount','bytes','checksums')}, ensure_ascii=False))
