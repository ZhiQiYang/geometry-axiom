def save(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


def store_stream(row, name, data, representation):
    path = 'command-%04d.%s' % (row['index'], name)
    (OUT / path).write_bytes(data)
    row[name] = {'path': path, 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest(),
                 'representation': representation}


class ObservedPopen(original_popen):
    """原 communicate 解碼前保留 bytes；不改原 Popen 參數或返回值。"""
    def __init__(self, *args, **kwargs):
        self.observed_row = current_run
        self.translation_index = 0
        super().__init__(*args, **kwargs)

    def _translate_newlines(self, data, encoding, errors):
        if self.observed_row is not None:
            streams = [name for name in ('stdout', 'stderr') if getattr(self, name) is not None]
            name = streams[self.translation_index]
            self.translation_index += 1
            store_stream(self.observed_row, 'raw-' + name, data, '原 Popen communicate 解碼前 bytes')
        return super()._translate_newlines(data, encoding, errors)


def observed_run(*args, **kwargs):
    global current_run
    command = args[0] if args else kwargs['args']
    row = {'index': len(records), 'command': list(map(str, command)),
           'cwd': str(kwargs.get('cwd', ROOT)), 'test': active['test'],
           'timeout': kwargs.get('timeout'),
           'started_utc': datetime.datetime.now(datetime.timezone.utc).isoformat()}
    records.append(row)
    current_run = row
    try:
        result = original_run(*args, **kwargs)
        row['returncode'] = result.returncode
        for name in ('stdout', 'stderr'):
            value = getattr(result, name)
            if value is not None:
                data = value.encode('utf-8') if isinstance(value, str) else value
                store_stream(row, name, data, '原 subprocess.run 返回字串以 UTF-8 保存' if isinstance(value, str) else '原返回 bytes')
        return result
    except BaseException as exc:
        row['exception'] = repr(exc)
        raise
    finally:
        current_run = None
        row['finished_utc'] = datetime.datetime.now(datetime.timezone.utc).isoformat()
        save('targeted-commands.json', records)


def source_hashes():
    paths = [ROOT / 'current_geometry.c']
    for directory in ('geometry-app/native', 'geometry-core/kernel', 'geometry-templates/woodworking/native'):
        paths.extend(p for p in (ROOT / directory).rglob('*') if p.is_file() and p.suffix in ('.c', '.h', '.inc'))
    paths.extend((ROOT / 'geometry-core').glob('*.inc'))
    paths.extend((ROOT / 'tests').glob('*.py'))
    paths.extend((ROOT / 'tests').glob('geo261_*harness.c'))
    return {str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(set(paths))}
