import ast
import hashlib
import json
import subprocess
from pathlib import Path

D = Path(__file__).resolve().parent
R = Path('/workspace/geometry-axiom')
W = Path('/workspace/geo261-host-authority-anchor-fix')
HEAD = '249571fe2477845d5330f6190dad965bf327f43b'
TEST = 'tests/test_geo261_host_authority.py'


def git(cwd, name, *args):
    argv = ['git', *args]
    run = subprocess.run(argv, cwd=cwd, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    (D / (name + '.stdout')).write_bytes(run.stdout)
    (D / (name + '.stderr')).write_bytes(run.stderr)
    (D / (name + '.exit-code')).write_text(str(run.returncode) + '\n')
    commands.append({'argv': argv, 'cwd': str(cwd), 'actual_exit_code': run.returncode})
    assert run.returncode == 0
    return run.stdout


def constant(tree):
    matches = [node for node in tree.body if isinstance(node, ast.Assign)
               and any(isinstance(target, ast.Name) and target.id == 'FOOTPRINT_ANCHOR'
                       for target in node.targets)]
    assert len(matches) == 1
    return matches[0]

assert git(R, 'main-status', 'status', '--porcelain') == b''
assert git(W, 'candidate-head', 'rev-parse', 'HEAD').decode().strip() == HEAD
assert git(W, 'changed-paths', 'diff', '--name-only').decode().splitlines() == [TEST]
git(W, 'diff-check', 'diff', '--check')
