"""只封裝既有錨點修復與定向原件，不執行產品。"""
from pathlib import Path
import hashlib
import json
import os

D = Path(__file__).resolve().parent
E = D.parent
P = E / 'source-S3-publication'
B = E / 'final-bundle/host-authority-anchor-fix-S3'
assert not B.exists()
assert json.loads((D / 'targeted-readonly-review.json').read_bytes())['validation_exit_code'] == 0
binding = json.loads((D / 'committed-binding/receipt.json').read_bytes())
assert binding['validation_exit_code'] == 0 and binding['tested_source_files_bound_to_S3_Git_blobs'] == 479
B.mkdir()
mapping = []


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


for root, label in [(D, 'fix-targeted'), (P, 'source-publication')]:
    for source in sorted(root.rglob('*')):
        if not source.is_file() or 'private-tmp' in source.parts or '__pycache__' in source.parts:
            continue
        assert not source.is_symlink()
        raw = source.read_bytes()
        assert not raw.startswith(b'\x7fELF') and source.suffix not in ('.tar', '.o', '.pyc', '.exe')
        raw.decode('utf-8')
        rel = Path(label) / source.relative_to(root)
        if source.suffix in ('.py', '.c', '.h', '.sh'):
            rel = Path(str(rel) + '.txt')
        target = B / rel
        target.parent.mkdir(parents=True, exist_ok=True)
        with target.open('xb') as f:
            f.write(raw)
        os.chmod(target, 0o644)
        assert target.read_bytes() == raw
        mapping.append({'source': str(source), 'published': str(rel), 'bytes': len(raw),
                        'sha256': digest(raw), 'copied_source_bytes_equal': True})
(B / 'artifact-mapping.json').write_text(json.dumps({'files': mapping,
    'excluded': ['私有 NativeBuild 暫存樹、ELF、物件檔、pycache；不列入文字證據'],
    'snapshot_policy': '方法 .py.txt 為原 bytes 的 0644 資料快照，不新增正式工具入口；歷史未執行 v1 重建 bytes 另明示其性質。'}, ensure_ascii=False, indent=2) + '\n')
(B / 'README.md').write_text('''# host authority 錨點修復與定向驗證

公開 S3：19401ca5448e162ee6140fed37b64b83b31ddec8；分支 codex/geo261-host-authority-anchor-fix-20261021；唯一 parent L2：249571fe2477845d5330f6190dad965bf327f43b。唯一改檔 tests/test_geo261_host_authority.py；只更新 FOOTPRINT_ANCHOR，保留 count==1、原 marker、原數值及突變流程。6c 的區域暫存調整使舊單行錨點失效；既有 L2 native 的兩案失敗原件仍由原正式包保存，未改成綠。

fixed-targeted 的 raw-stdout/raw-stderr 是 Popen communicate 解碼前 bytes；stdout/stderr 是原 subprocess.run 返回字串以 UTF-8 保存。建場 BridgeSession 的直接協定串流未另捕獲，僅保存完整建場 scene dump；不得稱所有 bridge 協定 bytes 已封存。NativeBuild 正常退出清理了暫存 binaries，包內不提供 ELF 或完整暫存來源樹。獨立靜態 peer 與頂層工具結果部分只有對話／工具觀察，轉錄均明示不是原 transport，也未假造接受器原碼。

files = sorted(p for p in B.rglob('*') if p.is_file())
(B / 'SHA256SUMS').write_text(''.join(digest(p.read_bytes()) + '  ' + str(p.relative_to(B)) + '\n' for p in files))
for line in (B / 'SHA256SUMS').read_text().splitlines():
    value, rel = line.split('  ', 1)
    assert digest((B / rel).read_bytes()) == value
for row in mapping:
    assert (B / row['published']).read_bytes() == Path(row['source']).read_bytes()
files = sorted(p for p in B.rglob('*') if p.is_file())
assert not any(p.suffix in ('.py', '.c', '.h', '.sh') for p in files)
result = {'bundle': str(B), 'file_count': len(files), 'bytes': sum(p.stat().st_size for p in files),
          'manifest_sha256': digest((B / 'SHA256SUMS').read_bytes()), 'original_copies': len(mapping),
          'all_source_copies_equal': True, 'all_manifest_hashes_match': True,
          'new_product_executions': 0, 'validation_exit_code': 0}
(E / 'host-authority-anchor-fix-S3-copy-check.json').write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
print(json.dumps(result, ensure_ascii=False, indent=2))
