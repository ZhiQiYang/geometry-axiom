"""只封裝本輪已完成的 L3 native 架構引用附件，不改既有架構包。"""
import hashlib
import json
from pathlib import Path
import stat
import sys

if sys.flags.optimize: raise RuntimeError('不可停用封裝核對')
BASE = Path('/workspace/geo261-evidence')
SOURCE = BASE / 'architecture-L3-native-reference-run01'
DEST = BASE / 'final-bundle/architecture-L3-native引用'
def sha(raw): return hashlib.sha256(raw).hexdigest()
def write(path, raw):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('xb') as stream: stream.write(raw)
    path.chmod(0o644)
assert json.loads((SOURCE/'execution.json').read_bytes())['actualWaitReturnCode'] == 0
assert sha((SOURCE/'binding.json').read_bytes()) == '41c8216749dad563dc1b2d5529490524955dd6975abd4bc742fcb3833f1cac1d'
DEST.mkdir(exist_ok=False)
rows = []
sources = [(p,p.relative_to(SOURCE).as_posix()) for p in sorted(SOURCE.rglob('*')) if p.is_file()]
sources.append((Path(__file__),'封裝方法-code.txt'))
for source, name in sources:
    assert not source.is_symlink() and stat.S_ISREG(source.lstat().st_mode)
    assert Path(name).suffix not in {'.py','.js','.c','.h','.inc','.exe','.dll','.so'}
    raw = source.read_bytes()
    write(DEST/name,raw)
    assert (DEST/name).read_bytes() == raw
    rows.append({'path':name,'sourcePath':str(source),'bytes':len(raw),'sha256':sha(raw)})
index = {'schema':1,'sourceL3':'6589248c9a94bdf0460d8f84dd1610e51048188b',
         'selfExcluded':'來源索引.json','generatedAlsoExcluded':['SHA256SUMS'],
         'entries':sorted(rows,key=lambda r:r['path'])}
write(DEST/'來源索引.json',(json.dumps(index,ensure_ascii=False,indent=2)+'\n').encode())
files = sorted(p for p in DEST.rglob('*') if p.is_file())
assert {p.relative_to(DEST).as_posix() for p in files} == {r['path'] for r in rows}|{'來源索引.json'}
write(DEST/'SHA256SUMS',''.join(sha(p.read_bytes())+'  '+p.relative_to(DEST).as_posix()+'\n' for p in files).encode())
for row in rows:
    raw=(DEST/row['path']).read_bytes()
    assert raw == Path(row['sourcePath']).read_bytes() and sha(raw)==row['sha256']
files.append(DEST/'SHA256SUMS')
assert all(stat.S_IMODE(p.stat().st_mode)==0o644 for p in files)
result={'path':str(DEST),'files':len(files),'bytes':sum(p.stat().st_size for p in files),
        'allCopiedBytesAndExactSetMatch':True,'indexSha256':sha((DEST/'來源索引.json').read_bytes()),
        'sumsSha256':sha((DEST/'SHA256SUMS').read_bytes()),'originalArchitectureBundleModified':False,
        'productOrCLIRerun':False}
write(BASE/'final-bundle/architecture-L3-native引用-copycheck.json',(json.dumps(result,ensure_ascii=False,indent=2)+'\n').encode())
print(json.dumps(result,ensure_ascii=False,indent=2))
