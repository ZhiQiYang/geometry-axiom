#!/usr/bin/env python3
"""外部正式 L3 來源準備器；只讀 Git 與解 archive，不建置／測試／取樣。"""
import argparse
import datetime
import hashlib
import io
import json
from pathlib import Path
import subprocess
import tarfile
import tempfile

p=argparse.ArgumentParser()
p.add_argument('--source-head',required=True)
p.add_argument('--expected-parent',required=True)
a=p.parse_args()
E=Path('/workspace/geo261-evidence');R=Path('/workspace/geometry-axiom')
L='249571fe2477845d5330f6190dad965bf327f43b'
C=a.source_head;LABEL='formal-L3';H='59796b1af9259eb23d30134c2ce5a56317a06091'
assert len(C)==40 and all(c in '0123456789abcdef' for c in C)
assert len(a.expected_parent)==40 and all(c in '0123456789abcdef' for c in a.expected_parent)
assert a.expected_parent=='19401ca5448e162ee6140fed37b64b83b31ddec8'
if not __debug__:
    raise RuntimeError('不可使用 Python -O／PYTHONOPTIMIZE')
W=Path(tempfile.mkdtemp(prefix='geo261-p189-'+LABEL+'-src-',dir='/workspace'))
O=Path(tempfile.mkdtemp(prefix='p189-'+LABEL+'-'+C[:12]+'-',dir=E))
commands=[]
def sha(b):return hashlib.sha256(b).hexdigest()
def save(path,value):
    with path.open('x') as f:json.dump(value,f,ensure_ascii=False,indent=2);f.write('\n')
def git(*args,cwd=R):
    r=subprocess.run(['git',*args],cwd=cwd,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    commands.append({'argv':['git',*args],'cwd':str(cwd),'actual_exit_code':r.returncode,'stdout_sha256':sha(r.stdout),'stderr':r.stderr.decode()})
    assert r.returncode==0,commands[-1]
    return r.stdout
assert git('rev-parse',C+'^{commit}').decode().strip()==C
assert git('rev-parse',a.expected_parent+'^{commit}').decode().strip()==a.expected_parent
assert git('rev-list','--parents','-n','1',C).decode().split()==[C,a.expected_parent]
assert git('rev-parse','HEAD').decode().strip()==C
assert git('status','--porcelain')==b''
changed=git('diff','--name-only',a.expected_parent,C).decode().splitlines()
assert changed==['tests/mutants_results.json'],changed
patch=git('diff','--binary','--full-index',a.expected_parent,C)
(O/'source-parent-ledger.patch').write_bytes(patch)
lpatch=git('diff','--binary','--full-index',L,C)
(O/'source-vs-previous-L.patch').write_bytes(lpatch)
changed_blobs=[]
for path in changed:
    raw=git('show',C+':'+path)
    changed_blobs.append({'path':path,'git_blob':git('rev-parse',C+':'+path).decode().strip(),'bytes':len(raw),'sha256':sha(raw)})
runner=R/'geometry-core/domain-decoupling/closeout-tools/p189_verify.py'
runner_raw=runner.read_bytes()
assert sha(runner_raw)=='fb42a801ce350e64444fcb2db433e59f8c2228546b4b594298581bdb365d0c44'
assert runner_raw==git('show',L+':'+str(runner.relative_to(R)))==git('show',C+':'+str(runner.relative_to(R)))
genpath='handoff/drafts/issue300/bench/gen_scene.py';calpath='handoff/drafts/issue300/bench/calib/f3-00.scene'
gen=git('show',H+':'+genpath);cal=git('show',H+':'+calpath)
assert sha(gen)=='347142beb7aa06c39969cd5bce35bacddc0734046eb7853e84d96f1ff9ef7c91'
assert sha(cal)=='35e24fb0a483ffd6fee8dd99f751edeb127418481aa717816af804cd218024b2'
(W/'gen_scene.py').write_bytes(gen);(W/'calib').mkdir();(W/'calib/f3-00.scene').write_bytes(cal)
commits={'main':'adbaebf9c73ea6801d7b4d5214e4b846b132923b','base':'3e600772076465fc65a26ac09055028301c0fb0b','final':C}
archives=[];(W/'archives').mkdir()
for label,commit in commits.items():
    assert git('rev-parse',commit+'^{commit}').decode().strip()==commit
    paths=[]
    for path in ('current_geometry.c','geometry-core','geometry-app','geometry-templates','geometry-web'):
        r=subprocess.run(['git','cat-file','-e',commit+':'+path],cwd=R,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
        commands.append({'argv':['git','cat-file','-e',commit+':'+path],'cwd':str(R),'actual_exit_code':r.returncode})
        if r.returncode==0:paths.append(path)
    excludes=[':(exclude)'+path for path in ('geometry-core/domain-decoupling','geometry-web/docs','geometry-web/prototypes','geometry-web/vendor','geometry-web/node_modules')]
    argv=['archive','--format=tar',commit,'--',*paths,*excludes]
    raw=git(*argv);arc=W/'archives'/(label+'.tar');arc.write_bytes(raw)
    src=W/commit[:8];src.mkdir();rows=[]
    with tarfile.open(fileobj=io.BytesIO(raw)) as tar:
        for member in tar.getmembers():
            if member.isdir():continue
            assert member.isfile() and not Path(member.name).is_absolute() and '..' not in Path(member.name).parts
            target=src/member.name;target.parent.mkdir(parents=True,exist_ok=True)
            data=tar.extractfile(member).read();target.write_bytes(data)
            assert target.read_bytes()==data
            rows.append({'path':member.name,'bytes':len(data),'sha256':sha(data)})
    assert not (src/'bridge').exists() and not (src/'.git').exists()
    save(O/('source-files-'+label+'.json'),rows)
    archives.append({'label':label,'sha':commit,'argv':['git',*argv],'cwd':str(R),'actual_exit_code':0,'archive_path':str(arc),'archive_bytes':len(raw),'archive_sha256':sha(raw),'source_directory':str(src),'source_file_count':len(rows),'source_index':'source-files-'+label+'.json','bridge_exists':False})
planned=[{'phase':'generator-selftest','argv':['python3','-B',str(W/'gen_scene.py'),'--selftest']}]
for phase in ('selftest','build','equivalence','timing','check'):
    planned.append({'phase':phase,'argv':['python3','-B','-u',str(runner),phase,'--repo',str(R),'--work',str(W),'--out',str(O),'--gen',str(W/'gen_scene.py'),'--final',C,'--timeout','900']})
supervisor=E/'p189-final-L-phase-supervisor.py'
assert sha(supervisor.read_bytes())=='e6e2dfdd321658084f33d175af01da22d0643103556de0fc9805d96f18f33413'
assert git('rev-parse','HEAD').decode().strip()==C
assert git('status','--porcelain')==b''
prep={'prepared_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'classification':'正式 L3 P-189 驗收；本來源需新實跑，不沿用 L2 或候選成績',
 'label':LABEL,'source_worktree':str(R),'repo':str(R),'work':str(W),'out':str(O),'commits':commits,
 'expected_parent':a.expected_parent,'source_sha':C,'source_head_clean_matches':True,'source_parent_matches':True,'parent_delta_only_ledger':True,'changed_paths':changed,
 'changed_blobs':changed_blobs,'source_parent_ledger_patch_sha256':sha(patch),'source_vs_previous_L_patch_sha256':sha(lpatch),
 'runner':str(runner),'runner_sha256':sha(runner_raw),'runner_git_blob':git('rev-parse',L+':'+str(runner.relative_to(R))).decode().strip(),
 'generator':str(W/'gen_scene.py'),'generator_sha256':sha(gen),'generator_source':H+':'+genpath,
 'generator_calibration':{'path':str(W/'calib/f3-00.scene'),'source':H+':'+calpath,'sha256':sha(cal),'bytes':len(cal)},
 'generator_dependency_review':'沿已讀固定原bytes的靜態依賴核：selftest唯一外部讀取為同目錄calib/f3-00.scene，其餘imports皆標準庫；本輪實際selftest尚待執行。',
 'archives':archives,'planned_commands':planned,'expected_equivalence_scene_count':175,'expected_timing_sample_count':36,
 'selftest_phase_rule':'本輪顯式selftest執行16控制，build/equivalence/timing/check不自動執行。',
 'timing_return_rule':'timing0僅表示收樣完成，check才判原門檻。','threshold_final_main_ratio':2.0,
 'environment_remove_before_execution':['PYTHONOPTIMIZE'],'stack_kib':16384,'timeout_seconds':900,
 'phase_commands_executed':[],'build_started':False,'timing_started':False,'preparation_tool_sha256':sha(Path(__file__).read_bytes())}
save(O/'prepare-receipt.json',prep);save(O/'planned-commands.json',planned);save(O/'preparation-commands.json',commands)
config={'prepared_utc':prep['prepared_utc'],'authorization':'主代理明確放行後才可啟動','released':False,
 'prepare_receipt':str(O/'prepare-receipt.json'),'supervisor':str(supervisor),'supervisor_sha256':sha(supervisor.read_bytes()),
 'out':str(O),'scheduled_launch_argv':['python3','-B',str(supervisor),str(O/'supervisor-released-config.json')],
 'launch_transport':'exec_command保有頂層session並真wait；stdout/stderr直檔，不launch-and-exit。','classification':prep['classification']}
save(O/'supervisor-prepared-config.json',config)
pointer={'out':str(O),'work':str(W),'source_sha':C,'ready_not_released':True}
save(E/('p189-'+LABEL+'-current.json'),pointer)
