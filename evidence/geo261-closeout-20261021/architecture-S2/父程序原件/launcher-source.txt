#!/usr/bin/env python3
"""保存外部 owned 刷新之父層命令、原輸出與真正程序 wait 碼。"""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import subprocess
import sys
import time

HERE = Path('/workspace/geo261-evidence/architecture-owned-preparation')
ROOT = Path('/workspace/geometry-axiom')
OUT = Path('/workspace/geo261-evidence/architecture-owned-S2-cabe8241')
LAUNCH = Path(str(OUT) + '-launcher')
HEAD = 'cabe824128be31d0039dc6c94fa3f04cfb722a6b'
argv = ['env', '-u', 'CC', '-u', 'PYTHONOPTIMIZE', 'python3', '-B',
        str(HERE / 'refresh_owned.py'), '--root', str(ROOT), '--out', str(OUT),
        '--source-head', HEAD, '--review-file', str(HERE / 'review-S2.json')]
LAUNCH.mkdir(exist_ok=False)
def now(): return datetime.now(timezone.utc).isoformat()
def dump(path, obj): path.write_text(json.dumps(obj, ensure_ascii=False, indent=2) + '\n')
started, tick = now(), time.monotonic()
command = {'argv': argv, 'cwd': str(ROOT), 'sourceHead': HEAD, 'startedUtc': started,
           'driverSha256': hashlib.sha256((HERE / 'refresh_owned.py').read_bytes()).hexdigest(),
           'reviewSha256': hashlib.sha256((HERE / 'review-S2.json').read_bytes()).hexdigest()}
dump(LAUNCH / 'command.json', command)
(LAUNCH / 'launcher-source.txt').write_bytes(Path(__file__).read_bytes())
with (LAUNCH / 'stdout.log').open('xb') as stdout, (LAUNCH / 'stderr.log').open('xb') as stderr:
    child = subprocess.Popen(argv, cwd=ROOT, stdout=stdout, stderr=stderr)
    code = child.wait()
receipt = {**command, 'finishedUtc': now(), 'elapsedSeconds': time.monotonic() - tick,
           'childPid': child.pid, 'returnCodeFromWait': code}
dump(LAUNCH / 'receipt.json', receipt)
(LAUNCH / '.exit-code').write_text(str(code) + '\n')
print(json.dumps(receipt, ensure_ascii=False, indent=2), flush=True)
raise SystemExit(code)
