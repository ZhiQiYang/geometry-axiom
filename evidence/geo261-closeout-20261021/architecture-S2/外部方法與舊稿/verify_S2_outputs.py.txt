#!/usr/bin/env python3
"""輕量只讀核對 S2 實跑原件；不重新執行掃描器或任何測試。"""
import hashlib
import json
from pathlib import Path
import subprocess

ROOT = Path('/workspace/geometry-axiom')
RUN = Path('/workspace/geo261-evidence/architecture-owned-S2-cabe8241')
LAUNCH = Path(str(RUN) + '-launcher')
AUDIT = Path(str(RUN) + '-audit')
HEAD = 'cabe824128be31d0039dc6c94fa3f04cfb722a6b'
PREFIX = 'geometry-core/domain-decoupling/'
OUTPUTS = [PREFIX + p for p in ('product-architecture.json', 'product-architecture-scan.json',
                               'PRODUCT-ARCHITECTURE.md', 'product-architecture-receipt.json')]
def sha(raw): return hashlib.sha256(raw).hexdigest()
def read(path): return json.loads(path.read_bytes())
def git(*args): return subprocess.check_output(['git', '-C', str(ROOT), *args])
summary = read(RUN / 'summary.json')
receipt = read(RUN / 'final/product-architecture-receipt.json')
review = read(RUN / 'review.json')
launcher = read(LAUNCH / 'receipt.json')
assert summary['sourceHead'] == receipt['sourceBaseHead'] == launcher['sourceHead'] == HEAD
assert summary['exitCode'] == launcher['returnCodeFromWait'] == 0
assert (LAUNCH / '.exit-code').read_text() == '0\n'
observed = git('rev-parse', 'HEAD').decode().strip()
changes_from_source = git('diff', '--name-only', '--no-renames', HEAD).decode().splitlines()
assert set(changes_from_source) <= set(OUTPUTS)
metadata = {}
for path in OUTPUTS:
    raw = (ROOT / path).read_bytes()
    assert raw == (RUN / 'final' / Path(path).name).read_bytes()
    assert sha(raw) == summary['outputs'][path]['rawSha256']
    old = (RUN / 'previous' / Path(path).name).read_bytes()
    assert old == git('show', HEAD + ':' + path)
    metadata[path] = {'rawSha256': sha(raw), 'bytes': len(raw), 'previousRawSha256': sha(old)}
old_manifest = read(RUN / 'previous/product-architecture.json')
manifest = read(RUN / 'final/product-architecture.json')
new_guard = manifest['policy']['forbiddenIdentifiers'].pop()
assert new_guard == 'WoodContactOrder' and manifest == old_manifest
items = receipt['inputs']
assert len(items) == 611 and len({r['path'] for r in items}) == 611
for item in items:
    raw = (ROOT / item['path']).read_bytes()
    assert sha(raw.replace(b'\r\n', b'\n')) == item['sha256']
    assert sha(raw) == receipt['completeInventoryVerification']['sourceHashesBeforeAndAfter'][item['path']]['rawSha256']
closure = sha(json.dumps(items, ensure_ascii=False, separators=(',', ':')).encode())
assert closure == receipt['inputClosureSha256']
assert set(r['path'] for r in receipt['manifestRefresh']['inputsChangedSincePreviousSeal']) == set(review['changedOwnedSources'])
assert len(review['changedOwnedSources']) == 5 and len(receipt['outsideInventoryReviewedTests']) == 7
for path, hashes in receipt['outsideInventoryReviewedTests'].items():
    assert sha((ROOT / path).read_bytes()) == hashes['rawSha256'] == review['changedOutsideInventory'][path]['rawSha256']
records = []
expected_codes = [('prior-manifest', 0), ('actual-cli-scan', 0), ('generic-tests', 0), ('product-tests', 0),
                  ('old-guard-accepts-0', 0), ('new-guard-rejects-0', 1), ('withdraw-guard-accepts-0', 0), ('neutral-accepts', 0)]
for name, code in expected_codes:
    row = read(RUN / name / 'result.json')
    assert row['exitCode'] == code and row['rawWaitStatus'] == code * 256
    for stream in ('stdout', 'stderr'):
        raw = (RUN / row[stream]['path']).read_bytes()
        assert sha(raw) == row[stream]['rawSha256']
    records.append({'name': name, 'argv': row['argv'], 'exitCode': code, 'rawWaitStatus': row['rawWaitStatus'], 'seconds': row['seconds']})
for name, count in [('generic-tests', 205), ('product-tests', 18)]:
    row = read(RUN / name / 'result.json')
    assert (row['tests'], row['passed'], row['skipped']) == (count, count, 0)
diagnostics = read(RUN / 'new-guard-rejects-0/stdout.log')['diagnostics']
assert len(diagnostics) == 1 and diagnostics[0]['code'] == 'FORBIDDEN_IDENTIFIER'
assert diagnostics[0]['path'] == 'geometry-core/kernel/geometry.c'
for name in ('old-guard-accepts-0', 'withdraw-guard-accepts-0', 'neutral-accepts'):
    value = read(RUN / name / 'stdout.log')
    assert value['ok'] and not value['diagnostics']
AUDIT.mkdir(exist_ok=False)
result = {'sourceBaseHead': HEAD, 'observedHead': observed, 'mode': '只讀原件、Git 差異與雜湊核對',
          'counts': summary['counts'], 'inputClosureSha256': closure, 'checkedInputCount': len(items),
          'changedOwnedPaths': list(review['changedOwnedSources']), 'checkedOutsidePaths': list(receipt['outsideInventoryReviewedTests']),
          'onlyMetadataDiffFromSource': changes_from_source, 'metadata': metadata, 'commands': records,
          'launcherWaitCode': launcher['returnCodeFromWait'], 'launcherElapsedSeconds': launcher['elapsedSeconds'],
          'newGuardDiagnostic': diagnostics[0],
          'notClaimed': ['重新執行正式掃描或測試', '產品行為或性能驗收', '後續 A2／L2 來源已實跑']}
(AUDIT / 'verification.json').write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
(AUDIT / 'command-code.txt').write_bytes(Path(__file__).read_bytes())
print(json.dumps(result, ensure_ascii=False, indent=2))
