#!/usr/bin/env python3
"""只封裝已凍結架構 S2 原件；執行來源改副檔名為文字，內容 bytes 不變。"""
import hashlib
import json
from pathlib import Path

BASE = Path('/workspace/geo261-evidence')
DEST = BASE / 'final-bundle/architecture-S2'
GROUPS = {
    '實跑原件': BASE / 'architecture-owned-S2-cabe8241',
    '父程序原件': BASE / 'architecture-owned-S2-cabe8241-launcher',
    '只讀核對': BASE / 'architecture-owned-S2-cabe8241-audit',
    'S2-A2綁定': BASE / 'architecture-owned-S2-A2-binding',
    '公開提交原件': BASE / 'source-A2-publication',
    '外部方法與舊稿': BASE / 'architecture-owned-preparation',
}
FILES = {
    '結果.md': BASE / 'architecture-owned-S2結果.md',
    '獨立審查.md': BASE / 'architecture-owned-S2-cabe8241-INDEPENDENT_REVIEW.md',
    'root審查結果.json': BASE / 'architecture-owned-S2-root-review.json',
}
DEST.mkdir(exist_ok=False)
entries = []
def copy(src, relative):
    assert src.is_file() and not src.is_symlink()
    if relative.endswith('.py'):
        relative += '.txt'
    raw = src.read_bytes()
    target = DEST / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open('xb') as stream:
        stream.write(raw)
    target.chmod(0o644)
    assert target.read_bytes() == raw
    entries.append({'path': relative, 'sourcePath': str(src), 'bytes': len(raw),
                    'sha256': hashlib.sha256(raw).hexdigest(), 'rawBytesPreserved': True})
for group, source in GROUPS.items():
    for src in sorted(source.rglob('*')):
        if src.is_file():
            copy(src, group + '/' + src.relative_to(source).as_posix())
for target, source in FILES.items():
    copy(source, target)
note = ('本包保存 S2 實際架構驗收、S2→A2 來源綁定與 root 公開提交原件。\n'
        'S2=cabe824128be31d0039dc6c94fa3f04cfb722a6b；A2=cbefb39b47fb0edcfaf003d2c1101929fea244c8。\n'
        '八條實際命令與輸出皆來自 S2；不聲稱 A2 重跑。root 審查結果的接受器程式僅存工具紀錄 0fea23，沒有另存原碼／stdout，不補造。\n'
        '獨立審查首次只讀接受器因 Git 中文路徑跳脫處理失誤為 1，改用 -z 後 0，詳獨立審查；不是正式 gate 失敗。\n'
        '執行來源一律以 .py.txt 或原有 .txt 保存相同 bytes，所有交付檔 mode 0644；未封裝 TMPDIR、測試副本或二進位。\n'
        'review-template.json 與歷史範本缺真來源值，不能正式執行驗收；本輪用 review-S2.json。\n'
        '原件中的絕對路徑記錄實跑環境，移機重現須提供新的 --root／--out 與固定來源 review；不能回寫舊收據。\n'
        '本包索引排除自身；原路徑與原 bytes SHA-256 逐檔列於來源索引.json。舊架構包完全不改。\n')
(DEST / '封裝說明.txt').write_text(note)
raw = (DEST / '封裝說明.txt').read_bytes()
entries.append({'path': '封裝說明.txt', 'sourcePath': '本次封裝新增說明', 'bytes': len(raw),
                'sha256': hashlib.sha256(raw).hexdigest(), 'rawBytesPreserved': False})
index = {'schema': 1, 'selfExcluded': '來源索引.json', 'entries': sorted(entries, key=lambda x: x['path'])}
(DEST / '來源索引.json').write_text(json.dumps(index, ensure_ascii=False, indent=2) + '\n')
files = sorted(p for p in DEST.rglob('*') if p.is_file())
assert {p.relative_to(DEST).as_posix() for p in files} == {e['path'] for e in entries} | {'來源索引.json'}
assert all(not p.is_symlink() and not p.stat().st_mode & 0o111 and p.suffix != '.py' for p in files)
result = {'destination': str(DEST), 'fileCount': len(files), 'totalBytes': sum(p.stat().st_size for p in files),
          'indexSha256': hashlib.sha256((DEST / '來源索引.json').read_bytes()).hexdigest(),
          'sourceBaseHeadActuallyTested': 'cabe824128be31d0039dc6c94fa3f04cfb722a6b',
          'metadataCommit': 'cbefb39b47fb0edcfaf003d2c1101929fea244c8', 'noExecutableSourceCopies': True}
receipt = BASE / 'architecture-owned-S2-package-receipt.json'
with receipt.open('x') as stream:
    json.dump(result, stream, ensure_ascii=False, indent=2)
    stream.write('\n')
print(json.dumps(result, ensure_ascii=False, indent=2))
