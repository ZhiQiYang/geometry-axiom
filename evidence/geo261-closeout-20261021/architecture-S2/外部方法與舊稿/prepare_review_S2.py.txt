def git(*args):
    return subprocess.check_output(['git', '-C', str(ROOT), *args])

def hashes(raw):
    return {'rawSha256': hashlib.sha256(raw).hexdigest(),
            'lfSha256': hashlib.sha256(raw.replace(b'\r\n', b'\n')).hexdigest()}

def dump(path, value):
    with path.open('x', encoding='utf-8') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2)
        stream.write('\n')

assert not git('status', '--porcelain=v1', '--untracked-files=no')
actual_diff = git('diff', '--name-status', '--no-renames', BASE, HEAD).decode().splitlines()
