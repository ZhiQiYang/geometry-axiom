#!/usr/bin/env python3
"""封存來源綁定後才完成的正常快轉原件，不回寫先前子包。"""
import hashlib
import json
from pathlib import Path

BASE = Path('/workspace/geo261-evidence')
SRC = BASE / 'source-S3-adoption'
DEST = BASE / 'final-bundle/source-S3-adoption'
DEST.mkdir(exist_ok=False)
def sha(raw): return hashlib.sha256(raw).hexdigest()
receipt = json.loads((SRC / 'receipt.json').read_bytes())
assert receipt['sourceSha'] == '19401ca5448e162ee6140fed37b64b83b31ddec8'
assert receipt['previousHead'] == '249571fe2477845d5330f6190dad965bf327f43b'
assert receipt['branch'] == 'codex/geo261-closeout-20261021' and receipt['remoteExact']
for row in receipt['commands']:
    assert row['exitCode'] == int((SRC / (row['label'] + '.exit-code')).read_text()) == 0
assert (SRC / 'remote.stdout').read_text() == receipt['sourceSha'] + '\trefs/heads/' + receipt['branch'] + '\n'
assert (SRC / 'head-before.stdout').read_text().strip() == receipt['previousHead']
assert (SRC / 'head-after.stdout').read_text().strip() == receipt['sourceSha']
assert not (SRC / 'status-before.stdout').read_bytes() and not (SRC / 'status-after.stdout').read_bytes()
records = []
for source in sorted(SRC.iterdir()) + [Path(__file__)]:
    assert source.is_file() and not source.is_symlink()
    name = source.name if source.parent == SRC else '封裝方法原碼.txt'
    raw = source.read_bytes()
    target = DEST / name
    target.write_bytes(raw)
    target.chmod(0o644)
    assert target.read_bytes() == source.read_bytes()
    records.append({'path': name, 'source': str(source), 'bytes': len(raw), 'sha256': sha(raw)})
(DEST / 'README.md').write_text('''# S3 正常快轉採入附錄

此為 `source-S3-binding` 封裝完成後才收錄的 root 原件。原綁定收據的「主工作樹仍 L2」是當時真值，不回填為 S3。此附錄保存 L2 原輪結束後，自己的 `codex/geo261-closeout-20261021` 分支正常快轉至 S3 `19401ca5448e162ee6140fed37b64b83b31ddec8`、一般 push 與 remote exact 原件。

這沒有更新 PR #288 的 `architecture/geo261-integration-20261016` 來源分支。採入收據的 freezeRelease 是協作觀察轉錄；原 canvas/native 收據另在各自子包，不能由本附錄代替完整產品驗收。

封裝僅讀取及逐 bytes 複製原件，核每條既有命令碼0與來源／remote／前後clean，不重新執行 Git 變更、連網、測試或建置。L3 尚未知。本子包索引只列來源複本，另有 README、原件清單及 SHA256SUMS 三個產生檔；SUMS 覆蓋除自身外全部檔案。
''')
(DEST / '原件清單.json').write_text(json.dumps(records, ensure_ascii=False, indent=2) + '\n')
files = sorted(p for p in DEST.iterdir() if p.is_file())
for path in files: path.chmod(0o644)
(DEST / 'SHA256SUMS').write_text(''.join(sha(p.read_bytes()) + '  ' + p.name + '\n' for p in files))
(DEST / 'SHA256SUMS').chmod(0o644)
files = sorted(p for p in DEST.iterdir() if p.is_file())
assert {p.name for p in files} == {r['path'] for r in records} | {'README.md', '原件清單.json', 'SHA256SUMS'}
result = {'bundle': str(DEST), 'copyCheckPassed': True, 'files': len(files), 'bytes': sum(p.stat().st_size for p in files),
          'indexSha256': sha((DEST / '原件清單.json').read_bytes()), 'sumsSha256': sha((DEST / 'SHA256SUMS').read_bytes())}
(BASE / 'source-S3-adoption-package-copycheck.json').write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
print(json.dumps(result, ensure_ascii=False, indent=2))
