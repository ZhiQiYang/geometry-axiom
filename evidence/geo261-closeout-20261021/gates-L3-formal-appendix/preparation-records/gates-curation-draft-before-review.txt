def utc():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def h(data):return hashlib.sha256(data).hexdigest()
def read(path):return json.loads(path.read_bytes())
def save(path,value):
    if path.exists():raise FileExistsError(str(path))
    path.write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n')
def ref(path):
    data=path.read_bytes();return dict(path=str(path),bytes=len(data),sha256=h(data),sourceMtimeNs=path.stat().st_mtime_ns)
def json_pointer(value,pointer):
    if not isinstance(pointer,str) or not pointer.startswith('/'):
        raise ValueError('完成後必須提供非空 JSON Pointer。')
    for token in pointer.split('/')[1:]:
        token=token.replace('~1','/').replace('~0','~')
        value=value[int(token)] if isinstance(value,list) else value[token]
    return value
def zero(value):return type(value) is int and value==0
def official_dir(phase):return G/('controls' if phase=='controls' else phase+'-run/'+phase)
def required_pins():
    paths=['preparation/instance.json','native/verification.json','native/completion-verification.json',
        'native/final-build-binary-index.json','gates/tool-observations.json',
        'gates/q38-progress-interpretation-correction.json','gates/q38-readonly-process-samples.jsonl']
    for phase in PHASES:
        paths += ['gates/verification/'+phase+'.json','gates/acceptance-review/'+phase+'.json',
            'gates/acceptance-execution/'+phase+'-primary-receipt.json',
            'gates/acceptance-execution/'+phase+'-detailed-receipt.json',
            'gates/launcher/'+phase+'/receipt.json',str((official_dir(phase)/'run-receipt.json').relative_to(BASE))]
    return paths

            if path.suffix in EXCLUDED_SUFFIXES or (mode&0o111 and path.suffix not in TEXT_SUFFIXES):

        assert not path.stat().st_mode&0o111 and target.suffix not in EXCLUDED_SUFFIXES

        '各 phase 的原始目錄結構完整保留；每份原件映射在 curation-source-mapping.json。正式 run-receipt、launcher/dispatch、acceptance-execution、verification、acceptance-review、全部進度與工具觀察一併保存。','',

    (OUT/'curation-README.zh-TW.md').write_text('\n'.join(lines)+'\n')
    assert git('rev-parse','HEAD')==SOURCE and git('status','--porcelain')==''
    for item in inventory:assert ref(Path(item['originalPath']))==dict(path=item['originalPath'],**{k:item[k] for k in ['bytes','sha256','sourceMtimeNs']})

        if path.is_file():

    APPENDIX.mkdir()
    proof=dict(sourceSha=SOURCE,curated=str(OUT),destination=str(BUNDLE),files=copied,fileCount=len(copied),

        artifactIndexSha256=h((OUT/'artifact-index.json').read_bytes()),copiedUtc=utc())

    result={k:proof[k] for k in ['sourceSha','curated','destination','fileCount','totalBytes','artifactIndexSha256']}
    result.update(copyVerificationSha256=h((APPENDIX/'copy-verification.json').read_bytes()),curationTool=ref(Path(__file__)),
        completionManifest=ref(manifest_path),formalToolOrProductExecuted=False,globalOrBinaryRead=False)
