cache={}
def h(data):return hashlib.sha256(data).hexdigest()
def data(path):
    path=Path(path)
    assert path.is_relative_to('/workspace/geo261-evidence') and not path.is_symlink(),str(path)
    if path not in cache:
        raw=path.read_bytes();raw.decode('utf-8');assert not raw.startswith(b'\x7fELF') and b'\0' not in raw,str(path)
        cache[path]=(raw,path.stat().st_mtime_ns)
    return cache[path][0]
def meta(path):
    raw=data(path);return dict(bytes=len(raw),sha256=h(raw),sourceMtimeNs=cache[Path(path)][1])
def load(path):return json.loads(data(path))
def checkref(ref):
    x=meta(Path(ref['path']));assert x['bytes']==ref['bytes'] and x['sha256']==ref['sha256'],ref['path']
    return x
def files(root):
    result=[]
    for p in root.rglob('*'):
        assert not p.is_symlink(),str(p)
        if p.is_file():result.append(p)
    return sorted(result)
