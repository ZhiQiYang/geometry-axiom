"""只啟動本輪唯讀接受器並保存真 wait 與完整 streams。"""
from pathlib import Path
import datetime,hashlib,json,subprocess,sys,time
r=Path('/workspace/geo261-evidence/browser-five-L3-independent-review');code=r/'peer-code-v2.txt'
argv=[sys.executable,'-B',str(code)];start=time.monotonic();utc=datetime.datetime.now(datetime.timezone.utc).isoformat()
with (r/'peer-v2.stdout').open('xb')as out,(r/'peer-v2.stderr').open('xb')as err:
    p=subprocess.Popen(argv,cwd='/workspace',stdin=subprocess.DEVNULL,stdout=out,stderr=err);rc=p.wait()
(r/'peer-v2.exit-code').write_text(str(rc)+'\n')
v={'argv':argv,'cwd':'/workspace','startedUTC':utc,'finishedUTC':datetime.datetime.now(datetime.timezone.utc).isoformat(),'seconds':time.monotonic()-start,'pid':p.pid,'observedReturncode':rc,'codeSha256':hashlib.sha256(code.read_bytes()).hexdigest(),'streams':{name:{'bytes':(r/name).stat().st_size,'sha256':hashlib.sha256((r/name).read_bytes()).hexdigest()}for name in ['peer-v2.stdout','peer-v2.stderr']},'scope':'只讀凍結五項 browser 原件，沒有產品／建置／browser 執行'}
(r/'execution-v2.json').write_text(json.dumps(v,ensure_ascii=False,indent=2)+'\n');print(json.dumps(v,ensure_ascii=False));raise SystemExit(rc)
