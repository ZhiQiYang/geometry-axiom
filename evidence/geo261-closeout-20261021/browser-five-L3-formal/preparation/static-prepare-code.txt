"""只讀核對 L3 browser 準備；不 build、不啟動 browser／服務。"""
from pathlib import Path
import ast, datetime, hashlib, json, subprocess

ROOT = Path('/workspace/geometry-axiom')
BASE = Path('/workspace/geo261-evidence')
OUT = BASE / 'browser-L3-preparation'
SHA = '6589248c9a94bdf0460d8f84dd1610e51048188b'
PARENT = '19401ca5448e162ee6140fed37b64b83b31ddec8'
OLD = json.loads((BASE/'browser-L2-preparation/L2-static-ready.json').read_bytes())
def git(*args): return subprocess.check_output(['git', *args], cwd=ROOT)
def sha(data): return hashlib.sha256(data).hexdigest()
def status(): return git('status', '--porcelain=v1', '--untracked-files=all').decode()
assert git('rev-parse', 'HEAD').decode().strip() == SHA
assert git('rev-parse', SHA+'^').decode().strip() == PARENT
assert status() == ''
paths = [row['path'] for row in OLD['sourceBindings']]
assert len(paths) == len(set(paths)) == 160
bindings = []
for name in paths:
    blob = git('show', SHA+':'+name)
    working = (ROOT/name).read_bytes()
    assert blob == working
    bindings.append({'path':name,'gitBlob':git('rev-parse',SHA+':'+name).decode().strip(),'bytes':len(blob),'sha256':sha(blob)})
wrappers=[]
for row in OLD['wrappers']:
    path=Path(row['path']); data=path.read_bytes()
    assert sha(data)==row['sha256'] and len(data)==row['bytes']
    ast.parse(data)
    wrappers.append({**row,'astValid':True})
assert len(wrappers)==4
golden_bytes=git('show',SHA+':geometry-web/cdp_sections.golden.json')
golden=json.loads(golden_bytes)
assert golden['mode']=='full' and len(golden['sections'])==85 and golden['total']==2261
canvas=BASE/'canvas-L3-formal-6589248c'
five=BASE/'browser-five-L3-formal-6589248c'
run_id='l3-6589248c-formal'
temp=Path('/tmp/geo261-browser-five-'+run_id)
assert not canvas.exists() and not five.exists() and not Path(str(five)+'-launch').exists() and not temp.exists()
assert not list(BASE.glob(canvas.name+'*')) and not list(BASE.glob(five.name+'*'))
base_env=['env','-u','CC','-u','PYTHONOPTIMIZE','python3','-B']
record={

 'sampledUTC':datetime.datetime.now(datetime.timezone.utc).isoformat(),

 'five':{'output':str(five),'bundle':str(BASE/'final-bundle/browser-five-L3-formal'),'launchDirectory':str(five)+'-launch','runId':run_id,'tempRoot':str(temp),'suiteOrder':['assembly','documents','grid','space','timber'],'limits':[900,600,900,900,1200],'launcherBaseArgv':base_env+[str(BASE/'browser-five-preparation/launch_browser_five.py'),'--sha',SHA,'--run-id',run_id,'--output',str(five)]},
 'browserRuntimeDirectory':OLD['browserRuntimeDirectory'],
 'buildExecuted':False,'browserExecuted':False,'serviceStarted':False,'releasePending':True,
