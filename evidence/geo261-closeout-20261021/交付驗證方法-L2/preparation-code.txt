#!/usr/bin/env python3
"""另建 E2 方法副本，只替三個已取得的來源常數，不執行驗證器。"""
import ast
import difflib
import hashlib
import json
from pathlib import Path

BASE = Path('/workspace/geo261-evidence')
OLD = BASE / 'source-binding-E-preparation/verify_E.py'
SNAPSHOT = BASE / 'source-binding-L2-preparation/snapshot-L2.json'
OUT = BASE / 'source-binding-E2-preparation'
L2 = '249571fe2477845d5330f6190dad965bf327f43b'
def sha(raw): return hashlib.sha256(raw).hexdigest()
old_raw = OLD.read_bytes()
assert sha(old_raw) == 'ffe3d3fe08ba3c51534b51724ae984e9c3a3774b02230e6d0f344f1bc4d897e3'
snapshot_raw = SNAPSHOT.read_bytes()
snapshot = json.loads(snapshot_raw)
assert snapshot['headBefore'] == snapshot['headAfter'] == L2
old_values = {'L': '43f85313c32410ac01406ab585846bbacafea278',
              'SNAPSHOT_HASH': '72283a36e47831867884f9682684267eddaf5d208acaf3891a0abeb9b3774455',
              'LEDGER_HASH': '868a80700c9d058e208684631e799a91659e7820b06c31c246b8af96c670a731'}
new_values = {'L': L2, 'SNAPSHOT_HASH': sha(snapshot_raw), 'LEDGER_HASH': snapshot['trackedLedger']['rawSha256']}
new_raw = old_raw
for key, value in old_values.items():
    old_line = (key + ' = "' + value + '"\n').encode()
    new_line = (key + ' = "' + new_values[key] + '"\n').encode()
    assert new_raw.count(old_line) == 1
    new_raw = new_raw.replace(old_line, new_line, 1)
restored = new_raw
for key, value in new_values.items():
    line = (key + ' = "' + value + '"\n').encode()
    assert restored.count(line) == 1
    restored = restored.replace(line, (key + ' = "' + old_values[key] + '"\n').encode(), 1)
assert restored == old_raw
tree = ast.parse(new_raw, filename='verify_E2.py')
assignments = {node.targets[0].id: ast.literal_eval(node.value) for node in tree.body
               if isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name)
               and node.targets[0].id in new_values}
assert assignments == new_values
args = [node.args[0].value for node in ast.walk(tree) if isinstance(node, ast.Call)
        and isinstance(node.func, ast.Attribute) and node.func.attr == 'add_argument' and node.args]
assert args == ['--root', '--expected-e', '--snapshot-l', '--review-file', '--review-sha256']
OUT.mkdir(exist_ok=False)
(OUT / 'verify_E2.py').write_bytes(new_raw)
(OUT / 'verify_E-original-code.txt').write_bytes(old_raw)
(OUT / 'constant-only.patch').write_text(''.join(difflib.unified_diff(old_raw.decode().splitlines(True), new_raw.decode().splitlines(True),
                                                               fromfile='既有已審/verify_E.py', tofile='L2三常數/verify_E2.py')))
(OUT / 'preparation-code.txt').write_bytes(Path(__file__).read_bytes())
result = {'sourceL2': L2, 'snapshotPath': str(SNAPSHOT), 'snapshotSha256': sha(snapshot_raw),
          'ledgerRawSha256': new_values['LEDGER_HASH'], 'originalVerifierPath': str(OLD),
          'originalVerifierSha256': sha(old_raw), 'newVerifierPath': str(OUT / 'verify_E2.py'),
          'newVerifierBytes': len(new_raw), 'newVerifierSha256': sha(new_raw),
          'oldConstants': old_values, 'newConstants': new_values,
          'reverseExactlyRestoresOriginalBytes': True, 'astParsePassed': True, 'parserArguments': args,
          'notExecuted': ['verify_E2', '任何 E2 controls', '產品測試／建置'],
          'notYetAvailable': ['E2 SHA', '最後總索引與 exact review']}
(OUT / 'preparation.json').write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
print(json.dumps(result, ensure_ascii=False, indent=2))
