"""唯讀比對已凍結 L3 canvas 證據包，不改包或執行產品。"""
from pathlib import Path
import collections,datetime,hashlib,json,stat
P=Path('/workspace/geo261-evidence/final-bundle/canvas-L3-formal')
sha=lambda b:hashlib.sha256(b).hexdigest()
index=json.loads((P/'source-index.json').read_bytes())
assert sha((P/'source-index.json').read_bytes())=='ac5f5d136afd7e0fd7bb7f8722d5d62681606cbe4f58a9ea64ab53149493aa3b'
assert sha((P/'SHA256SUMS').read_bytes())=='990c272bd510ad24bcd782fde5312adbd11274d3baabf8fc0a1e93305e0b3213'
assert index['sourceSha']=='6589248c9a94bdf0460d8f84dd1610e51048188b'
assert len(index['files'])==index['originalCount']==238
assert len({r['artifactPath'] for r in index['files']})==238
rows=[]
for r in index['files']:
    a=Path(r['sourcePath']).read_bytes();b=(P/r['artifactPath']).read_bytes()
    assert a==b and len(a)==r['bytes'] and sha(a)==r['sha256']
    rows.append({'path':r['artifactPath'],'bytes':len(b),'sha256':sha(b)})
assert sum(r['bytes'] for r in rows)==index['originalBytes']
allfiles={str(f.relative_to(P)):f for f in P.rglob('*') if f.is_file()}
assert not any(f.is_symlink() for f in P.rglob('*'))
assert len(allfiles)==241 and sum(f.stat().st_size for f in allfiles.values())==19863222
assert set(allfiles)=={r['artifactPath'] for r in index['files']}|{'README.md','source-index.json','SHA256SUMS'}
assert all(stat.S_IMODE(f.stat().st_mode)==0o644 for f in allfiles.values())
assert not any(f.suffix.lower() in ('.py','.js','.cjs','.exe','.so') for f in allfiles.values())
sums={}
for line in (P/'SHA256SUMS').read_text().splitlines():
    value,path=line.split('  ',1);assert path not in sums;sums[path]=value
assert set(sums)==set(allfiles)-{'SHA256SUMS'} and len(sums)==240
for path,value in sums.items():assert sha(allfiles[path].read_bytes())==value
suffixes=collections.Counter(f.suffix for f in allfiles.values())
assert suffixes['.png']==135 and suffixes['.gcanvas']==8 and suffixes['.csv']==6
text=(P/'README.md').read_text()
import re
readme_links=re.findall(r'\[[^\]]+\]\(([^)]+)\)',text)
for path in readme_links:assert (P/path).exists(), path
assert len([r for r in index['files'] if r['artifactPath'].startswith('peer/')])==18
for boundary in ('本輪均為 **reused**','1 個預期 uncaught exception','wait trace 為 disabled','Node 已開始後','不由本包推論全七項已完成','parentEnvironment 新增欄位而誤判 1','在建立封包前退出 1'):
    assert boundary in text
out={'schema':'geo261-canvas-L3-bundle-peer/1','checkedUTC':datetime.datetime.now(datetime.timezone.utc).isoformat(),
 'validationExitCode':0,'readOnly':True,'sourceSha':index['sourceSha'],'originalCount':238,'fileCount':241,'bytes':19863222,
 'checksums':240,'pngCount':135,'gcanvasCount':8,'csvCount':6,'exactOriginalBytes':True,'exactFileSet':True,
 'allRegular0644':True,'noExecutableCopies':True,'sourceIndexSha256':sha((P/'source-index.json').read_bytes()),
 'sha256sumsSha256':sha((P/'SHA256SUMS').read_bytes()),'files':rows,'readmeLinksChecked':readme_links,'includedPeerOriginals':18,
 'limitations':['只讀原件/索引/README，不重跑產品，不改子包；本包只證明 L3 canvas，不代替其餘驗收。',
 '本次核對雜湊與原 bytes，沒有逐張目視圖片。','原工具退出碼轉錄限制、首版 peer schema 誤判及封包路徑誤判均由原包保存。']}
target=Path('/workspace/geo261-evidence/canvas-L3-bundle-peer.json')
with target.open('x',encoding='utf-8') as f:json.dump(out,f,ensure_ascii=False,indent=2);f.write('\n')
print(json.dumps({k:out[k] for k in ('validationExitCode','originalCount','fileCount','bytes','checksums')},ensure_ascii=False))
