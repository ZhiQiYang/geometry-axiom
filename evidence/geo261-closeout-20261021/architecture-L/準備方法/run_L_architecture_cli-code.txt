#!/usr/bin/env python3
"""放行後才執行 L 架構 CLI；只寫全新外部收據，不重產 metadata 或重跑單元測試。"""
import argparse
import datetime
import hashlib
import json
import os
from pathlib import Path
import platform
import shutil
import subprocess
import sys
import time

sys.dont_write_bytecode = True
HEAD = "43f85313c32410ac01406ab585846bbacafea278"
MANIFEST = "geometry-core/domain-decoupling/product-architecture.json"
ARTIFACTS = ["geometry-core/domain-decoupling/" + name for name in (
    "product-architecture.json", "product-architecture-scan.json",
    "PRODUCT-ARCHITECTURE.md", "product-architecture-receipt.json")]
TOOLS = ["tests/architecture_contract.py", "tests/test_architecture_contract.py", "tests/test_product_architecture.py"]
COMMAND = ["env", "-u", "CC", "-u", "PYTHONOPTIMIZE", "python3", "-B",
           "tests/architecture_contract.py", "--root", ".", "--manifest", MANIFEST]
NATIVE_CWD = "/workspace/geo261-native-l"
NATIVE_OUT = "/workspace/geo261-evidence/native-final"
NATIVE_LAUNCHER = "/workspace/geo261-evidence/native-final-launcher"
NATIVE_LOG_SOURCE = "/tmp/geometry-tests/run_all-logs"
NATIVE_START = "2026-10-20T23:39:12.072242+00:00"
NATIVE_COMMAND = ["timeout", "2100", "python3", "tests/run_all.py", "--build"]
NATIVE_PINS = {
    "command.json": "8db09dd28b443c8289d395ec20d2545e7438a3bd9546b249b103e06a8776e2b6",
    "launcher/launcher-code.txt": "6c32561bf1e6fbe72f78167918be60e64418655fc201ac924d58d34ef0d10e7c",
    "launcher/wrapper-code.txt": "64ff6ee9c5a506c5ae89ec19df3bd1a03fe93ab1943bda5a6ef7a60bbdf328c4",
    "launcher/preflight.json": "dd6ad4cfb742a62ef774ae991eac1b36e01a12e835d06ac36a3d85283f90d440",
    "launcher/launcher-command.json": "4a573447714429fea8a69784244f8d214fad666e752f0162711677e6c7d9d346",
    "launcher/wrapper-pid.json": "3bc6ea1ca227e911fb8a3ca9866bb530c72cf64196b6f3d62088920507a9190c",
}


def require(ok, message):
    if not ok:
        raise RuntimeError(message)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def utc():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--root", required=True)
    parser.add_argument("--out", required=True)
    parser.add_argument("--native-dir", required=True)
    parser.add_argument("--native-launcher", required=True)
    parser.add_argument("--native-verification", required=True)
    parser.add_argument("--native-verification-sha256", required=True)
    args = parser.parse_args()
    root, out, native = (Path(value).resolve() for value in (args.root, args.out, args.native_dir))
    launcher = Path(args.native_launcher).resolve()
    require(str(native) == NATIVE_OUT and str(launcher) == NATIVE_LAUNCHER,
            "本次一次性 runner 只接受已固定的 L native 原收據目錄")
    require(out != root and root not in out.parents, "收據目錄必須在 repo 外")
    require(not out.exists(), "收據目錄已存在，禁止覆寫")
    out.mkdir(parents=True)
    git_commands = []

    def write_json(name, obj):
        (out / name).write_text(json.dumps(obj, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")

    def git(*argv):
        command = ["git", "-c", "core.fsmonitor=false", "-C", str(root), *argv]
        proc = subprocess.run(command, capture_output=True, check=False,
                              env=dict(os.environ, GIT_OPTIONAL_LOCKS="0"))
        git_commands.append({"argv": command, "exitCode": proc.returncode,
                             "stdoutBytes": len(proc.stdout), "stdoutSha256": sha(proc.stdout),
                             "stdout": proc.stdout.decode() if len(proc.stdout) < 4000 else None,
                             "stderr": proc.stderr.decode()})
        require(proc.returncode == 0, "來源讀取 Git 命令失敗")
        return proc.stdout

    def state():
        head = git("rev-parse", "HEAD").decode().strip()
        status = git("status", "--porcelain", "--untracked-files=all").decode()
        branch = git("branch", "--show-current").decode().strip()
        result = {"head": head, "status": status, "branch": branch, "files": {}}
        require(head == HEAD and not status, "來源不是固定 L 或工作樹有非忽略變更")
        for path in sorted(set(TOOLS + ARTIFACTS)):
            raw = (root / path).read_bytes()
            committed = git("show", HEAD + ":" + path)
            require(raw == committed, "工具或 metadata 工作bytes不符Git L：" + path)
            result["files"][path] = {"bytes": len(raw), "rawSha256": sha(raw)}
        return result

    def native_binding():
        references, raw_files = [], {}
        ref_out = out / "native-references"
        ref_out.mkdir()

        def read(path, relative):
            path = Path(path)
            require(path.is_file() and not path.is_symlink(), "native 原件缺漏或是連結：" + str(path))
            raw = path.read_bytes()
            if relative in NATIVE_PINS:
                require(sha(raw) == NATIVE_PINS[relative], "本輪 native 啟動原件 hash 不符：" + relative)
            target = ref_out / relative
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(raw)
            references.append({"source": str(path), "copiedAs": "native-references/" + relative,
                               "bytes": len(raw), "rawSha256": sha(raw)})
            raw_files[relative] = raw
            return raw

        def read_json(path, relative):
            return json.loads(read(path, relative))

        command = read_json(native / "command.json", "command.json")
        receipt = read_json(native / "receipt.json", "receipt.json")
        preflight = read_json(launcher / "preflight.json", "launcher/preflight.json")
        launch_command = read_json(launcher / "launcher-command.json", "launcher/launcher-command.json")
        launch_receipt = read_json(launcher / "launcher-receipt.json", "launcher/launcher-receipt.json")
        wrapper_pid = read_json(launcher / "wrapper-pid.json", "launcher/wrapper-pid.json")
        wrapper = read(launcher / "wrapper-code.txt", "launcher/wrapper-code.txt")
        launch_code = read(launcher / "launcher-code.txt", "launcher/launcher-code.txt")
        require(all(receipt.get(key) == value for key, value in command.items()),
                "native 結束收據與本輪啟動原件不一致")
        require(command["sourceSha"] == command["headBefore"] == receipt["headAfter"] == HEAD
                and command["cwd"] == NATIVE_CWD and command["command"] == NATIVE_COMMAND
                and command["startedUtc"] == NATIVE_START and command["jobLimitSeconds"] == 2100
                and not command["statusBefore"] and not receipt["statusAfter"]
                and receipt["sourceHashesUnchanged"] is True
                and command["sourceHashesBefore"] == receipt["sourceHashesAfter"],
                "native 命令、L 來源或前後工作樹不符")
        for path, digest in command["sourceHashesBefore"].items():
            require(sha(git("show", HEAD + ":" + path)) == digest,
                    "native 受記錄工具與 Git L bytes 不符：" + path)
        require(preflight["sourceSha"] == preflight["head"] == HEAD and not preflight["status"]
                and preflight["cwd"] == NATIVE_CWD and preflight["command"] == NATIVE_COMMAND
                and preflight["out"] == NATIVE_OUT and preflight["launcherOut"] == NATIVE_LAUNCHER
                and preflight["suiteLogDirectory"] == NATIVE_LOG_SOURCE
                and preflight["nativeGlobalAbsentBeforeStart"] is True
                and preflight["CCAbsent"] is True and preflight["PYTHONOPTIMIZEAbsent"] is True,
                "native 啟動前來源／環境／舊全域產物隔離未證明")
        require(command["wrapperSource"].encode() == wrapper
                and command["wrapperArgv"][-1].encode() == wrapper
                and launch_command["wrapperArgv"] == command["wrapperArgv"]
                and launch_command["launcherArgv"][3].encode() == launch_code
                and launch_command["wrapperCodeSha256"] == sha(wrapper)
                and preflight["wrapperSha256"] == sha(wrapper)
                and preflight["launcherSha256"] == sha(launch_code), "兩層執行碼原件不符")
        require(all(launch_receipt.get(key) == value for key, value in launch_command.items())
                and launch_command["sourceShaExpected"] == HEAD
                and launch_command["cwd"] == NATIVE_CWD
                and launch_command["nativeOutputPath"] == NATIVE_OUT
                and launch_receipt["wrapperPid"] == wrapper_pid["wrapperPid"] == command["wrapperPid"]
                and launch_receipt["wrapperStarted"] is True
                and launch_receipt["nativeReceiptPath"] == str(native / "receipt.json")
                and launch_receipt["nativeReceiptExists"] is True
                and launch_receipt["nativeReceiptReadError"] is None
                and launch_receipt["launcherError"] is None
                and launch_receipt["nativeSourceSha"] == HEAD
                and launch_receipt["nativeReceiptSha256"] == sha(raw_files["receipt.json"]),
                "父 launcher 與 native 收據綁定不符")
        raw_wait = receipt["mainChildRawWaitStatus"]
        require(type(raw_wait) is int and os.WIFEXITED(raw_wait)
                and os.waitstatus_to_exitcode(raw_wait) == receipt["exitCode"] == 0
                and launch_receipt["nativeMainChildRawWaitStatus"] == raw_wait
                and launch_receipt["nativeMainTimeoutReturnCode"] == 0
                and launch_receipt["wrapperReturnCode"] == launch_receipt["launcherExitCode"] == 0,
                "native 或父 launcher 尚未真實成功結束")
        for name in (".exit-code", ".launcher-exit-code"):
            require(read(launcher / name, "launcher/" + name) == b"0\n", "launcher 結束碼原件不符")
        reaped = read_json(native / "reaped.json", "reaped.json")
        events = [item for item in receipt["reaped"] if item["isMainProcess"]]
        require(len(events) == 1 and events[0]["pid"] == command["mainChildPid"]
                and events[0]["rawWaitStatus"] == raw_wait and events[0]["exitCode"] == 0
                and reaped["mainChildPid"] == command["mainChildPid"]
                and reaped["mainChildExitCode"] == 0 and reaped["events"] == receipt["reaped"],
                "native 主子程序 wait 紀錄不符")
        started = datetime.datetime.fromisoformat(NATIVE_START)
        finished = datetime.datetime.fromisoformat(receipt["finishedUtc"])
        require(started <= finished <= datetime.datetime.fromisoformat(launch_receipt["finishedUtc"]),
                "native 結束時間順序不符")

        # 完整 native 224 套件判定引用專責核對收據；此處不另造 run_all 裁判。
        verification = read_json(args.native_verification, "native-verification.json")
        require(sha(raw_files["native-verification.json"]) == args.native_verification_sha256,
                "native 完整核對收據 hash 與明示 pin 不符")
        index = read_json(native / "suite-log-source-index.json", "suite-log-source-index.json")
        require(index["sourceSha"] == HEAD and index["sourceDirectory"] == NATIVE_LOG_SOURCE
                and index["fileCount"] == len(index["files"])
                and index["totalBytes"] == sum(item["bytes"] for item in index["files"]),
                "本輪 native suite 原件索引來源／總數不符")
        indexed = {item["artifactPath"]: item for item in index["files"]}
        require(len(indexed) == len(index["files"]), "native suite 索引有重複路徑")
        named = read_json(native / "named-required-suites.json", "named-required-suites.json")
        expected_raw = read(native / "source-reference/run_all.expected.json", "source-reference/run_all.expected.json")
        require(expected_raw == git("show", HEAD + ":tests/run_all.expected.json"),
                "native 完整套件預期清單不是 Git L 原 bytes")
        expected = json.loads(expected_raw)["suites"]
        run_log = read(native / "run.log", "run.log")
        package_path = Path("/workspace/geo261-evidence/native-l-preparation/package-native-l.py")
        package_code = read(package_path, "packaging-tool-code.txt")
        ref_specs = {
            "command": (native / "command.json", raw_files["command.json"]),
            "receipt": (native / "receipt.json", raw_files["receipt.json"]),
            "launcherReceipt": (launcher / "launcher-receipt.json", raw_files["launcher/launcher-receipt.json"]),
            "sourceIndex": (native / "suite-log-source-index.json", raw_files["suite-log-source-index.json"]),
            "expected": (native / "source-reference/run_all.expected.json", expected_raw),
            "runLog": (native / "run.log", run_log),
            "packagingTool": (package_path, package_code),
        }
        expected_refs = {key: {"path": str(path), "bytes": len(raw), "sha256": sha(raw)}
                         for key, (path, raw) in ref_specs.items()}
        require(named["artifactReferences"] == expected_refs
                and verification["artifactReferences"] == {
                    **expected_refs, "namedRequiredSuites": {
                        "path": str(native / "named-required-suites.json"),
                        "bytes": len(raw_files["named-required-suites.json"]),
                        "sha256": sha(raw_files["named-required-suites.json"])}}
                and named["sourceSha"] == verification["sourceSha"] == HEAD,
                "native 完整核對／具名收據與真實原件路徑／bytes／hash 不符")
        require(verification["acceptedNative"] is True and verification["all224ExpectedMatched"] is True
                and verification["all224Passed"] is True
                and verification["expectedSuiteCount"] == verification["observedCompletedTableRows"]
                    == verification["observedProgressCount"] == len(verification["suiteRows"]) == len(expected) == 224
                and set(verification["suiteRows"]) == set(expected)
                and verification["expectedMismatches"] == verification["missingSuiteLogs"]
                    == verification["pythonFailures"] == []
                and verification["unallowedSkipCount"] == 0 and verification["timedOut"] is False
                and verification["runExitCode"] == verification["wrapperReturnCode"]
                    == verification["launcherExitCode"] == verification["persistedWrapperExitCode"]
                    == verification["persistedLauncherExitCode"] == 0
                and verification["mainChildRawWaitStatus"] == raw_wait
                and verification["startedUtc"] == NATIVE_START
                and verification["finishedUtc"] == receipt["finishedUtc"]
                and verification["elapsedSeconds"] == receipt["elapsedSeconds"] < 2100
                and verification["headBefore"] == verification["headAfter"]
                    == verification["packagingHead"] == HEAD
                and not verification["statusBefore"] and not verification["statusAfter"]
                and not verification["packagingStatus"] and verification["sourceHashesUnchanged"] is True
                and verification["rawLogCount"] == index["fileCount"]
                and verification["rawLogBytes"] == index["totalBytes"],
                "native 完整 224 核對尚未通過或不屬本輪；不開 CLI")
        suite_summaries = []
        for stem, source, total in [
            ("py_test_architecture_contract", "tests/test_architecture_contract.py", 205),
            ("py_test_product_architecture", "tests/test_product_architecture.py", 18),
        ]:
            bindings = []
            for suffix in (".json", ".log", ".unittest.log"):
                relative = "suite-logs/" + stem + suffix
                item = indexed[relative]
                require(item["sourcePath"] == NATIVE_LOG_SOURCE + "/" + stem + suffix
                        and type(item["sourceMtimeNs"]) is int
                        and started.timestamp() <= item["sourceMtimeNs"] / 1e9 <= finished.timestamp(),
                        "具名套件原件不是本輪來源路徑／時間：" + relative)
                raw = read(native / relative, relative)
                require(len(raw) == item["bytes"] and sha(raw) == item["sha256"],
                        "具名套件封存 bytes 與來源索引不符：" + relative)
                bindings.append(item)
            result = json.loads(raw_files["suite-logs/" + stem + ".json"])
            require(result["file"] == source
                    and result["total"] == result["run"] == result["passed"] == total
                    and all(result.get(key) == 0 for key in ("failed", "skipped", "explicit_skipped",
                            "unrun_skipped", "blocked_by_error", "expected_failed", "extra_errors"))
                    and result.get("import_error") is None
                    and result.get("failures") == [] and result.get("expected_failures") == [],
                    "L native 具名套件不是指定檔案的完整通過：" + stem)
            suite_name = "py:" + stem.removeprefix("py_")
            named_suite = named["suites"][suite_name]
            require(named_suite["expected"] == expected[suite_name] == {"mode": "count", "count": total, "unit": "test"}
                    and named_suite["row"] == verification["suiteRows"][suite_name]
                    and named_suite["row"]["total"] == named_suite["row"]["passed"] == total
                    and named_suite["row"]["failed"] == named_suite["row"]["skipped"] == 0
                    and sorted(named_suite["rawFiles"], key=lambda item: item["artifactPath"])
                        == sorted(bindings, key=lambda item: item["artifactPath"]),
                    "完整 native／具名核對與六份原件不一致：" + suite_name)
            suite_summaries.append({"suite": suite_name, "file": source,
                                    "passed": total, "skipped": 0, "rawFiles": bindings})
        write_json("native-references.json", {
            "說明": "只引用本輪 L native 原件；完整 224 核對沿用專責收據，不重跑 205／18 或另造 run_all 裁判",
            "sourceSha": HEAD, "nativeStartedUtc": NATIVE_START,
            "nativeVerificationSha256": args.native_verification_sha256,
            "nativeAggregateMeaning": verification["all224PassedMeaning"],
            "nativeAllowedSkips": verification["skips"],
            "nativeExitCodeOnlySuites": verification["exitCodeOnlySuites"],
            "具名套件": suite_summaries, "references": references,
        })

    launched = False
    try:
        before = state()
        write_json("before.json", before)
        native_binding()
        # 完整列出實際交給 CLI 的環境；不繼承無關服務或憑證變數。
        inherited = ("PATH", "HOME", "LANG", "LC_ALL", "LC_CTYPE", "TMPDIR", "TMP", "TEMP",
                     "SYSTEMROOT", "VIRTUAL_ENV")
        child_env = {key: os.environ[key] for key in inherited if key in os.environ}
        child_env["PYTHONDONTWRITEBYTECODE"] = "1"
        require("PATH" in child_env and "CC" not in child_env and "PYTHONOPTIMIZE" not in child_env,
                "CLI 環境不符")
        python_path = shutil.which("python3", path=child_env["PATH"])
        require(python_path and Path(python_path).resolve() == Path(sys.executable).resolve(),
                "runner 與命令內 python3 不是同一解譯器")
        require(shutil.which("env", path=child_env["PATH"]), "找不到 env 程式")
        write_json("environment.json", {"完整子程序環境": child_env,
                   "原環境CC是否存在": "CC" in os.environ,
                   "原環境PYTHONOPTIMIZE是否存在": "PYTHONOPTIMIZE" in os.environ,
                   "未繼承環境變數名稱": sorted(set(os.environ) - set(child_env)),
                   "pythonResolved": str(Path(python_path).resolve()), "pythonVersion": sys.version,
                   "platform": platform.platform(), "說明": "完整 argv 仍用 env -u CC -u PYTHONOPTIMIZE；不保存無關憑證值"})
        started, monotonic = utc(), time.monotonic()
        command_record = {"sourceSha": HEAD, "argv": COMMAND, "cwd": str(root), "startedUTC": started,
                          "runnerPath": str(Path(__file__).resolve()), "runnerRawSha256": sha(Path(__file__).read_bytes()),
                          "不執行": ["架構 metadata 重產", "generic 205 套件重跑", "product 18 套件重跑", "native 重跑"]}
        write_json("command.json", command_record)
        with (out / "stdout.json").open("xb") as stdout, (out / "stderr.log").open("xb") as stderr:
            pid = os.fork()
            if pid == 0:
                try:
                    os.chdir(root)
                    os.dup2(stdout.fileno(), 1)
                    os.dup2(stderr.fileno(), 2)
                    os.execvpe(COMMAND[0], COMMAND, child_env)
                except BaseException as error:
                    os.write(2, ("CLI 啟動失敗：" + repr(error) + "\n").encode())
                    os._exit(127)
            launched = True
            waited, raw_wait = os.waitpid(pid, 0)
            require(waited == pid, "回收的子程序PID不符")
        cli_exit = os.waitstatus_to_exitcode(raw_wait)
        finished, elapsed = utc(), time.monotonic() - monotonic
        (out / "cli.exit-code").write_text(str(cli_exit) + "\n", encoding="utf-8")
        post_error = None
        after = None
        try:
            after = state()
            require(after == before, "架構 CLI 前後來源／工具／四產物改變")
        except Exception as error:
            post_error = repr(error)
        write_json("after.json", {"state": after, "error": post_error})
        raw_stdout, raw_stderr = (out / "stdout.json").read_bytes(), (out / "stderr.log").read_bytes()
        parse_error, summary = None, None
        try:
            result = json.loads(raw_stdout)
            summary = {"ok": result["ok"], "sources": len(result["sources"]), "edges": len(result["edges"]),
                       "diagnostics": len(result["diagnostics"]), "trustedSourcesNotParsed": len(result["trustedSourcesNotParsed"])}
            require((cli_exit == 0) == (result["ok"] is True and not result["diagnostics"]), "CLI 退出碼與原 JSON 診斷不符")
        except Exception as error:
            parse_error = repr(error)
        wrapper_exit = cli_exit if cli_exit >= 0 else 128 - cli_exit
        if post_error or parse_error:
            wrapper_exit = wrapper_exit or 1
        write_json("receipt.json", {**command_record, "finishedUTC": finished, "elapsedSeconds": elapsed,
                   "childPid": pid, "rawWaitStatus": raw_wait, "cliExitCode": cli_exit, "wrapperExitCode": wrapper_exit,
                   "headBefore": before["head"], "headAfter": after["head"] if after else None,
                   "sourceAndMetadataUnchanged": after == before and post_error is None,
                   "postflightError": post_error, "stdoutParseError": parse_error, "summary": summary,
                   "stdoutBytes": len(raw_stdout), "stdoutSha256": sha(raw_stdout),
                   "stderrBytes": len(raw_stderr), "stderrSha256": sha(raw_stderr)})
        write_json("git-commands.json", git_commands)
        print(json.dumps({"out": str(out), "cliExitCode": cli_exit, "wrapperExitCode": wrapper_exit,
                          "rawWaitStatus": raw_wait, "summary": summary}, ensure_ascii=False))
        return wrapper_exit
    except Exception as error:
        write_json("failure.json", {"atUTC": utc(), "cliLaunched": launched, "error": repr(error)})
        write_json("git-commands.json", git_commands)
        raise


if __name__ == "__main__":
    raise SystemExit(main())
