"""只為已指定 exact 來源產生外部前置碼；不執行前置、建立工作樹或碰全域快取。"""
import argparse
import ast
import datetime
import difflib
import hashlib
import json
from pathlib import Path
import re
import shutil
import subprocess
import sys

HERE = Path(__file__).resolve().parent
TEMPLATE_SHA = '8f43fb35c0fb22882390d13cfbf92e3b71ecd5f616ff7d8b5add3d2a358019d5'
IDENTITY_SHA = '2fe0e01186664bc99ef466c7c6c2be6da1d79bf843cb25bd89fec69f82165f82'

def digest(data):
    return hashlib.sha256(data).hexdigest()

def main():
    if sys.flags.optimize:
        raise RuntimeError('拒絕 -O；本外部方法的檢查不可停用。')
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--instance', required=True)
    parser.add_argument('--source', required=True)
    args = parser.parse_args()
    if not re.fullmatch('[0-9a-f]{40}', args.source):
        parser.error('--source 只接受 root 指定的真實完整 SHA。')
    instance = Path(args.instance).resolve(strict=True)
    plan = json.loads(instance.read_bytes())
    if plan['sourceSha'] != args.source:
        raise ValueError('來源與 instance 不符。')
    if plan['expectedScope'] != dict(suites=224, pythonModules=188, countedTotal=5188):
        raise ValueError('事前範圍不是 224／188／5188。')
    if plan['formalNativeCommand'] != ['timeout', '2100', 'python3', 'tests/run_all.py', '--build'] or plan['stackBytes'] != 16777216:
        raise ValueError('原期限、正式 argv 或堆疊被改動。')
    if not {'CC', 'PYTHONOPTIMIZE'} <= set(plan['unsetEnvironment']):
        raise ValueError('父環境 unset 契約不完整。')
    if plan['readyToLaunch'] or plan['productExecuted'] or plan['worktreesCreated'] or plan['globalStorageTouched']:
        raise ValueError('此入口只處理尚未啟動的新實例。')
    root = Path(plan['root']).resolve(strict=True)
    if instance != root/'preparation/instance.json':
        raise ValueError('instance 路徑不在其新輪根目錄。')
    for key in ['nativeWorktree', 'gatesWorktree', 'nativeTmpdir', 'globalArchive', 'nativeOut', 'nativeLauncherOut']:
        if Path(plan[key]).exists():
            raise ValueError('新輪路徑已存在，拒絕覆寫：'+plan[key])
    sha = subprocess.check_output(['git', '-C', plan['repo'], 'rev-parse', '--verify', args.source+'^{commit}'], text=True).strip()
    if sha != args.source:
        raise ValueError('來源不是 exact 可解析 commit。')
    witness = subprocess.check_output(['git', '-C', plan['repo'], 'show', sha+':tests/test_geo261_host_authority.py'])
    tree = ast.parse(witness)
    methods = [c.name+'.'+n.name for c in tree.body if isinstance(c, ast.ClassDef)
               for n in c.body if isinstance(n, ast.FunctionDef) and n.name.startswith('test')]
    if len(methods) != 18 or len(set(methods)) != 18:
        raise ValueError('host_authority 事前 18 methods 契約不符。')
    if any(isinstance(n, ast.FunctionDef) and n.name == 'load_tests' for n in tree.body):
        raise ValueError('host_authority 有自訂 loader，AST 計數不足，需先審查。')
    original = (HERE/'preflight-template-code.txt').read_bytes()
    identity = HERE/'historical-l2-preflight/proc-infrastructure-identity.json'
    if digest(original) != TEMPLATE_SHA or digest(identity.read_bytes()) != IDENTITY_SHA:
        raise ValueError('已審前置碼或精確四身分原件 hash 改變。')
    text = original.decode()
    replacements = [
        ("BASE=Path('/workspace/geo261-evidence/revalidation-l2-249571fe2477-20261021')", 'BASE=Path('+repr(str(root))+')'),
        ("SHA='249571fe2477845d5330f6190dad965bf327f43b'", 'SHA='+repr(sha)),
    ]
    for old, new in replacements:
        if text.count(old) != 1:
            raise ValueError('原前置常數不唯一。')
        text = text.replace(old, new)
    reverse = text
    for old, new in reversed(replacements):
        if reverse.count(new) != 1:
            raise ValueError('新前置常數不唯一。')
        reverse = reverse.replace(new, old)
    if reverse.encode() != original:
        raise ValueError('除了 BASE／SHA 兩常數，原前置碼 bytes 改變。')
    ast.parse(text)
    destination = instance.parent/'prepare-native.py'
    diff = instance.parent/'prepare-native.diff'
    receipt = instance.parent/'preflight-materialization.json'
    if any(p.exists() for p in [destination, diff, receipt]):
        raise ValueError('已有前置碼或收據，拒絕覆寫。')
    launcher = Path(plan['nativeLauncherOut'])
    launcher.mkdir()
    saved_identity = launcher/identity.name
    shutil.copy2(identity, saved_identity)
    if saved_identity.read_bytes() != identity.read_bytes():
        raise ValueError('精確身分原件拷貝不一致。')
    destination.write_text(text)
    diff.write_text(''.join(difflib.unified_diff(original.decode().splitlines(True), text.splitlines(True), fromfile='已驗 L2 外部前置碼', tofile='新來源兩常數實例')))
    result = dict(sourceSha=sha, createdUtc=datetime.datetime.now(datetime.timezone.utc).isoformat(),
                  instance=str(instance), instanceSha256=digest(instance.read_bytes()),
                  preparedOnly=True, readyToLaunch=False, preflightExecuted=False,
                  productExecuted=False, worktreesCreated=False, globalStorageTouched=False,
                  templateSha256=TEMPLATE_SHA, generatedPath=str(destination), generatedSha256=digest(destination.read_bytes()),
                  onlyBaseAndSourceConstantsChanged=True,
                  identityReferenceSha256=IDENTITY_SHA, identityReferencePath=str(saved_identity),
                  identityScope='僅原四個 PID 的 UID 四值／start ticks／PPid／comm／完整 cmdline bytes 全符時可用；任何新增或不同身分 PermissionError 一律停止。',
                  identityLimitation='原四個身分的 fd/exe 未能讀取；未宣稱查過未讀 fd。新輪執行前置時重新核身分，歷史原件不是新輪採樣。',
                  hostAuthorityStaticMethods=methods, hostAuthoritySourceSha256=digest(witness),
                  futurePreflightArgv=['env', '-u', 'CC', '-u', 'PYTHONOPTIMIZE', 'python3', '-B', str(destination)],
                  nextStep='root 完成正式 P189 並放行後才執行 futurePreflightArgv；本入口不執行。')
    receipt.write_text(json.dumps(result, ensure_ascii=False, indent=2)+'\n')
    print(json.dumps(result, ensure_ascii=False))
    return 0

if __name__ == '__main__':
    raise SystemExit(main())
