def safe_text(path,data=None):
    assert path.is_file() and not path.is_symlink(),str(path)
    assert path.suffix not in ('.py','.js','.mjs','.cjs','.exe','.so','.dll','.o','.a'),str(path)
    assert not path.stat().st_mode&0o111,str(path)
    data=path.read_bytes() if data is None else data
    assert not data.startswith(b'\x7fELF') and b'\0' not in data,str(path)
    data.decode('utf-8')
    return data
