"""已獲放行的 L2 native 啟動前保全；不執行 build 或 test。"""
import ast,datetime,fcntl,hashlib,json,os,resource,subprocess,sys
from pathlib import Path

BASE=Path('/workspace/geo261-evidence/revalidation-l3-6589248c9a94-20261021')
PLAN=json.loads((BASE/'preparation/instance.json').read_text())
SHA='6589248c9a94bdf0460d8f84dd1610e51048188b'
assert PLAN['sourceSha']==SHA and not sys.flags.optimize
LAUNCHER=Path(PLAN['nativeLauncherOut']);assert LAUNCHER.is_dir()
def utc():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def save(name,value):(LAUNCHER/name).write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n')
def git(cwd,*args):return subprocess.check_output(['git',*args],cwd=cwd,text=True).strip()
def h(data):return hashlib.sha256(data).hexdigest()
ancestors={os.getpid()};pid=os.getppid()
while pid>0 and pid not in ancestors:
    ancestors.add(pid)
    try:
        fields=Path('/proc',str(pid),'stat').read_text().rsplit(')',1)[1].split();pid=int(fields[1])
    except FileNotFoundError:break
matches=[];denied=[]
for item in Path('/proc').iterdir():
    if not item.name.isdigit() or int(item.name) in ancestors:continue
    try:
        fields=(item/'stat').read_text().rsplit(')',1)[1].split()
        if fields[0]=='Z':continue
        raw=(item/'cmdline').read_bytes();argv=[v.decode(errors='replace') for v in raw.split(b'\0') if v]
        reason=[]
        try:
            exe=os.readlink(item/'exe')
            if exe.startswith('/tmp/geometry-tests/'):reason.append('global binary 正在執行')
        except FileNotFoundError:pass
        if any(v=='tests/run_all.py' or v.endswith('/tests/run_all.py') for v in argv):reason.append('run_all 正在執行')
        for fd in (item/'fd').iterdir():
            try:
                target=os.readlink(fd)
                if target.startswith('/tmp/geometry-tests/'):
                    info=(item/'fdinfo'/fd.name).read_text()
                    flags=int(next(line.split()[1] for line in info.splitlines() if line.startswith('flags:')),8)
                    if flags&os.O_ACCMODE:reason.append('全域檔案 writable fd: '+target)
            except (FileNotFoundError,ProcessLookupError):pass
        if reason:matches.append(dict(pid=int(item.name),argv=argv,reasons=reason))
    except (FileNotFoundError,ProcessLookupError):pass
    except PermissionError:denied.append(int(item.name))
locks=[]
for name in ['build.lock','run_all.lock']:
    path=Path('/tmp/geometry-tests')/name
    if path.exists():
        with path.open('r+') as f:
            fcntl.flock(f.fileno(),fcntl.LOCK_EX|fcntl.LOCK_NB)
            locks.append(dict(path=str(path),available=True))
            fcntl.flock(f.fileno(),fcntl.LOCK_UN)
save('no-live-writer-preflight-rescan.json',dict(sampledUtc=utc(),matches=matches,permissionDenied=denied,locks=locks,scope='當下 /proc exe、run_all argv、global writable fd 與兩鎖；不是未來並行寫入保證。'))
known=json.loads((LAUNCHER/'proc-infrastructure-identity.json').read_text())['items']
verified=[];unexplained=[]
for pid in denied:
    try:
        status=Path('/proc',str(pid),'status').read_text()
        stat=Path('/proc',str(pid),'stat').read_text().rsplit(')',1)[1].split()
        actual=dict(pid=pid,uid=[int(x) for x in next(x for x in status.splitlines() if x.startswith('Uid:')).split()[1:]],
            startTimeTicks=int(stat[19]),parentPid=int(stat[1]),comm=Path('/proc',str(pid),'comm').read_text().strip(),
            cmdlineHex=Path('/proc',str(pid),'cmdline').read_bytes().hex())
        expected=known.get(str(pid),{})
        if expected and all(actual[k]==expected[k] for k in actual):verified.append(actual)
        else:unexplained.append(dict(actual=actual,expected=expected))
    except OSError as exc:unexplained.append(dict(pid=pid,error=str(exc)))
save('no-live-writer-preflight-revised.json',dict(sampledUtc=utc(),matches=matches,unreadableInfrastructureVerified=verified,unexplainedPermissionErrors=unexplained,locks=locks,
    limitation='上述精確身份的環境程序 fd/exe 未能讀取；其餘可讀程序無 global writable fd／run_all／global exe。此為當下檢查，不宣稱未讀 fd 已核。'))
assert not matches and not unexplained,(matches,unexplained)
source_counts={}
for module,want in [('test_geo261_wood_contact_world',12),('test_geo261_host_facts_authority',27),('test_geo261_placement_definition',15)]:
    data=subprocess.check_output(['git','show',SHA+':tests/'+module+'.py'],cwd=PLAN['repo'])
    tree=ast.parse(data)
    methods=[c.name+'.'+n.name for c in tree.body if isinstance(c,ast.ClassDef) for n in c.body if isinstance(n,ast.FunctionDef) and n.name.startswith('test')]
    assert len(methods)==want and len(methods)==len(set(methods))
    assert not any(isinstance(n,ast.FunctionDef) and n.name=='load_tests' for n in tree.body)
    source_counts[module]=dict(expected=want,observed=len(methods),methods=methods,sourceSha256=h(data))
save('static-source-counts.json',dict(sourceSha=SHA,expectedScope=PLAN['expectedScope'],changedModules=source_counts,executedTests=False))
for key in ['nativeWorktree','gatesWorktree']:
    path=Path(PLAN[key]);assert not path.exists()
    command=['git','worktree','add','--detach',str(path),SHA]
    result=subprocess.run(command,cwd=PLAN['repo'],stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    (LAUNCHER/(key+'.stdout.log')).write_bytes(result.stdout);(LAUNCHER/(key+'.stderr.log')).write_bytes(result.stderr)
    save(key+'-creation.json',dict(command=command,exitCode=result.returncode,startedFrom=PLAN['repo']))
    assert result.returncode==0
    assert git(path,'rev-parse','HEAD')==SHA and not git(path,'status','--porcelain')
global_dir=Path(PLAN['globalDirectory']);archive=Path(PLAN['globalArchive'])
assert not archive.exists()
entries=[]
if global_dir.exists():
    for path in sorted(global_dir.rglob('*')):
        stat=path.lstat()
        item=dict(path=str(path.relative_to(global_dir)),mtimeNs=stat.st_mtime_ns,mode=stat.st_mode)
        if path.is_symlink():item.update(kind='symlink',target=os.readlink(path))
        elif path.is_file():
            data=path.read_bytes();item.update(kind='file',bytes=len(data),sha256=h(data))
        else:item.update(kind='directory')
        entries.append(item)
    save('prior-global-artifacts.json',dict(source=str(global_dir),destination=str(archive),sampledUtc=utc(),entries=entries))
    global_dir.rename(archive)
    for item in entries:
        path=archive/item['path']
        if item['kind']=='file':assert path.stat().st_size==item['bytes'] and path.stat().st_mtime_ns==item['mtimeNs']
else:save('prior-global-artifacts.json',dict(source=str(global_dir),destination=str(archive),sampledUtc=utc(),entries=[],sourceAbsent=True))
assert not global_dir.exists()
tmp=Path(PLAN['nativeTmpdir']);tmp.mkdir()
env=os.environ.copy()
prior_env={k:env.get(k) for k in PLAN['unsetEnvironment']}
for k in PLAN['unsetEnvironment']:env.pop(k,None)
env['PYTHONDONTWRITEBYTECODE']='1'
resource.setrlimit(resource.RLIMIT_STACK,(16777216,resource.getrlimit(resource.RLIMIT_STACK)[1]))
validation=subprocess.run(PLAN['nativeValidateOnlyArgv'],cwd=PLAN['nativeWorktree'],env=env,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
(LAUNCHER/'argument-validation.stdout.log').write_bytes(validation.stdout);(LAUNCHER/'argument-validation.stderr.log').write_bytes(validation.stderr)
save('argument-validation.json',dict(command=PLAN['nativeValidateOnlyArgv'],exitCode=validation.returncode,productExecuted=False))
assert validation.returncode==0
versions={}
for command in [['cc','--version'],['python3','--version'],['timeout','--version']]:
    r=subprocess.run(command,env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
    versions[command[0]]=dict(command=command,exitCode=r.returncode,stdout=r.stdout.decode(errors='replace'))
save('preflight.json',dict(sourceSha=SHA,checkedUtc=utc(),head=git(PLAN['nativeWorktree'],'rev-parse','HEAD'),status=git(PLAN['nativeWorktree'],'status','--porcelain'),
    parentEnvironmentBeforeUnset=prior_env,effectiveUnset={k:k not in env for k in PLAN['unsetEnvironment']},CCAbsent='CC' not in env,PYTHONOPTIMIZEAbsent='PYTHONOPTIMIZE' not in env,
    stackKiB=resource.getrlimit(resource.RLIMIT_STACK)[0]//1024,command=PLAN['formalNativeCommand'],jobLimitSeconds=2100,timeoutIncludesBuild=True,
    cwd=PLAN['nativeWorktree'],tmpdir=str(tmp),out=PLAN['nativeOut'],launcherOut=str(LAUNCHER),suiteLogDirectory=PLAN['suiteLogDirectory'],oldGlobalArchive=str(archive),
    oldEntries=len(entries),oldFileBytes=sum(i.get('bytes',0) for i in entries),nativeGlobalAbsentBeforeStart=not global_dir.exists(),versions=versions,
    expectedSuiteCount=224,expectedPythonCount=188,expectedCountedTotal=5188,allowedSkips=PLAN['allowedSkips'],sourceFiles=PLAN['sourceFiles']))
print(json.dumps(dict(preflight=str(LAUNCHER/'preflight.json'),sourceSha=SHA,readyForNative=True,productBuildOrTestsStarted=False,globalArchive=str(archive)),ensure_ascii=False))
