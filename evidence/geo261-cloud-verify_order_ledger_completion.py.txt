"""唯讀核對新來源的完整帳本、78筆原始判決及本人監督；不取代原 --check。"""
from collections import Counter
import hashlib
import json
from pathlib import Path
import re
import sys

HERE=Path(__file__).resolve().parent
PRODUCT=sys.argv[1]
assert re.fullmatch('[0-9a-f]{40}',PRODUCT)
OUT=HERE/('ledger-'+PRODUCT[:8])
def read(path):return json.loads(path.read_text(encoding='utf-8-sig'))
def main():
    assert not (HERE/'STOP-WSL').exists()
    assert not (OUT/'verified.json').exists()
    rec=read(OUT/'receipt.json')
    sup=read(HERE/('ledger-'+PRODUCT[:8]+'-supervisor.json'))
    assert rec['sha']==PRODUCT and rec['pass_check'] is True
    assert len(rec['steps'])==4 and all(row['exit']==0 for row in rec['steps'])
    assert rec['steps'][0]['command'][:5]==['git','-c','safe.directory=*','clone','-q']
    assert rec['steps'][1]['command']==['git','-c','advice.detachedHead=false','checkout','-q',PRODUCT]
    assert [row['command'] for row in rec['steps'][2:]]==[
        ['python3','-B','-u','tests/mutate.py'],['python3','-B','tests/mutate.py','--check']]
    assert sup['status']=='completed' and sup['child_exit']==sup['exit_code']==0
    assert not sup['cleanup']['remaining']
    assert PRODUCT in sup['command']
    raw=(OUT/'mutants_results.json').read_bytes()
    digest=hashlib.sha256(raw).hexdigest()
    assert digest==rec['ledger_sha256']
    records={key:value for key,value in json.loads(raw).items() if re.fullmatch(r'M\d+',key)}
    assert len(records)==78
    counts=Counter(value['verdict'] for value in records.values())
    assert counts==Counter(killed=70,crashed=1,equivalent=4,retired=3)==Counter(rec['counts'])
    active={key:value for key,value in records.items() if value['verdict']!='retired'}
    assert len(active)==75 and all(value['execution']['head']==PRODUCT for value in active.values())
    schemas=Counter(value['execution']['schema'] for value in active.values())
    assert schemas==Counter({'geometry-mutation-execution/1':57,'geometry-bridge-mutation-execution/1':18})
    rows=re.findall(r'^(M\d+)\s+(KILLED|CRASHED|EQUIVALENT|RETIRED|BLOCKED|SURVIVED)\b',
                    (OUT/'mutate.log').read_text(encoding='utf-8'),re.MULTILINE)
    assert len(rows)==len({key for key,_ in rows})==78
    assert dict(rows)=={key:value['verdict'].upper() for key,value in records.items()}
    result={'product_source_sha':PRODUCT,'passed':True,'ledger_sha256':digest,'counts':dict(counts),
            'execution_heads':[PRODUCT],'execution_schemas':dict(schemas),'raw_log_verdicts':len(rows),
            'steps':rec['steps'],'supervisor_exit':sup['exit_code'],
            'verifier_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
    (OUT/'verified.json').write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    print(json.dumps(result,ensure_ascii=False))
if __name__=='__main__':main()
