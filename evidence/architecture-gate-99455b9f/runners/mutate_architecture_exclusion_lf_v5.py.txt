"""Read-only source / TEMP mutant controls for architecture scanner extensions."""
import argparse
import ast
import difflib
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys
import tempfile

CONTROLS = {
    'LAYER_DIRECTION': ['test_direct_c_template_include_is_rejected', 'test_javascript_nested_template_tracks_dependency',
                        'test_javascript_template_keeps_transitive_closure', 'test_html_host_route_keeps_transitive_closure',
                        'test_core_cannot_depend_on_even_digest_locked_vendor', 'test_core_helper_cannot_launder_vendor',
                        'test_trusted_vendor_cannot_be_reached_by_core_or_template'],
    'SOURCE_UNREGISTERED': ['test_new_executable_cannot_be_omitted_from_manifest', 'test_standalone_source_cannot_be_unregistered'],
    'CORE_DYNAMIC_IMPORT': ['test_python_dynamic_import_alias_is_rejected', 'test_python_from_loader_alias_is_rejected',
                           'test_python_builtin_loader_escape_is_rejected'],
    'CORE_PATH_INJECTION': ['test_python_sys_path_alias_injection_is_rejected', 'test_python_from_sys_path_alias_is_rejected'],
    'FORBIDDEN_IDENTIFIER': ['test_concrete_template_enum_is_rejected', 'test_javascript_template_forbidden_identifier_is_executable'],
    'FORBIDDEN_STRING': ['test_concrete_material_string_is_rejected', 'test_javascript_template_forbidden_string_is_executable'],
    'TEMPLATE_EXPRESSION_TOKENS': ['test_javascript_nested_template_tracks_dependency',
                                 'test_javascript_template_keeps_transitive_closure',
                                 'test_javascript_template_forbidden_identifier_is_executable',
                                 'test_javascript_template_forbidden_string_is_executable',
                                 'test_javascript_template_regex_comment_cannot_swallow_dependency'],
    'HTML_ROUTE_OWNERSHIP_EDGE': ['test_html_host_route_cannot_bypass_layer_direction', 'test_html_host_route_keeps_transitive_closure'],
    'JS_ROUTE_OWNERSHIP_EDGE': ['test_host_literal_import_route_is_an_owned_edge_inside_template'],
    'VENDOR_TRUST_POLICY': ['test_trusted_vendor_bundle_must_be_host_only'],
    'VENDOR_TRUST_UNDECLARED': ['test_trusted_vendor_bundle_cannot_omit_owned_vendor_file'],
    'VENDOR_DIGEST_CHECK': ['test_trusted_vendor_digest_is_still_mandatory_and_current'],
    'C_STANDARD_HEADER_POLICY': ['test_unknown_c_header_cannot_be_declared_standard', 'test_platform_c_header_cannot_be_declared_standard'],
    'EXTERNAL_HOST_LAYER': ['test_explicit_host_platform_header_does_not_allow_core',
                            'test_host_platform_header_cannot_launder_through_core_helper',
                            'test_explicit_host_external_allowance_does_not_allow_core'],
    'IMMUTABLE_DIGEST': ['test_lf_exclusion_changed_character_is_rejected', 'test_raw_exclusion_remains_byte_exact_by_default',
                         'test_lf_exclusion_preserves_bare_carriage_return'],
    'LF_NORMALIZATION': ['test_lf_exclusion_same_content_accepts_crlf_and_lf'],
    'EXCLUSION_HASH_MODE': ['test_exclusion_rejects_unknown_hash_mode'],
}
COUNTS = {'LAYER_DIRECTION': 1, 'SOURCE_UNREGISTERED': 1, 'CORE_DYNAMIC_IMPORT': 3, 'CORE_PATH_INJECTION': 2,
          'FORBIDDEN_IDENTIFIER': 1, 'FORBIDDEN_STRING': 1, 'TEMPLATE_EXPRESSION_TOKENS': 1, 'HTML_ROUTE_OWNERSHIP_EDGE': 1}
COUNTS.update(JS_ROUTE_OWNERSHIP_EDGE=1, VENDOR_TRUST_POLICY=2, VENDOR_TRUST_UNDECLARED=1, VENDOR_DIGEST_CHECK=1)
COUNTS.update(C_STANDARD_HEADER_POLICY=1, EXTERNAL_HOST_LAYER=1)
COUNTS.update(IMMUTABLE_DIGEST=1, LF_NORMALIZATION=1, EXCLUSION_HASH_MODE=1)

def check(value, message):
    if not value:
        raise RuntimeError(message)

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def mutate(source, control):
    if control == 'EXTERNAL_HOST_LAYER':
        needle = 'and self.entries[path]["layer"] in entry["layers"]'
        check(source.count(needle) == 1, 'EXTERNAL_HOST_LAYER: wrong predicate count')
        result = source.replace(needle, 'and True  # TEMP removed H-only external allowance predicate\n                  ')
        ast.parse(result)
        return result
    tree = ast.parse(source)
    selected = []
    for node in ast.walk(tree):
        if control == 'LF_NORMALIZATION' and isinstance(node, ast.Assign) and ast.unparse(node) == "raw = raw.replace(b'\\r\\n', b'\\n')":
            selected.append(node)
            continue
        if not isinstance(node, ast.Expr) or not isinstance(node.value, ast.Call):
            continue
        call = node.value
        if control == 'TEMPLATE_EXPRESSION_TOKENS':
            match = ast.unparse(call) == 'pieces.extend(expression)'
        elif control == 'HTML_ROUTE_OWNERSHIP_EDGE':
            match = ast.unparse(call) == 'self.owned(self.root / target, path, line)'
        elif control == 'JS_ROUTE_OWNERSHIP_EDGE':
            match = ast.unparse(call) == 'self.owned(self.root / self.javascript_routes[path, tail[1].value], path, token.line)'
        elif control == 'VENDOR_DIGEST_CHECK':
            match = ast.unparse(call) == "self.check_digest(path, entry.get('sha256'), 'VENDOR_DIGEST')"
        elif control == 'C_STANDARD_HEADER_POLICY':
            match = isinstance(call.func, ast.Name) and call.func.id == 'require' and call.args and ast.unparse(call.args[0]) == 'name in C_STANDARD_HEADERS'
        elif control == 'IMMUTABLE_DIGEST':
            match = isinstance(call.func, ast.Name) and call.func.id == 'require' and len(call.args) > 1 and isinstance(call.args[1], ast.Name) and call.args[1].id == 'code'
        else:
            match = isinstance(call.func, ast.Name) and call.func.id == 'require' and len(call.args) > 1 and isinstance(call.args[1], ast.Constant) and call.args[1].value == control
        if match:
            selected.append(node)
    check(len(selected) == COUNTS[control], f'{control}: wrong AST target count')
    lines = source.splitlines(keepends=True)
    for node in sorted(selected, key=lambda item: item.lineno, reverse=True):
        lines[node.lineno - 1:node.end_lineno] = [' ' * node.col_offset + f'pass  # TEMP removed {control}\n']
    mutated = ''.join(lines)
    ast.parse(mutated)
    return mutated

def run(root, test, out, name, cases=()):
    command = [sys.executable, '-B', str(test), *['ArchitectureContract.' + case for case in cases]]
    proc = subprocess.run(command, cwd=root, env=dict(os.environ, PYTHONDONTWRITEBYTECODE='1', PYTHONIOENCODING='utf-8'), capture_output=True, timeout=60)
    raw = proc.stdout + proc.stderr
    (out / (name + '.log')).write_bytes(raw)
    text = raw.decode().replace('\r\n', '\n')
    count = re.findall(r'^Ran (\d+) tests? in ', text, re.M)
    return {'exit': proc.returncode, 'testsRun': int(count[0]) if len(count) == 1 else None,
            'failedCases': re.findall(r'^FAIL: ([^ ]+)', text, re.M), 'errorCases': re.findall(r'^ERROR: ([^ ]+)', text, re.M),
            'skippedCases': re.findall(r'^(test_[^ ]+) .* \.\.\. skipped ', text, re.M), 'logSha256': sha(raw)}

def main():
    p = argparse.ArgumentParser()
    p.add_argument('--root', type=Path, required=True)
    p.add_argument('--out', type=Path, required=True)
    p.add_argument('--expected-tests', type=int, required=True)
    a = p.parse_args()
    root, out = a.root.resolve(), a.out.resolve()
    check(not out.is_relative_to(root) and not out.exists(), 'output must be new and outside source checkout')
    out.mkdir(parents=True)
    paths = ['tests/architecture_contract.py', 'tests/test_architecture_contract.py', 'tests/run_all.py', 'tests/run_all.expected.json']
    before = {path: (root / path).read_bytes() for path in paths}
    report = {'head': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root, text=True).strip(),
              'runnerSha256': sha(Path(__file__).read_bytes()), 'inputsBefore': {path: sha(raw) for path, raw in before.items()},
              'baseline': None, 'mutants': [], 'scope': 'Full synthetic baseline, then focused independent TEMP removal controls; no production completion claim.'}
    status = 0
    try:
        expected = json.loads(before[paths[3]])['suites']['py:test_architecture_contract']['count']
        check(expected == a.expected_tests, 'registered test count mismatch')
        registry = ast.parse(before[paths[2]].decode())
        values = [ast.literal_eval(node.value) for node in registry.body if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == 'PY_NEEDS' for t in node.targets)]
        check(len(values) == 1 and values[0].get('test_architecture_contract') == [], 'suite not registered with no native needs')
        baseline = report['baseline'] = run(root, root / paths[1], out, 'baseline')
        check(baseline['exit'] == 0 and baseline['testsRun'] == a.expected_tests and not baseline['failedCases'] and not baseline['errorCases'], 'baseline failed')
        check(baseline['skippedCases'] in ([], ['test_symlink_escape_is_rejected'] if os.name == 'nt' else []), 'unexpected skip')
        source = before[paths[0]].decode().replace('\r\n', '\n')
        with tempfile.TemporaryDirectory(prefix='geo261-template-mutants-') as temp:
            temp_path = Path(temp)
            check(not temp_path.is_relative_to(root), 'TEMP inside checkout')
            for control, cases in CONTROLS.items():
                mutated = mutate(source, control)
                (temp_path / 'architecture_contract.py').write_text(mutated, encoding='utf-8')
                (temp_path / 'test_architecture_contract.py').write_bytes(before[paths[1]])
                diff = ''.join(difflib.unified_diff(source.splitlines(True), mutated.splitlines(True), fromfile=paths[0], tofile='TEMP/architecture_contract.py'))
                (out / (control + '.diff')).write_text(diff, encoding='utf-8')
                item = run(root, temp_path / 'test_architecture_contract.py', out, control, cases)
                item.update(control=control, expectedFailedCases=cases, mutantSha256=sha(mutated.encode()))
                report['mutants'].append(item)
                check(item['exit'] == 1 and item['testsRun'] == len(cases) and not item['errorCases'] and not item['skippedCases'] and sorted(item['failedCases']) == sorted(cases), control + ': unexpected mutant outcome')
        report['temporaryCopiesDeleted'] = not temp_path.exists()
    except Exception as error:
        status = 1
        report['failure'] = str(error)
    finally:
        report['inputsUnchanged'] = {path: (root / path).read_bytes() == raw for path, raw in before.items()}
        if not all(report['inputsUnchanged'].values()):
            status = 1
        report['exit'] = status
        report['status'] = 'passed-with-explicit-symlink-skip' if status == 0 and report['baseline']['skippedCases'] else ('passed' if status == 0 else 'failed')
        (out / 'mutation-results.json').write_text(json.dumps(report, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
    print(json.dumps({'status': report['status'], 'mutants': len(report['mutants']), 'failure': report.get('failure'), 'report': str(out / 'mutation-results.json')}, ensure_ascii=False))
    return status

if __name__ == '__main__':
    raise SystemExit(main())
