"""TEMP guard-removal evidence for supported comprehension result provenance."""
import ast
import hashlib
import importlib.machinery
import importlib.util
import json
from pathlib import Path
import sys

BASE = Path(__file__).with_name('mutate_architecture_embedded_v8.py.txt')
loader = importlib.machinery.SourceFileLoader('architecture_mutation_v8', str(BASE))
spec = importlib.util.spec_from_loader(loader.name, loader)
prior = importlib.util.module_from_spec(spec)
loader.exec_module(prior)
runner = prior.runner
previous_mutate = prior.mutate
NEW = {
    'PY_COMPREHENSION_RESULT_PROVENANCE': [
        'test_python_list_comprehension_unpack_cannot_read_host_loaded_template',
        'test_python_list_comprehension_index_cannot_read_host_loaded_template',
        'test_python_set_comprehension_result_preserves_sys_provenance',
        'test_python_generator_result_preserves_sys_provenance',
        'test_python_dict_comprehension_value_preserves_sys_provenance',
        'test_python_dict_comprehension_key_preserves_sys_provenance',
        'test_python_nested_comprehension_result_preserves_sys_provenance',
    ],
}
runner.CONTROLS.update(NEW)
runner.COUNTS.update({name: 1 for name in NEW})

def mutate(source, control):
    if control != 'PY_COMPREHENSION_RESULT_PROVENANCE':
        return previous_mutate(source, control)
    wanted = 'isinstance(node, (ast.ListComp, ast.SetComp, ast.DictComp, ast.GeneratorExp))'
    selected = [node for node in ast.walk(ast.parse(source))
                if isinstance(node, ast.If) and ast.unparse(node.test) == wanted]
    runner.check(len(selected) == 1, control + ': wrong AST result-provenance branch count')
    node = selected[0]
    lines = source.splitlines(keepends=True)
    lines[node.lineno - 1:node.end_lineno] = [' ' * node.col_offset + 'pass  # TEMP removed ' + control + '\n']
    result = ''.join(lines)
    ast.parse(result)
    return result

runner.mutate = mutate
runner.__file__ = __file__
if __name__ == '__main__':
    status = runner.main()
    out = Path(sys.argv[sys.argv.index('--out') + 1])
    path = out / 'mutation-results.json'
    report = json.loads(path.read_text(encoding='utf-8'))
    report['baseRunners'] = [{'path': str(item), 'sha256': hashlib.sha256(item.read_bytes()).hexdigest()}
                             for item in [BASE, prior.BASE, prior.prior.BASE, prior.prior.prior.BASE]]
    report['comprehensionResultControls'] = list(NEW)
    path.write_text(json.dumps(report, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
    raise SystemExit(status)
