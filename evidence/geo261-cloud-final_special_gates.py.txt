#!/usr/bin/env python3
"""經既有鎖與停止監督依序驗最終 SHA；任一關卡失敗就停，禁止吞退出碼。"""
from datetime import datetime,timezone
import json
import os
from pathlib import Path
import re
import subprocess
import sys
import time

SHA=sys.argv[1]
if not re.fullmatch('[0-9a-f]{40}',SHA):raise SystemExit('需完整 SHA')
HERE=Path(__file__).resolve().parent
TOOLS=Path('/mnt/d/geometry-axiom-work/wsl-ci-tools')
OUT=HERE/('special-'+SHA[:8])
ENV=dict(os.environ,PATH='/opt/node/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin',LANG='C.UTF-8',LC_ALL='C.UTF-8',PYTHONIOENCODING='utf-8',PYTHONDONTWRITEBYTECODE='1')

def main():
    OUT.mkdir()
    rec={'sha':SHA,'started':datetime.now(timezone.utc).isoformat(),'steps':[]}
    def run(name,argv,log):
        if log.exists():raise RuntimeError('拒絕覆蓋舊收據：'+str(log))
        print('開始 '+name,flush=True);start=time.monotonic()
        with log.open('wb') as f:p=subprocess.run(argv,env=ENV,stdout=f,stderr=subprocess.STDOUT)
        row={'name':name,'command':list(map(str,argv)),'exit':p.returncode,'seconds':time.monotonic()-start,'log':str(log)}
        rec['steps'].append(row);(OUT/'receipt.json').write_text(json.dumps(rec,ensure_ascii=False,indent=2)+'\n')
        print(json.dumps(row,ensure_ascii=False),flush=True)
        if p.returncode:raise SystemExit(p.returncode if p.returncode>0 else 128-p.returncode)
    run('P52 最終交錯量測及五個等待對照',['python3','-B','-u',str(HERE/'postfix_validate.py'),SHA],HERE/('postfix-'+SHA[:8]+'-launch.log'))
    run('P52 原始退出碼與裁判雜湊獨立核對',['python3','-B',str(HERE/'verify_postfix_receipt.py'),SHA],OUT/'p52-verify.log')
    run('Q38 全斷面突變閘門',['bash',str(TOOLS/'q38/q38_gate.sh'),SHA,'codexfinal','-','gate'],OUT/'q38.log')
    run('§7 完整卡片參照與公理突變',['bash',str(TOOLS/'card_ref_mutations.sh'),SHA,'codexfinal'],OUT/'card-reference.log')
    run('P-189 三方載回、dump 與交錯效能',['python3','-B','-u',str(HERE/'p189_verify.py'),SHA],OUT/'p189.log')
    rec.update(finished=datetime.now(timezone.utc).isoformat(),passed=True)
    (OUT/'receipt.json').write_text(json.dumps(rec,ensure_ascii=False,indent=2)+'\n')
    print('專項全部通過 '+SHA,flush=True)

if __name__=='__main__':main()
