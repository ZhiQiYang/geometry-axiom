"""Repository-external Reading A policy draft; not installed or executed.

All oracle timbers are complete output facts. Native W must additionally use
its real placed/member/placement facts; this draft is not a native shortcut.
Oracle Region has no FIXED/FREE or placement field: this draft presumes the
oracle's existing all-FIXED, one-placement model and cannot verify those native
negative cases. The bool interface is explanatory only; the installed policy
needs a tri-state decision so an explicit bare-long-RC refusal cannot fall back
to an older indoor-only acceptance path.
"""


def reading_a_end_supported(reg, timber, others, axis, at):
    """Check the approved fixed-boundary endpoint using actual RC faces.

    Uses the oracle's existing rng/iv_norm/iv_subtract/leftover primitives when
    inserted into framing_oracle.py. It does not choose a layout or alter K's
    complete-face primitive. At most a typed W policy admission is returned.
    """
    if timber["kind"] != "boundary" or timber.get("core_role", "boundary") != "boundary":
        return False
    rect = timber["rect"]
    span, cross = rng(rect, axis), rng(rect, 1 - axis)
    if at not in span or cross[1] <= cross[0] or leftover(rect, reg.rects()):
        return False
    # Preserve the product's exact locating-edge rule: covering an entire
    # edge wins, then longest positive overlap, then original edge order.
    edge = None
    for p, q in reg.outer_edges():
        if p[1 - axis] != q[1 - axis] or p[1 - axis] not in cross:
            continue
        lo, hi = sorted((p[axis], q[axis]))
        overlap_width = min(span[1], hi) - max(span[0], lo)
        if overlap_width <= 0:
            continue
        covers = span[0] <= lo and hi <= span[1]
        if edge is None or (covers and not edge[2]) or (covers == edge[2] and overlap_width > edge[3]):
            edge = (lo, hi, covers, overlap_width)
    if edge is None or not edge[2] or at not in edge[:2]:
        return False
    # Do not constrain the target timber's longitudinal axis: I20 boundary
    # members have the same direction, including approved 13 mm step example B.
    contacts = []
    low_end = at == span[0]
    for other in others:
        if other is timber:
            continue
        other_span, other_cross = rng(other["rect"], axis), rng(other["rect"], 1 - axis)
        if (other_span[1] if low_end else other_span[0]) != at:
            continue
        lo, hi = max(cross[0], other_cross[0]), min(cross[1], other_cross[1])
        if hi > lo:
            contacts.append((lo, hi))
    contacts = iv_norm(contacts)
    if not contacts:
        return False
    width = cross[1] - cross[0]
    # Measure each actual continuous RC face before intersecting it with this
    # end or subtracting timber coverage. Extra collinear vertices must not
    # turn a 30 mm face into two supposedly short 15 mm faces.
    for ring in reg.rings:
        faces = []
        for i, p in enumerate(ring):
            q = ring[(i + 1) % len(ring)]
            if p[axis] == q[axis] == at:
                faces.append(tuple(sorted((p[1 - axis], q[1 - axis]))))
        for lo, hi in iv_norm(faces):
            visible = (max(lo, cross[0]), min(hi, cross[1]))
            if hi - lo >= width and visible[1] > visible[0] and iv_subtract([visible], contacts):
                return False
    return True
