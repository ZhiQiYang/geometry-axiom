def read(path):

def sha(data):
    return hashlib.sha256(data).hexdigest()

def check_node_raw(directory, observation):
    text = (directory / 'stdout.log').read_text() + (directory / 'stderr.log').read_text()
    for field, marker in [('failures', 'fail'), ('skips', 'skipped')]:
        match = re.search(r'(?:#|ℹ) ' + marker + r' (\d+)', text)
        assert match and int(match[1]) == observation[field]
    assert observation['failed_names'] == [line for line in text.splitlines() if line.startswith(('✖ ', 'not ok '))]

    mutated = source.decode()

        assert mutated.count(before) == 1
        mutated = mutated.replace(before, after, 1)

    red, restored = row['red'], row['restored']
    check_node_raw(extracted / '13118903439' / name, red)
    check_node_raw(extracted / '13118903439' / name / 'restored', restored)
    verdict = 'killed' if red['exit_code'] == 1 and red['failures'] and red['skips'] == 0 and restored['exit_code'] == 0 and any(expected_tags[name] in value for value in red['failed_names']) else 'NOT_KILLED'
    assert row['verdict'] == verdict == 'killed'
    assert restored['failures'] == restored['skips'] == 0 and restored['failed_names'] == []
    card_rows.append({'name': name, 'verdict': verdict, 'expectedTag': expected_tags[name],
                      'redExit': red['exit_code'], 'redFailures': red['failures'], 'restoredExit': restored['exit_code']})
