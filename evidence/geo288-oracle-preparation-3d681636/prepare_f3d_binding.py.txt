def git(*args):
    return recorder.run(['git','--git-dir',str(GIT),*args],cwd=BASE,timeout=120,
                        source_variant='read-only-immutable-F3d-oracle-binding')[1]


def literal(tree,name):
    return next(ast.literal_eval(node.value) for node in tree.body if isinstance(node,ast.Assign)
                and any(isinstance(target,ast.Name) and target.id==name for target in node.targets))


def tree_rows(raw):
    result={}
    for row in raw.split(b'\0'):
        if not row:continue
        meta,path=row.split(b'\t',1);mode,kind,oid=meta.decode().split()
        assert kind=='blob'
        result[path.decode()]={'mode':mode,'git_blob_oid':oid}
    return result
