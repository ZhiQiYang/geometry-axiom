#!/usr/bin/env python3
"""Reproduce the GEO-261 P0 proposal gate's four guard-removal witnesses.

Usage (the script may retain its .py.txt suffix):
    python mutate_proposal_gate.py.txt --root /path/to/repo --out /path/to/new-receipts
    python mutate_proposal_gate.py.txt --root /path/to/repo --out /path/to/new-receipts --python /path/to/python3

The official 29-test baseline must pass first. Only controlled copies in a
system temporary directory are mutated; that directory is deleted on exit.
Repository files remain read-only. Logs and mutation-results.json are retained
under --out. Existing receipt files are never overwritten. Runner exit 0 means
the baseline passed and all four mutants failed in the expected named cases;
their expected child exit code is 1. Runner exit 1 means a contract/evidence
check failed; exit 2 means execution was blocked by the environment.

This proves the P0 proposal validator has executable rejection witnesses. It
does not prove implementation decoupling, semantic approval, or product parity.
"""
from __future__ import annotations

import argparse
import ast
from collections import Counter
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys
import tempfile


EXPECTED_TESTS = 29
GUARDS = (
    ("SOURCE_HASH", 1, [
        "test_rejects_changed_ssot_even_if_manifest_stays_self_consistent",
        "test_rejects_source_hash_drift",
        "test_rejects_joint_ssot_map_and_review_baseline_rewrite",
    ]),
    ("CLAUSE_BEFORE", 1, ["test_rejects_clause_before_drift"]),
    ("MIXED_LEGACY", 2, ["test_rejects_joint_map_and_review_legacy_deletion"]),
    ("PREAMBLE_SCOPES", 1, ["test_rejects_joint_map_and_review_preamble_scope_drift"] * 3),
)
INPUTS = {
    "test": "tests/test_domain_decoupling_proposal.py",
    "map": "geometry-core/domain-decoupling/clause-map.json",
    "review": "geometry-core/domain-decoupling/AXIOM-PROPOSAL.md",
    "ssot": "公理層.md",
    "expected": "tests/run_all.expected.json",
}


class EvidenceFailure(Exception):
    pass


def check(condition, message):
    if not condition:
        raise EvidenceFailure(message)


def sha256(data):
    return hashlib.sha256(data).hexdigest()


def remove_guard(source, code, expected_guard_count):
    guards = []
    for node in ast.walk(ast.parse(source)):
        if not isinstance(node, ast.Expr) or not isinstance(node.value, ast.Call):
            continue
        call = node.value
        if (isinstance(call.func, ast.Name) and call.func.id == "require"
                and len(call.args) > 1 and isinstance(call.args[1], ast.Constant)
                and call.args[1].value == code):
            guards.append(node)
    check(len(guards) == expected_guard_count,
          f"{code}: expected {expected_guard_count} guard statements, found {len(guards)}")
    lines = source.splitlines(keepends=True)
    for guard in sorted(guards, key=lambda node: node.lineno, reverse=True):
        indent = lines[guard.lineno - 1][:guard.col_offset]
        lines[guard.lineno - 1:guard.end_lineno] = [
            indent + f"pass  # TEMP MUTANT: removed {code} guard\n"
        ]
    return "".join(lines).encode("utf-8")


def run_suite(python, path, root, out, name, timeout):
    env = dict(os.environ, GEOMETRY_PROPOSAL_ROOT=str(root), PYTHONIOENCODING="utf-8")
    process = subprocess.run([python, str(path)], cwd=root, env=env,
                             capture_output=True, timeout=timeout)
    raw_log = process.stdout + process.stderr
    log = raw_log.decode("utf-8").replace("\r\n", "\n")
    log_name = name + ".log"
    (out / log_name).write_bytes(raw_log)
    counts = re.findall(r"^Ran (\d+) tests? in ", log, re.MULTILINE)
    return {
        "exit": process.returncode,
        "testsRun": int(counts[-1]) if counts else None,
        "failedCases": re.findall(r"^FAIL: ([^ ]+)", log, re.MULTILINE),
        "errors": len(re.findall(r"^ERROR:", log, re.MULTILINE)),
        "cleanOK": bool(re.search(r"^OK$", log, re.MULTILINE)),
        "log": log_name,
        "logSha256": sha256((out / log_name).read_bytes()),
    }


def main():
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("--root", type=Path, required=True, help="Repository containing the frozen P0 proposal and 29-test gate")
    parser.add_argument("--out", type=Path, required=True, help="Receipt directory; existing receipt files are refused")
    parser.add_argument("--python", default=sys.executable, help="Python executable for all child test runs (default: this interpreter)")
    parser.add_argument("--timeout", type=float, default=60, help="Maximum seconds per test run (default: 60)")
    args = parser.parse_args()
    root, out = args.root.resolve(), args.out.resolve()
    if not root.is_dir() or args.timeout <= 0:
        parser.error("--root must be a directory and --timeout must be positive")
    for relative in INPUTS.values():
        if not (root / relative).is_file():
            parser.error(f"required input missing: {root / relative}")
    receipt_names = ["baseline.log", "mutation-results.json"] + [code.lower() + ".log" for code, _, _ in GUARDS]
    if out.exists() and (not out.is_dir() or any((out / name).exists() for name in receipt_names)):
        parser.error("--out must not contain existing receipt files; choose a fresh output directory")
    out.mkdir(parents=True, exist_ok=True)
    snapshots = {name: (root / relative).read_bytes() for name, relative in INPUTS.items()}
    report = {
        "schema": "geo261-p0-mutation-evidence/1",
        "createdAt": datetime.now(timezone.utc).isoformat(),
        "pythonExecutable": args.python,
        "runnerScriptSha256": sha256(Path(__file__).read_bytes()),
        "runnerScriptLfSha256": sha256(Path(__file__).read_bytes().replace(b"\r\n", b"\n")),
        "expectedTests": EXPECTED_TESTS,
        "officialTestSha256": sha256(snapshots["test"]),
        "officialTestLfSha256": sha256(snapshots["test"].replace(b"\r\n", b"\n")),
        "proposalMapSha256": sha256(snapshots["map"]),
        "proposalMapLfSha256": sha256(snapshots["map"].replace(b"\r\n", b"\n")),
        "proposalReviewSha256": sha256(snapshots["review"]),
        "proposalReviewLfSha256": sha256(snapshots["review"].replace(b"\r\n", b"\n")),
        "ssotLfSha256": sha256(snapshots["ssot"].replace(b"\r\n", b"\n")),
        "hashNormalization": "lfSha256 replaces CRLF with LF only; raw sha256 is also retained.",
        "inputFiles": {name: {"path": INPUTS[name], "sha256": sha256(data),
                              "lfSha256": sha256(data.replace(b"\r\n", b"\n"))} for name, data in snapshots.items()},
        "temporaryPolicy": "Only controlled copies are mutated; the system temporary directory is deleted on exit.",
        "temporaryCopiesDeleted": False,
        "baseline": None,
        "mutants": [],
    }
    exit_code = 0
    temp_root = None
    try:
        expected = json.loads(snapshots["expected"])["suites"]["py:test_domain_decoupling_proposal"]
        check(expected == {"mode": "count", "count": EXPECTED_TESTS, "unit": "test"},
              f"expected test registration differs from {EXPECTED_TESTS}: {expected}")
        baseline = run_suite(args.python, root / INPUTS["test"], root, out, "baseline", args.timeout)
        report["baseline"] = baseline
        check(baseline["exit"] == 0 and baseline["testsRun"] == EXPECTED_TESTS
              and baseline["cleanOK"] and not baseline["failedCases"] and baseline["errors"] == 0,
              "official baseline did not pass all 29 tests without skips; no mutants were run")
        with tempfile.TemporaryDirectory(prefix="geo261-p0-mutants-") as temporary:
            temp_root = Path(temporary).resolve()
            check(temp_root.parent == Path(tempfile.gettempdir()).resolve()
                  and temp_root.name.startswith("geo261-p0-mutants-"),
                  "temporary cleanup path is outside the intended system temporary directory")
            for code, expected_guard_count, expected_failures in GUARDS:
                mutant = remove_guard(snapshots["test"].decode("utf-8"), code, expected_guard_count)
                target = temp_root / ("test_proposal_without_" + code.lower() + ".py")
                target.write_bytes(mutant)
                result = run_suite(args.python, target, root, out, code.lower(), args.timeout)
                result.update(guard=code, removedStatements=expected_guard_count, mutantSha256=sha256(mutant),
                              mutantLfSha256=sha256(mutant.replace(b"\r\n", b"\n")))
                report["mutants"].append(result)
                check(result["exit"] == 1 and result["testsRun"] == EXPECTED_TESTS
                      and result["errors"] == 0
                      and Counter(result["failedCases"]) == Counter(expected_failures),
                      f"{code}: expected named rejection witnesses did not fail exactly as specified")
        report["temporaryCopiesDeleted"] = not temp_root.exists()
        check(report["temporaryCopiesDeleted"], "temporary mutant directory was not removed")
    except EvidenceFailure as error:
        report["failure"] = str(error)
        exit_code = 1
    except (OSError, subprocess.TimeoutExpired, UnicodeError, KeyError, ValueError, SyntaxError) as error:
        report["failure"] = f"{type(error).__name__}: {error}"
        exit_code = 2
    finally:
        report["temporaryCopiesDeleted"] = temp_root is None or not temp_root.exists()
        unchanged = {name: (root / INPUTS[name]).is_file()
                     and (root / INPUTS[name]).read_bytes() == data for name, data in snapshots.items()}
        report["inputsUnchanged"] = unchanged
        report["officialTestUnchanged"] = unchanged["test"]
        report["ssotUnchanged"] = unchanged["ssot"]
        if not all(unchanged.values()):
            report["failure"] = "official inputs changed during the run; these receipts are not a frozen-version result"
            exit_code = 1
        report["runnerExit"] = exit_code
        report["status"] = "passed" if exit_code == 0 else ("failed" if exit_code == 1 else "blocked")
        (out / "mutation-results.json").write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"status": report["status"], "runnerExit": exit_code,
                      "baselineTests": report["baseline"]["testsRun"] if report["baseline"] else None,
                      "mutants": [{"guard": result["guard"], "exit": result["exit"],
                                   "testsRun": result["testsRun"], "failures": len(result["failedCases"]),
                                   "errors": result["errors"]} for result in report["mutants"]],
                      "result": str(out / "mutation-results.json"), "failure": report.get("failure")}, indent=2))
    return exit_code


if __name__ == "__main__":
    raise SystemExit(main())
